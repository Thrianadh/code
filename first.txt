def process_pdf(self, s3_key: str) -> None:
    """
    Process a single PDF file from S3.
    Extracts content using Textract, generates Markdown and text outputs,
    and saves them to the correct S3 path.
    """
    logger.info(f"Processing PDF: {s3_key}")

    # Step 1: Start Textract job
    job_id = self.textract_manager.start_document_analysis(s3_key)
    if not self.textract_manager.wait_for_job_completion(job_id):
        logger.error(f"Textract job {job_id} failed or timed out for {s3_key}")
        return

    # Step 2: Retrieve and parse results
    results = self.textract_manager.get_job_results(job_id)
    blocks = results.get("Blocks", [])
    plain_text, markdown = TextractManager.parse_blocks(blocks)

    # Step 3: Generate output filenames
    base_name = os.path.splitext(os.path.basename(s3_key))[0]

    # ✅ FIX: Use the full textract_output path (remove s3://bucket_name/ prefix)
    textract_output_path = Config.S3_CONFIG["textract_output"].replace(
        f"s3://{Config.S3_CONFIG['bucket_name']}/", ""
    )
    output_key_base = f"{textract_output_path}{base_name}"

    # Step 4: Save outputs to S3
    with tempfile.NamedTemporaryFile(mode="w+", suffix=".txt", delete=False) as txt_file:
        txt_file.write(plain_text)
        txt_file_path = txt_file.name

    with tempfile.NamedTemporaryFile(mode="w+", suffix=".md", delete=False) as md_file:
        md_file.write(markdown)
        md_file_path = md_file.name

    # Upload to S3 with the correct full path
    self.s3_manager.upload_file(txt_file_path, f"{output_key_base}.txt")
    self.s3_manager.upload_file(md_file_path, f"{output_key_base}.md")

    # Clean up temp files
    os.unlink(txt_file_path)
    os.unlink(md_file_path)

    logger.info(f"Successfully processed and saved outputs to {output_key_base}")

    def wait_for_job_completion(self, job_id: str, max_retries: int = 30, delay: int = 5) -> bool:
        """
        Poll Textract job status until completion.

        Args:
            job_id: Textract job ID.
            max_retries: Maximum number of retries.
            delay: Delay between retries in seconds.

        Returns:
            bool: True if job succeeded, False otherwise.
        """
        for _ in range(max_retries):
            status = self.get_job_status(job_id)
            if status == "SUCCEEDED":
                logger.info(f"Textract job {job_id} succeeded.")
                return True
            elif status == "FAILED":
                logger.error(f"Textract job {job_id} failed.")
                return False
            logger.info(f"Job {job_id} status: {status}. Retrying in {delay} seconds...")
            time.sleep(delay)
        logger.error(f"Textract job {job_id} timed out after {max_retries * delay} seconds.")
        return False

        Args:
            job_id: Textract job ID.

            raise

    @staticmethod
    def parse_blocks(blocks: List[Dict[str, Any]]) -> Tuple[str, str]:
        """
        Parse Textract blocks into plain text and Markdown.
        Preserves reading order, headings, tables, and lists.

        Args:
            blocks: List of Textract blocks.

        Returns:
            Tuple[str, str]: (plain_text, markdown_text)
        """
        plain_text = []
        markdown_text = []

        # Sort blocks by their position (top to bottom, left to right)
        sorted_blocks = sorted(
            blocks,
            key=lambda x: (
                x.get("Geometry", {}).get("BoundingBox", {}).get("Top", 0),
                x.get("Geometry", {}).get("BoundingBox", {}).get("Left", 0),
            ),
        )

        for block in sorted_blocks:
            block_type = block["BlockType"]
            text = block.get("Text", "")

            if block_type == "LINE" or block_type == "WORD":
                plain_text.append(text)
                markdown_text.append(text)
            elif block_type == "TABLE":
                # Placeholder: Extract table as Markdown (simplified)
                table_md = TextractManager._parse_table(block, sorted_blocks)
                plain_text.append("[TABLE]")
                markdown_text.append(table_md)
            elif block_type == "KEY_VALUE_SET":
                # Placeholder: Format key-value pairs
                key = block.get("Key", {}).get("Text", "")
                value = block.get("Value", {}).get("Text", "")
                plain_text.append(f"{key}: {value}")
                markdown_text.append(f"**{key}**: {value}")
            elif block_type == "SELECTION_ELEMENT":
                # Placeholder: Format selection elements (e.g., checkboxes)
                if block.get("SelectionStatus") == "SELECTED":
                    plain_text.append(f"[x] {text}")
                    markdown_text.append(f"- [x] {text}")
                else:
                    plain_text.append(f"[ ] {text}")
                    markdown_text.append(f"- [ ] {text}")

        return "\n".join(plain_text), "\n".join(markdown_text)

    @staticmethod
    def _parse_table(table_block: Dict[str, Any], all_blocks: List[Dict[str, Any]]) -> str:
        """
        Parse a Textract TABLE block into Markdown.

        Args:
            table_block: Textract TABLE block.
            all_blocks: All blocks in the document (for resolving relationships).

        Returns:
            str: Markdown-formatted table.
        """
        # Find all cells in the table
        cells = []
        for rel in table_block.get("Relationships", []):
            if rel["Type"] == "CHILD":
                for cell_id in rel["Ids"]:
                    cell = next((b for b in all_blocks if b["Id"] == cell_id), None)
                    if cell:
                        cells.append(cell)

        # Sort cells by row and column indices
        cells.sort(key=lambda x: (x.get("RowIndex", 0), x.get("ColumnIndex", 0)))

        # Extract text from each cell
        rows = []
        current_row = []
        prev_row_index = None
        for cell in cells:
            row_index = cell.get("RowIndex", 0)
            if row_index != prev_row_index and current_row:
                rows.append(current_row)
                current_row = []
            prev_row_index = row_index

            # Get cell text (from child blocks)
            cell_text = []
            for rel in cell.get("Relationships", []):
                if rel["Type"] == "CHILD":
                    for content_id in rel["Ids"]:
                        content_block = next(
                            (b for b in all_blocks if b["Id"] == content_id), None
                        )
                        if content_block and content_block["BlockType"] in ["LINE", "WORD"]:
                            cell_text.append(content_block.get("Text", ""))
            current_row.append(" | ".join(cell_text))
        if current_row:
            rows.append(current_row)

        # Format as Markdown table
        if not rows:
            return ""

        markdown_table = []
        markdown_table.append("| " + " | ".join(rows[0]) + " |")
        markdown_table.append("|" + "|".join(["---"] * len(rows[0])) + "|")
        for row in rows[1:]:
            markdown_table.append("| " + " | ".join(row) + " |")

        return "\n".join(markdown_table)

        self.textract_manager = TextractManager()

    def process_pdf(self, s3_key: str) -> None:
        """
        Process a single PDF file from S3.
        Extracts content using Textract, generates Markdown and text outputs,
        and saves them to S3.

        Args:
            s3_key: S3 key of the PDF file.
        """
        logger.info(f"Processing PDF: {s3_key}")

        # Step 1: Start Textract job
        job_id = self.textract_manager.start_document_analysis(s3_key)
        if not self.textract_manager.wait_for_job_completion(job_id):
            logger.error(f"Textract job {job_id} failed or timed out for {s3_key}")
            return

        # Step 2: Retrieve and parse results
        results = self.textract_manager.get_job_results(job_id)
        blocks = results.get("Blocks", [])
        plain_text, markdown = TextractManager.parse_blocks(blocks)

        # Step 3: Generate output filenames
        base_name = os.path.splitext(os.path.basename(s3_key))[0]
        textract_output_prefix = Config.S3_CONFIG["textract_output"].split("s3://")[1].split("/")[1]
        output_prefix = f"{textract_output_prefix}/{base_name}"

        # Step 4: Save outputs to S3
        with tempfile.NamedTemporaryFile(mode="w+", suffix=".txt", delete=False) as txt_file:
            txt_file.write(plain_text)
            txt_file_path = txt_file.name

        with tempfile.NamedTemporaryFile(mode="w+", suffix=".md", delete=False) as md_file:
            md_file.write(markdown)
            md_file_path = md_file.name

        # Upload to S3
        self.s3_manager.upload_file(txt_file_path, f"{output_prefix}.txt")
        self.s3_manager.upload_file(md_file_path, f"{output_prefix}.md")

        # Clean up temp files
        os.unlink(txt_file_path)
        os.unlink(md_file_path)

        logger.info(f"Successfully processed and saved outputs for {s3_key}")

    def process_all_pdfs(self) -> None:
        """Process all PDFs in the source S3 prefix."""
        source_prefix = Config.S3_CONFIG["source_prefix"]
        pdf_keys = [
            key for key in self.s3_manager.list_objects(source_prefix)
            if key.lower().endswith(".pdf")
        ]
        logger.info(f"Found {len(pdf_keys)} PDFs to process in {source_prefix}")

        for s3_key in pdf_keys:
            self.process_pdf(s3_key)

def lambda_handler(event, context):
    """
    AWS Lambda handler for the Textract RAG pipeline.
    Processes PDFs from S3 and saves outputs to another S3 location.
    """
    try:
        logger.info("Lambda function started")
        pipeline = TextractPipeline()
        pipeline.process_all_pdfs()
        logger.info("Lambda function completed successfully")
        return {
            "statusCode": 200,
            "body": "PDF processing completed successfully",
        }
    except Exception as e:
        logger.error(f"Lambda function failed: {e}", exc_info=True)
        return {
            "statusCode": 500,
            "body": f"Error: {str(e)}",
        }
