"""
Step 9 -- answer generation with Nova Lite via the Bedrock Converse API,
using chain-of-thought reasoning, grounded in Step 8's fused chunks.

Nova is asked for exactly two things: an answer, and which single chunk_id
it drew that answer from. It is NOT asked to write the citation's "source"
or "excerpt" -- those are looked up and sliced directly from the actual
chunk text in code, after the model responds. This keeps the citation
grounded to what the source document literally says: the model can only
choose *which* chunk to point at, it cannot phrase what the chunk contains
into the citation itself.

Chain-of-thought: Nova is asked to reason step by step in a REASONING
section first (checking which chunk(s), if any, actually support an
answer) before committing to a FINAL_JSON line. The reasoning is logged
for auditing but is not part of the returned answer -- only FINAL_JSON is
parsed and returned to the caller.

If no chunk supports an answer, Nova is instructed to say so explicitly
rather than guess, the citation is left as null, AND the answer text
itself is overridden with a fixed refusal message -- Step 10's hallucination
control doesn't just leave a missing citation sitting next to whatever
Nova wrote; a claim with no supporting chunk is replaced outright.

Final return shape (this is the contract the whole chunking/retrieval
design was built backwards from):
    {
      "answer": "...",
      "citation": {"chunk_id": "...", "source": "...", "excerpt": "..."} | null
    }

Usage (standalone, for testing):
    python generation_pipeline.py --query "what is the deadline to respond to this notice?"
"""

import argparse
import json
import logging
import re

import boto3

from config import (
    AWS_REGION,
    NOVA_MODEL_ID,
    GENERATION_TEMPERATURE,
    GENERATION_MAX_TOKENS,
    EXCERPT_CHAR_LENGTH,
)
from retrieval_pipeline import HybridRetriever
from reranking_pipeline import rerank

logging.basicConfig(level=logging.INFO, format="%(asctime)s %(levelname)s %(message)s", datefmt="%H:%M:%S")
log = logging.getLogger(__name__)

SYSTEM_PROMPT = """You are ECMC's document assistant. Answer questions using ONLY the \
CONTEXT chunks provided in the user message. Do not use any outside knowledge, \
and do not guess.

Respond in exactly this format, with these two labeled sections and nothing \
else before, between, or after them:

REASONING:
Think step by step. Identify which context chunk(s), if any, are relevant to \
the question. Check whether a relevant chunk actually answers the question, \
or only partially. If no chunk answers the question, say so here.

FINAL_JSON:
A single-line JSON object, exactly this shape:
{"answer": "<your answer in plain language, or a statement that the documents do not contain this information>", "chunk_id": "<the chunk_id of the ONE chunk that most directly supports your answer, or null if none do>"}
"""

FINAL_JSON_MARKER = "FINAL_JSON:"


def build_context_block(chunks):
    parts = []
    for chunk in chunks:
        parts.append(
            f"[chunk_id: {chunk['chunk_id']} | source: {chunk['source']} | "
            f"page: {chunk.get('page')} | type: {chunk.get('element_type')}]\n"
            f"{chunk['text']}"
        )
    return "\n\n".join(parts)


def parse_model_response(raw_text):
    """Splits the REASONING / FINAL_JSON response, logs the reasoning, and
    parses the JSON object out of the FINAL_JSON section."""
    if FINAL_JSON_MARKER in raw_text:
        reasoning_part, _, json_part = raw_text.partition(FINAL_JSON_MARKER)
    else:
        # Model didn't follow the format -- fall back to scanning the whole
        # response for a JSON object rather than failing outright.
        reasoning_part, json_part = "", raw_text

    log.info("Model reasoning: %s", reasoning_part.strip().replace("REASONING:", "").strip())

    match = re.search(r"\{.*\}", json_part, re.DOTALL)
    if not match:
        raise ValueError(f"No JSON object found in model response: {raw_text!r}")

    return json.loads(match.group(0))


def build_citation(chunk_id, chunks):
    if not chunk_id:
        return None
    for chunk in chunks:
        if chunk["chunk_id"] == chunk_id:
            text = chunk["text"]
            excerpt = text[:EXCERPT_CHAR_LENGTH] + ("..." if len(text) > EXCERPT_CHAR_LENGTH else "")
            return {
                "chunk_id": chunk["chunk_id"],
                "source": chunk["source"],
                "excerpt": excerpt,
            }
    # Model named a chunk_id that wasn't actually in the context we gave it --
    # treat that as no citation rather than fabricating one.
    log.error("Model returned chunk_id %r not present in the provided chunks", chunk_id)
    return None


def generate_answer(bedrock_client, retriever, query):
    fused_chunks = rerank(bedrock_client, retriever, query)

    if not fused_chunks:
        return {
            "answer": "The provided documents do not contain information to answer this question.",
            "citation": None,
        }

    context_block = build_context_block(fused_chunks)
    user_message = f"CONTEXT:\n{context_block}\n\nQUESTION:\n{query}"

    response = bedrock_client.converse(
        modelId=NOVA_MODEL_ID,
        system=[{"text": SYSTEM_PROMPT}],
        messages=[{"role": "user", "content": [{"text": user_message}]}],
        inferenceConfig={
            "temperature": GENERATION_TEMPERATURE,
            "maxTokens": GENERATION_MAX_TOKENS,
        },
    )

    raw_text = response["output"]["message"]["content"][0]["text"]
    parsed = parse_model_response(raw_text)

    answer = parsed.get("answer", "").strip()
    citation = build_citation(parsed.get("chunk_id"), fused_chunks)

    # Step 10, hallucination control: if no chunk actually backs the answer,
    # the answer text itself is replaced -- not just left to accompany a
    # missing citation. Nova can still have written a speculative answer
    # even after saying chunk_id: null; an uncited claim in a bankruptcy/
    # financial context is exactly the failure mode this pipeline exists to
    # avoid, so it isn't passed through as-is.
    if citation is None:
        answer = "The provided documents do not contain information to answer this question."

    return {"answer": answer, "citation": citation}


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--query", required=True, help="Question to answer")
    args = parser.parse_args()

    session = boto3.Session(region_name=AWS_REGION)
    s3_client = session.client("s3")
    bedrock_client = session.client("bedrock-runtime")

    retriever = HybridRetriever(s3_client)
    result = generate_answer(bedrock_client, retriever, args.query)

    print(json.dumps(result, indent=2, default=str))


if __name__ == "__main__":
    main()

"""
Runs a fixed list of test questions through the full RAG pipeline
(retrieval -> RRF fusion -> Nova generation, Steps 7-9) and records each
result and its latency into an .xlsx file. This is a manual test harness
for exactly this kind of question -- "does the pipeline answer correctly
and cite the right chunk, and how long does it take" -- run across a
batch, rather than one at a time via generation_pipeline.py's CLI.

Input: a plain text file, one question per line (default: questions.txt
next to this script). Pass --questions-file to point at a different file.

Output: an .xlsx file with two sheets:
    - Results  -> one row per question: question, answer, chunk_id,
                  source, excerpt, has_citation, latency_seconds
    - Summary  -> total questions, how many got a citation vs. were
                  refused (Step 10's hallucination control), and
                  average/min/max latency -- all as formulas over the
                  Results sheet, so they recalculate if a row is edited.

Install:
    pip install boto3 python-dotenv openpyxl

Usage:
    python eval_pipeline.py --questions-file questions.txt --output eval_results.xlsx
"""

import argparse
import logging
import time

import boto3
from openpyxl import Workbook
from openpyxl.styles import Font, Alignment

from config import AWS_REGION
from retrieval_pipeline import HybridRetriever
from generation_pipeline import generate_answer

logging.basicConfig(level=logging.INFO, format="%(asctime)s %(levelname)s %(message)s", datefmt="%H:%M:%S")
log = logging.getLogger(__name__)

FONT_NAME = "Arial"
HEADER_FONT = Font(name=FONT_NAME, bold=True)
BODY_FONT = Font(name=FONT_NAME)


def load_questions(path):
    questions = []
    with open(path, "r", encoding="utf-8") as f:
        for line in f:
            line = line.strip()
            if line:
                questions.append(line)
    return questions


def run_questions(bedrock_client, retriever, questions):
    """Calls Step 9's generate_answer() once per question, timing each call
    individually. A failure on one question is recorded as an error row
    and does not stop the rest of the batch."""
    rows = []
    for i, question in enumerate(questions, start=1):
        log.info("[%d/%d] %s", i, len(questions), question)
        start = time.perf_counter()
        error = None
        try:
            result = generate_answer(bedrock_client, retriever, question)
        except Exception as e:
            log.error("Question %d failed: %s", i, e)
            result = {"answer": None, "citation": None}
            error = str(e)
        elapsed = time.perf_counter() - start

        citation = result.get("citation")
        rows.append({
            "question": question,
            "answer": f"ERROR: {error}" if error else result.get("answer"),
            "chunk_id": citation["chunk_id"] if citation else "",
            "source": citation["source"] if citation else "",
            "excerpt": citation["excerpt"] if citation else "",
            "has_citation": "Yes" if citation else "No",
            "latency_seconds": round(elapsed, 3),
        })
    return rows


def write_xlsx(rows, output_path):
    wb = Workbook()
    ws = wb.active
    ws.title = "Results"

    headers = ["#", "Question", "Answer", "Chunk ID", "Source", "Excerpt", "Has Citation", "Latency (s)"]
    ws.append(headers)
    for cell in ws[1]:
        cell.font = HEADER_FONT

    for i, row in enumerate(rows, start=1):
        ws.append([
            i, row["question"], row["answer"], row["chunk_id"],
            row["source"], row["excerpt"], row["has_citation"], row["latency_seconds"],
        ])

    for row in ws.iter_rows(min_row=2):
        for cell in row:
            cell.font = BODY_FONT
            cell.alignment = Alignment(wrap_text=True, vertical="top")

    widths = {"A": 4, "B": 40, "C": 45, "D": 22, "E": 20, "F": 45, "G": 12, "H": 12}
    for col, width in widths.items():
        ws.column_dimensions[col].width = width

    last_row = 1 + len(rows)

    summary = wb.create_sheet("Summary")
    summary_rows = [
        ("Metric", "Value"),
        ("Total questions", f"=COUNTA(Results!B2:B{last_row})"),
        ("Answered with citation", f'=COUNTIF(Results!G2:G{last_row},"Yes")'),
        ("Refused (no citation)", f'=COUNTIF(Results!G2:G{last_row},"No")'),
        ("Average latency (s)", f"=AVERAGE(Results!H2:H{last_row})"),
        ("Min latency (s)", f"=MIN(Results!H2:H{last_row})"),
        ("Max latency (s)", f"=MAX(Results!H2:H{last_row})"),
    ]
    for r in summary_rows:
        summary.append(r)
    for cell in summary[1]:
        cell.font = HEADER_FONT
    for row in summary.iter_rows(min_row=2):
        for cell in row:
            cell.font = BODY_FONT
    summary.column_dimensions["A"].width = 24
    summary.column_dimensions["B"].width = 16

    wb.save(output_path)


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--questions-file", default="questions.txt")
    parser.add_argument("--output", default="eval_results.xlsx")
    args = parser.parse_args()

    questions = load_questions(args.questions_file)
    if not questions:
        log.error("No questions found in %s", args.questions_file)
        return

    log.info("Loaded %d questions", len(questions))

    session = boto3.Session(region_name=AWS_REGION)
    s3_client = session.client("s3")
    bedrock_client = session.client("bedrock-runtime")

    retriever = HybridRetriever(s3_client)
    rows = run_questions(bedrock_client, retriever, questions)

    write_xlsx(rows, args.output)
    log.info("Wrote %s (%d rows)", args.output, len(rows))


if __name__ == "__main__":
    main()
