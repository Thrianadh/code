"""Environment configuration loader.

Merges .env secrets (region, bucket, model IDs) with paths.py constants.
Returns a single Settings dataclass that every module imports.
"""

import os
from dataclasses import dataclass
from typing import Optional

from dotenv import load_dotenv

from .paths import S3Paths, ChunkingParams, HybridSearchParams, TextractParams, NovaPrompts
from .paths import S3_PATHS, CHUNKING, HYBRID, TEXTRACT, PROMPTS


@dataclass(frozen=True)
class Settings:
    """Single source of truth for all configuration.

    Attributes:
        AWS_REGION: AWS region for all boto3 clients.
        S3_BUCKET: Source-of-truth S3 bucket name.
        BEDROCK_CHAT_MODEL_ID: Nova Lite model ID for generation.
        BEDROCK_EMBED_MODEL_ID: Titan Embed model ID for vectorization.
        s3_paths: Immutable S3 path definitions.
        chunking: Parent-child chunking parameters.
        hybrid: Hybrid search (RRF) parameters.
        textract: Textract API configuration.
        prompts: Nova prompt templates.
    """

    AWS_REGION: str
    S3_BUCKET: str
    BEDROCK_CHAT_MODEL_ID: str
    BEDROCK_EMBED_MODEL_ID: str

    s3_paths: S3Paths = S3_PATHS
    chunking: ChunkingParams = CHUNKING
    hybrid: HybridSearchParams = HYBRID
    textract: TextractParams = TEXTRACT
    prompts: NovaPrompts = PROMPTS


# Module-level singleton cache
_SETTINGS_CACHE: Optional[Settings] = None


def load_settings(dotenv_path: Optional[str] = None) -> Settings:
    """Load .env and merge with path constants.

    Args:
        dotenv_path: Optional path to .env file. If None, searches cwd upward.

    Returns:
        Settings dataclass with all configuration merged.

    Raises:
        ValueError: If required .env variables are missing.
    """
    global _SETTINGS_CACHE
    if _SETTINGS_CACHE is not None:
        return _SETTINGS_CACHE

    load_dotenv(dotenv_path=dotenv_path)

    region = os.getenv("AWS_REGION", "us-east-1")
    bucket = os.getenv("S3_BUCKET", "bankrupcy-data-bkt")
    chat_model = os.getenv("BEDROCK_CHAT_MODEL_ID", "us.amazon.nova-2-lite-v1:0")
    embed_model = os.getenv("BEDROCK_EMBED_MODEL_ID", "amazon.titan-embed-text-v2:0")

    # Validate required vars
    missing = []
    if not region:
        missing.append("AWS_REGION")
    if not bucket:
        missing.append("S3_BUCKET")
    if not chat_model:
        missing.append("BEDROCK_CHAT_MODEL_ID")
    if not embed_model:
        missing.append("BEDROCK_EMBED_MODEL_ID")

    if missing:
        raise ValueError(
            f"Missing required environment variables: {', '.join(missing)}. "
            f"Please set them in .env file."
        )

    _SETTINGS_CACHE = Settings(
        AWS_REGION=region,
        S3_BUCKET=bucket,
        BEDROCK_CHAT_MODEL_ID=chat_model,
        BEDROCK_EMBED_MODEL_ID=embed_model,
    )
    return _SETTINGS_CACHE


def clear_settings_cache() -> None:
    """Clear the cached Settings instance. Useful for testing."""
    global _SETTINGS_CACHE
    _SETTINGS_CACHE = None

######aws_clients.py
"""Singleton boto3 clients with retry configuration.

All modules import clients from here. Never create boto3.client() directly.
This ensures consistent region, retry policy, and connection pooling.
"""

import boto3
from botocore.config import Config
from typing import Optional

from config.settings import load_settings

# Module-level singleton cache
_bedrock_runtime: Optional[boto3.client] = None
_textract: Optional[boto3.client] = None
_s3: Optional[boto3.client] = None


def _get_region() -> str:
    """Fetch region from Settings singleton."""
    return load_settings().AWS_REGION


def get_bedrock_runtime_client() -> boto3.client:
    """Return singleton Bedrock Runtime client.

    Used for: Converse API (Nova Lite), InvokeModel (Titan Embed).
    Retry mode: adaptive (handles throttling automatically).
    """
    global _bedrock_runtime
    if _bedrock_runtime is None:
        _bedrock_runtime = boto3.client(
            service_name="bedrock-runtime",
            region_name=_get_region(),
            config=Config(
                retries={"max_attempts": 3, "mode": "adaptive"},
                max_pool_connections=25,
            ),
        )
    return _bedrock_runtime


def get_textract_client() -> boto3.client:
    """Return singleton Textract client.

    Used for: analyze_document, start_document_analysis, get_document_analysis.
    Retry mode: standard (Textract has long-running jobs, adaptive not needed).
    """
    global _textract
    if _textract is None:
        _textract = boto3.client(
            service_name="textract",
            region_name=_get_region(),
            config=Config(
                retries={"max_attempts": 3, "mode": "standard"},
            ),
        )
    return _textract


def get_s3_client() -> boto3.client:
    """Return singleton S3 client.

    Used for: all upload/download/list operations.
    No transfer acceleration (cost control).
    """
    global _s3
    if _s3 is None:
        _s3 = boto3.client(
            service_name="s3",
            region_name=_get_region(),
            config=Config(
                retries={"max_attempts": 3, "mode": "standard"},
                max_pool_connections=50,
            ),
        )
    return _s3


def clear_client_cache() -> None:
    """Clear all cached boto3 clients. Useful for testing or region changes."""
    global _bedrock_runtime, _textract, _s3
    _bedrock_runtime = None
    _textract = None
    _s3 = None

##s3_utils.py
"""All S3 read/write operations with centralized error handling.

Every S3 interaction in the pipeline goes through this module.
Provides format-aware upload/download (JSON, text, markdown, bytes)
and utility functions (list, exists, prefix normalization).
"""

import json
import logging
from io import BytesIO
from typing import Any, Dict, List, Optional

import boto3
from botocore.exceptions import ClientError

from .aws_clients import get_s3_client
from .logging_config import get_logger

logger = get_logger("s3_utils")


# ── Upload Operations ──────────────────────────────────────────────────────

def upload_json(bucket: str, key: str, data: Dict[str, Any]) -> None:
    """Serialize dict to JSON bytes and upload to S3.

    Args:
        bucket: S3 bucket name.
        key: S3 object key (should end with .json).
        data: Python dictionary to serialize.

    Raises:
        ClientError: On S3 upload failure.
    """
    body = json.dumps(data, indent=2, ensure_ascii=False).encode("utf-8")
    upload_bytes(bucket, key, body, content_type="application/json")
    logger.info("Uploaded JSON", extra={"bucket": bucket, "key": key, "bytes": len(body)})


def upload_text(bucket: str, key: str, text: str) -> None:
    """Upload plain text string to S3.

    Args:
        bucket: S3 bucket name.
        key: S3 object key (should end with .txt).
        text: UTF-8 string content.
    """
    body = text.encode("utf-8")
    upload_bytes(bucket, key, body, content_type="text/plain; charset=utf-8")
    logger.info("Uploaded text", extra={"bucket": bucket, "key": key, "bytes": len(body)})


def upload_markdown(bucket: str, key: str, markdown: str) -> None:
    """Upload markdown string to S3 with validation.

    Args:
        bucket: S3 bucket name.
        key: S3 object key (must end with .md).
        markdown: Markdown-formatted string.

    Raises:
        ValueError: If key does not end with .md.
    """
    if not key.endswith(".md"):
        raise ValueError(f"Markdown upload requires .md extension. Got: {key}")
    body = markdown.encode("utf-8")
    upload_bytes(bucket, key, body, content_type="text/markdown; charset=utf-8")
    logger.info("Uploaded markdown", extra={"bucket": bucket, "key": key, "bytes": len(body)})


def upload_bytes(
    bucket: str,
    key: str,
    data: bytes,
    content_type: str = "application/octet-stream",
) -> None:
    """Upload raw bytes to S3.

    Args:
        bucket: S3 bucket name.
        key: S3 object key.
        data: Raw bytes payload.
        content_type: MIME type for the object.

    Raises:
        ClientError: On S3 upload failure.
    """
    s3 = get_s3_client()
    try:
        s3.put_object(
            Bucket=bucket,
            Key=key,
            Body=data,
            ContentType=content_type,
        )
    except ClientError as exc:
        logger.error(
            "S3 upload failed",
            extra={"bucket": bucket, "key": key, "error": str(exc)},
        )
        raise


# ── Download Operations ──────────────────────────────────────────────────────

def download_json(bucket: str, key: str) -> Dict[str, Any]:
    """Download S3 object and deserialize JSON.

    Args:
        bucket: S3 bucket name.
        key: S3 object key.

    Returns:
        Python dictionary. Returns empty dict if key does not exist (404).

    Raises:
        ClientError: On non-404 S3 errors (permissions, network).
    """
    try:
        body = download_bytes(bucket, key)
        return json.loads(body.decode("utf-8"))
    except ClientError as exc:
        error_code = exc.response["Error"]["Code"]
        if error_code == "NoSuchKey":
            logger.warning("S3 key not found, returning empty dict", extra={"bucket": bucket, "key": key})
            return {}
        raise


def download_text(bucket: str, key: str) -> str:
    """Download S3 object as UTF-8 string.

    Args:
        bucket: S3 bucket name.
        key: S3 object key.

    Returns:
        UTF-8 decoded string.

    Raises:
        ClientError: On S3 download failure.
    """
    body = download_bytes(bucket, key)
    return body.decode("utf-8")


def download_bytes(bucket: str, key: str) -> bytes:
    """Download S3 object as raw bytes.

    Args:
        bucket: S3 bucket name.
        key: S3 object key.

    Returns:
        Raw bytes payload.

    Raises:
        ClientError: On S3 download failure (including 404).
    """
    s3 = get_s3_client()
    try:
        response = s3.get_object(Bucket=bucket, Key=key)
        body = response["Body"].read()
        logger.info(
            "Downloaded bytes",
            extra={"bucket": bucket, "key": key, "bytes": len(body)},
        )
        return body
    except ClientError as exc:
        logger.error(
            "S3 download failed",
            extra={"bucket": bucket, "key": key, "error": str(exc)},
        )
        raise


# ── Listing & Existence Operations ─────────────────────────────────────────

def list_objects(bucket: str, prefix: str) -> List[str]:
    """List all object keys under a prefix.

    Handles pagination automatically (S3 returns max 1000 per call).

    Args:
        bucket: S3 bucket name.
        prefix: S3 key prefix (e.g., 'ecmc_org/ECMC_RAG/01_markdown/').

    Returns:
        List of full object keys.
    """
    s3 = get_s3_client()
    keys: List[str] = []
    paginator = s3.get_paginator("list_objects_v2")

    for page in paginator.paginate(Bucket=bucket, Prefix=prefix):
        for obj in page.get("Contents", []):
            keys.append(obj["Key"])

    logger.info(
        "Listed S3 objects",
        extra={"bucket": bucket, "prefix": prefix, "count": len(keys)},
    )
    return keys


def check_exists(bucket: str, key: str) -> bool:
    """Check if an S3 object exists using HeadObject.

    Args:
        bucket: S3 bucket name.
        key: S3 object key.

    Returns:
        True if object exists, False otherwise.
    """
    s3 = get_s3_client()
    try:
        s3.head_object(Bucket=bucket, Key=key)
        return True
    except ClientError as exc:
        error_code = exc.response["Error"]["Code"]
        if error_code == "404":
            return False
        logger.error(
            "S3 head_object failed",
            extra={"bucket": bucket, "key": key, "error": str(exc)},
        )
        raise


def ensure_prefix(prefix: str) -> str:
    """Ensure prefix ends with '/' for safe concatenation.

    Args:
        prefix: S3 key prefix string.

    Returns:
        Prefix with trailing slash.
    """
    return prefix if prefix.endswith("/") else prefix + "/"

"""Text processing utilities for chunking, ID generation, and table conversion.

All token counting is approximate (whitespace-based) for fast, dependency-free
chunking decisions. Exact tokenization happens inside Bedrock.
"""

import re
import uuid
from typing import Dict, List, Optional, Set, Tuple

from config.paths import CHUNKING


# ── Token Estimation ───────────────────────────────────────────────────────

def estimate_token_count(text: str) -> int:
    """Approximate token count using whitespace-based heuristic.

    Rule of thumb: 1 token ≈ 0.75 English words.
    This is fast, dependency-free, and accurate within ±15% for chunking.

    Args:
        text: Input text string.

    Returns:
        Estimated integer token count.

    Example:
        >>> estimate_token_count("Income-driven repayment plans calculate.")
        6
    """
    if not text or not text.strip():
        return 0
    # Split on whitespace to get word count
    word_count = len(text.split())
    # Apply 0.75 tokens-per-word ratio and round up
    return max(1, int(word_count * 0.75))


# ── ID Generation ──────────────────────────────────────────────────────────

def generate_doc_id(filename: str, existing_ids: Optional[Set[str]] = None) -> str:
    """Generate sequential doc_id from filename.

    Format: doc_001, doc_002, ...
    If existing_ids provided, finds next available sequence number.

    Args:
        filename: Original file name (e.g., 'Federal_Student_Aid_Guide_2024.pdf').
        existing_ids: Set of already-assigned doc_ids. If None, generates doc_001.

    Returns:
        New unique doc_id string.

    Example:
        >>> generate_doc_id("guide.pdf")
        'doc_001'
        >>> generate_doc_id("guide2.pdf", {"doc_001"})
        'doc_002'
    """
    existing_ids = existing_ids or set()
    if not existing_ids:
        return "doc_001"

    # Extract numeric suffixes from existing doc_ids
    max_num = 0
    for doc_id in existing_ids:
        match = re.match(r"doc_(\d{3})", doc_id)
        if match:
            max_num = max(max_num, int(match.group(1)))

    next_num = max_num + 1
    return f"doc_{next_num:03d}"


def generate_parent_id(doc_id: str, sequence: int) -> str:
    """Generate parent chunk ID.

    Format: {doc_id}_parent_{sequence:03d}

    Args:
        doc_id: Document identifier (e.g., 'doc_001').
        sequence: Parent sequence number within document (1-indexed).

    Returns:
        Parent chunk ID string.
    """
    return f"{doc_id}_parent_{sequence:03d}"


def generate_child_id(doc_id: str, parent_sequence: int, child_sequence: int) -> str:
    """Generate child chunk ID.

    Format: {doc_id}_child_{parent_sequence:03d}_{child_sequence:03d}
    This embeds parent_sequence for easy tracing.

    Args:
        doc_id: Document identifier.
        parent_sequence: Parent sequence number.
        child_sequence: Child sequence within parent (1-indexed).

    Returns:
        Child chunk ID string.
    """
    return f"{doc_id}_child_{parent_sequence:03d}_{child_sequence:03d}"


def generate_uuid() -> str:
    """Generate a short unique identifier for internal tracking.

    Returns:
        8-character hex string (first 8 chars of UUID4).
    """
    return uuid.uuid4().hex[:8]


# ── Table Conversion ───────────────────────────────────────────────────────

def textract_table_to_markdown(
    table_block: Dict[str, Any],
    blocks_by_id: Dict[str, Dict[str, Any]],
) -> str:
    """Convert an AWS Textract TABLE block to pipe-delimited markdown.

    Handles merged cells by filling empty cells with placeholder.
    Detects header row from WORD block confidence or first row.

    Args:
        table_block: Textract Block with BlockType="TABLE".
        blocks_by_id: Dict mapping all block IDs to their block objects.

    Returns:
        Markdown table string.

    Example Output:
        | Income Range | Family Size 1 | Family Size 2 |
        | --- | --- | --- |
        | $0 - $20,000 | $0 | $0 |
    """
    # Extract relationships: TABLE → CELL blocks
    cells = []
    for rel in table_block.get("Relationships", []):
        if rel.get("Type") == "CHILD":
            for cell_id in rel.get("Ids", []):
                cell = blocks_by_id.get(cell_id)
                if cell and cell.get("BlockType") == "CELL":
                    cells.append(cell)

    if not cells:
        return ""

    # Build grid: row_index -> {col_index -> text}
    grid: Dict[int, Dict[int, str]] = {}
    max_row = 0
    max_col = 0

    for cell in cells:
        row = cell.get("RowIndex", 1)
        col = cell.get("ColumnIndex", 1)
        # Extract text from cell's child WORD blocks
        cell_text = ""
        for rel in cell.get("Relationships", []):
            if rel.get("Type") == "CHILD":
                for word_id in rel.get("Ids", []):
                    word = blocks_by_id.get(word_id)
                    if word and word.get("BlockType") == "WORD":
                        cell_text += word.get("Text", "") + " "
        cell_text = cell_text.strip()

        if row not in grid:
            grid[row] = {}
        grid[row][col] = cell_text
        max_row = max(max_row, row)
        max_col = max(max_col, col)

    # Build markdown rows
    rows: List[str] = []
    for r in range(1, max_row + 1):
        cols = [grid.get(r, {}).get(c, "") for c in range(1, max_col + 1)]
        rows.append("| " + " | ".join(cols) + " |")

    # Insert header separator after first row (assume first row is header)
    if len(rows) >= 1:
        separator = "| " + " | ".join(["---"] * max_col) + " |"
        rows.insert(1, separator)

    return "\n".join(rows)


# ── Markdown Cleaning ──────────────────────────────────────────────────────

def clean_markdown(text: str) -> str:
    """Normalize markdown text for consistent formatting.

    Operations:
      - Collapse multiple blank lines to single blank line
      - Strip trailing whitespace from each line
      - Ensure single newline between paragraphs
      - Standardize heading levels (no space between # and text)

    Args:
        text: Raw markdown string.

    Returns:
        Cleaned markdown string.
    """
    # Normalize line endings
    text = text.replace("\r\n", "\n").replace("\r", "\n")

    # Strip trailing whitespace per line
    lines = [line.rstrip() for line in text.split("\n")]

    # Collapse multiple blank lines
    cleaned_lines = []
    prev_blank = False
    for line in lines:
        is_blank = line == ""
        if is_blank and prev_blank:
            continue
        cleaned_lines.append(line)
        prev_blank = is_blank

    # Fix heading spacing: "#  Title" → "# Title"
    result_lines = []
    for line in cleaned_lines:
        match = re.match(r"^(#{1,6})\s+(.*)", line)
        if match:
            result_lines.append(f"{match.group(1)} {match.group(2).strip()}")
        else:
            result_lines.append(line)

    return "\n".join(result_lines)


# ── Smart Truncation ───────────────────────────────────────────────────────

def truncate_at_boundary(
    text: str,
    max_tokens: int,
    boundary: str = "sentence",
) -> str:
    """Truncate text at sentence or paragraph boundary without mid-sentence cuts.

    Args:
        text: Input text to truncate.
        max_tokens: Maximum estimated tokens allowed.
        boundary: "sentence" or "paragraph". Default is sentence.

    Returns:
        Truncated text ending at a natural boundary.

    Raises:
        ValueError: If boundary is not 'sentence' or 'paragraph'.
    """
    if boundary not in ("sentence", "paragraph"):
        raise ValueError(f"boundary must be 'sentence' or 'paragraph', got {boundary}")

    if estimate_token_count(text) <= max_tokens:
        return text

    if boundary == "sentence":
        # Split on sentence endings (.!? followed by space or newline)
        sentences = re.split(r"(?<=[.!?])\s+", text)
    else:
        sentences = text.split("\n\n")

    result = []
    current_tokens = 0
    for segment in sentences:
        seg_tokens = estimate_token_count(segment)
        if current_tokens + seg_tokens > max_tokens:
            break
        result.append(segment)
        current_tokens += seg_tokens

    return " ".join(result) if boundary == "sentence" else "\n\n".join(result)


# ── Context Extraction ─────────────────────────────────────────────────────

def extract_context_around_table(
    reading_order: List[Dict[str, Any]],
    table_block_id: str,
    max_tokens: int = 200,
) -> str:
    """Extract text blocks immediately before and after a table.

    Used for table enrichment: provides Nova Lite with surrounding context
    to generate accurate captions.

    Args:
        reading_order: List of blocks sorted by reading order.
        table_block_id: The block ID of the TABLE block.
        max_tokens: Maximum tokens of context to extract (split before/after).

    Returns:
        Concatenated surrounding text.
    """
    # Find table position in reading order
    table_idx = -1
    for i, block in enumerate(reading_order):
        if block.get("block_id") == table_block_id:
            table_idx = i
            break

    if table_idx == -1:
        return ""

    half_tokens = max_tokens // 2
    before_text = []
    after_text = []

    # Collect text before table (reverse order, stop when token limit reached)
    before_tokens = 0
    for i in range(table_idx - 1, -1, -1):
        block = reading_order[i]
        if block.get("block_type") in ("LAYOUT_TEXT", "LAYOUT_SECTION_HEADER"):
            text = block.get("text", "")
            tokens = estimate_token_count(text)
            if before_tokens + tokens > half_tokens:
                break
            before_text.insert(0, text)
            before_tokens += tokens

    # Collect text after table
    after_tokens = 0
    for i in range(table_idx + 1, len(reading_order)):
        block = reading_order[i]
        if block.get("block_type") in ("LAYOUT_TEXT", "LAYOUT_SECTION_HEADER"):
            text = block.get("text", "")
            tokens = estimate_token_count(text)
            if after_tokens + tokens > half_tokens:
                break
            after_text.append(text)
            after_tokens += tokens

    return " ".join(before_text + after_text)

"""Structured JSON logging for production observability.

All modules use get_logger() to emit JSON-formatted logs.
CloudWatch automatically parses these for metric filters and dashboards.
"""

import json
import logging
import sys
from datetime import datetime, timezone
from typing import Any, Dict, Optional


class JSONFormatter(logging.Formatter):
    """Custom formatter that outputs log records as JSON lines.

    Format:
        {
          "timestamp": "2026-08-01T10:30:00Z",
          "level": "INFO",
          "module": "textract_processor",
          "function": "analyze_document",
          "message": "Job started",
          "doc_id": "doc_001",
          "duration_ms": 4500,
          "status": "success"
        }
    """

    def format(self, record: logging.LogRecord) -> str:
        log_entry: Dict[str, Any] = {
            "timestamp": datetime.now(timezone.utc).isoformat().replace("+00:00", "Z"),
            "level": record.levelname,
            "module": record.module,
            "function": record.funcName,
            "message": record.getMessage(),
        }

        # Merge extra fields from record.__dict__ (passed via extra=)
        # Filter out standard LogRecord attributes to avoid noise
        standard_attrs = {
            "name", "msg", "args", "levelname", "levelno", "pathname",
            "filename", "module", "exc_info", "exc_text", "stack_info",
            "lineno", "funcName", "created", "msecs", "relativeCreated",
            "thread", "threadName", "processName", "process", "message",
            "asctime", "taskName",
        }
        for key, value in record.__dict__.items():
            if key not in standard_attrs and not key.startswith("_"):
                log_entry[key] = value

        # Include exception info if present
        if record.exc_info:
            log_entry["exception"] = self.formatException(record.exc_info)

        return json.dumps(log_entry, default=str)


def get_logger(name: str) -> logging.Logger:
    """Get a configured logger with JSON formatting.

    Args:
        name: Logger name (typically module name).

    Returns:
        Configured logging.Logger instance.
    """
    logger = logging.getLogger(name)

    # Avoid duplicate handlers if called multiple times
    if logger.handlers:
        return logger

    logger.setLevel(logging.INFO)

    # Stream handler → stdout (CloudWatch captures stdout in ECS/Lambda)
    handler = logging.StreamHandler(sys.stdout)
    handler.setFormatter(JSONFormatter())
    logger.addHandler(handler)

    # Prevent propagation to root logger (avoids duplicate output)
    logger.propagate = False

    return logger


def log_operation(
    logger: logging.Logger,
    operation: str,
    doc_id: Optional[str] = None,
    duration_ms: Optional[float] = None,
    status: Optional[str] = None,
    extra: Optional[Dict[str, Any]] = None,
) -> None:
    """Standardized operation log entry.

    Args:
        logger: Logger instance from get_logger().
        operation: Name of the operation (e.g., 'analyze_document').
        doc_id: Document identifier if applicable.
        duration_ms: Operation duration in milliseconds.
        status: 'success', 'failure', 'started', etc.
        extra: Additional key-value pairs to include in log.
    """
    log_data = {
        "operation": operation,
    }
    if doc_id:
        log_data["doc_id"] = doc_id
    if duration_ms is not None:
        log_data["duration_ms"] = round(duration_ms, 2)
    if status:
        log_data["status"] = status
    if extra:
        log_data.update(extra)

    logger.info(f"Operation: {operation}", extra=log_data)

"""Cross-cutting decorators for retry logic, timing, and input validation.

Apply these to pipeline functions to get automatic retries, performance logging,
and schema validation without cluttering business logic.
"""

import functools
import time
from typing import Any, Callable, Dict, Tuple, Type, Union

from .logging_config import get_logger

logger = get_logger("decorators")


# Retry Decorator 
def retry(
    max_attempts: int = 3,
    backoff_seconds: int = 2,
    exceptions: Tuple[Type[Exception], ...] = (Exception,),
) -> Callable:
    """Exponential backoff retry decorator.

    Retries the decorated function up to max_attempts with exponential backoff.
    Logs each retry attempt.

    Args:
        max_attempts: Maximum number of attempts (including first call).
        backoff_seconds: Initial backoff in seconds. Doubles each retry.
        exceptions: Tuple of exception types to catch and retry.

    Returns:
        Decorated function.

    Example:
        @retry(max_attempts=3, backoff_seconds=2, exceptions=(ClientError,))
        def call_bedrock_api(...):
            ...
    """
    def decorator(func: Callable) -> Callable:
        @functools.wraps(func)
        def wrapper(*args: Any, **kwargs: Any) -> Any:
            attempt = 1
            current_backoff = backoff_seconds

            while attempt <= max_attempts:
                try:
                    return func(*args, **kwargs)
                except exceptions as exc:
                    if attempt == max_attempts:
                        logger.error(
                            f"Function {func.__name__} failed after {max_attempts} attempts",
                            extra={
                                "function": func.__name__,
                                "attempt": attempt,
                                "max_attempts": max_attempts,
                                "error": str(exc),
                                "status": "failure",
                            },
                        )
                        raise

                    logger.warning(
                        f"Function {func.__name__} failed, retrying in {current_backoff}s",
                        extra={
                            "function": func.__name__,
                            "attempt": attempt,
                            "max_attempts": max_attempts,
                            "backoff_seconds": current_backoff,
                            "error": str(exc),
                            "status": "retry",
                        },
                    )
                    time.sleep(current_backoff)
                    current_backoff *= 2
                    attempt += 1

            # Should never reach here
            raise RuntimeError(f"Unexpected exit from retry loop in {func.__name__}")

        return wrapper
    return decorator


# ── Timing Decorator ───────────────────────────────────────────────────────

def timing(logger_name: str = "performance") -> Callable:
    """Log function execution duration in milliseconds.

    Args:
        logger_name: Logger name for timing logs. Default 'performance'.

    Returns:
        Decorated function.

    Example:
        @timing()
        def process_document(doc_id: str):
            ...
    """
    perf_logger = get_logger(logger_name)

    def decorator(func: Callable) -> Callable:
        @functools.wraps(func)
        def wrapper(*args: Any, **kwargs: Any) -> Any:
            start = time.perf_counter()
            try:
                result = func(*args, **kwargs)
                duration_ms = (time.perf_counter() - start) * 1000
                perf_logger.info(
                    f"Function {func.__name__} completed",
                    extra={
                        "function": func.__name__,
                        "duration_ms": round(duration_ms, 2),
                        "status": "success",
                    },
                )
                return result
            except Exception as exc:
                duration_ms = (time.perf_counter() - start) * 1000
                perf_logger.error(
                    f"Function {func.__name__} failed",
                    extra={
                        "function": func.__name__,
                        "duration_ms": round(duration_ms, 2),
                        "status": "failure",
                        "error": str(exc),
                    },
                )
                raise

        return wrapper
    return decorator


# ── Input Validation Decorator ────────────────────────────────────────────

def validate_input(schema: Dict[str, Union[Type, Tuple[Type, ...]]]) -> Callable:
    """Validate function keyword arguments against expected types.

    Args:
        schema: Dict mapping arg name to expected type(s).
                Example: {"doc_id": str, "s3_key": str, "max_pages": (int, type(None))}

    Returns:
        Decorated function.

    Raises:
        TypeError: If argument type does not match schema.
        ValueError: If required argument (no None in types) is missing.

    Example:
        @validate_input({"doc_id": str, "bucket": str})
        def process_document(doc_id: str, bucket: str):
            ...
    """
    def decorator(func: Callable) -> Callable:
        @functools.wraps(func)
        def wrapper(*args: Any, **kwargs: Any) -> Any:
            import inspect
            sig = inspect.signature(func)
            bound = sig.bind(*args, **kwargs)
            bound.apply_defaults()

            for arg_name, expected_types in schema.items():
                if isinstance(expected_types, type):
                    expected_types = (expected_types,)

                value = bound.arguments.get(arg_name)
                if value is None and type(None) not in expected_types:
                    raise ValueError(
                        f"Missing required argument '{arg_name}' in {func.__name__}"
                    )

                if value is not None and not isinstance(value, expected_types):
                    raise TypeError(
                        f"Argument '{arg_name}' in {func.__name__} "
                        f"expected {expected_types}, got {type(value).__name__}"
                    )

            return func(*args, **kwargs)

        return wrapper
    return decorator
