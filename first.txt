"""
Terminal-based chat interface for the RAG ChatBot.
Displays conversation like a real chatbot in the terminal.

Commands:
  /new      — Start a new chat (clear history, keep session)
  /history  — Show conversation history
  /sources  — Show sources used in last response
  /quit     — Exit

Usage:
    python chat_cli.py
"""

import os
import sys
import textwrap
from datetime import datetime

from rag_chatbot import RAGChatBot


# ── Terminal Colors ─────────────────────────────────────────────
class Colors:
    HEADER = "\033[95m"      # Purple
    BLUE = "\033[94m"       # Blue
    CYAN = "\033[96m"       # Cyan
    GREEN = "\033[92m"      # Green
    YELLOW = "\033[93m"     # Yellow
    RED = "\033[91m"        # Red
    BOLD = "\033[1m"        # Bold
    UNDERLINE = "\033[4m"   # Underline
    END = "\033[0m"         # Reset


def print_banner():
    """Print welcome banner."""
    banner = f"""
{Colors.CYAN}{Colors.BOLD}╔══════════════════════════════════════════════════════════════╗
║                  ECMC RAG CHATBOT                              ║
║         Student Loan & Financial Aid Assistant                ║
╚══════════════════════════════════════════════════════════════╝{Colors.END}

{Colors.YELLOW}Commands:{Colors.END}
  {Colors.GREEN}/new{Colors.END}      — Start a new chat
  {Colors.GREEN}/history{Colors.END}  — Show conversation history
  {Colors.GREEN}/sources{Colors.END}  — Show sources from last response
  {Colors.GREEN}/quit{Colors.END}     — Exit

{Colors.CYAN}Type your question below...{Colors.END}
"""
    print(banner)


def print_user(msg: str):
    """Print user message."""
    width = 60
    wrapped = textwrap.fill(msg, width=width)
    lines = wrapped.split("\n")
    print(f"\n{Colors.GREEN}┌─ You ─{'─' * 52}┐{Colors.END}")
    for line in lines:
        print(f"{Colors.GREEN}│{Colors.END}  {line:<{width}}  {Colors.GREEN}│{Colors.END}")
    print(f"{Colors.GREEN}└{'─' * 60}┘{Colors.END}")


def print_bot(msg: str, sources: list = None):
    """Print bot response."""
    width = 60
    wrapped = textwrap.fill(msg, width=width)
    lines = wrapped.split("\n")
    print(f"\n{Colors.BLUE}┌─ ECMC Bot ─{'─' * 47}┐{Colors.END}")
    for line in lines:
        print(f"{Colors.BLUE}│{Colors.END}  {line:<{width}}  {Colors.BLUE}│{Colors.END}")
    print(f"{Colors.BLUE}└{'─' * 60}┘{Colors.END}")

    if sources:
        print(f"{Colors.YELLOW}   📚 Sources: {', '.join(sources[:3])}{Colors.END}")


def print_system(msg: str):
    """Print system message."""
    print(f"\n{Colors.YELLOW}⚡ {msg}{Colors.END}")


def print_history(history: list):
    """Print conversation history."""
    if not history:
        print_system("No conversation history yet.")
        return

    print(f"\n{Colors.CYAN}{Colors.BOLD}─── Conversation History ───{Colors.END}")
    for i, msg in enumerate(history, 1):
        role = msg["role"]
        content = msg["content"][:100] + "..." if len(msg["content"]) > 100 else msg["content"]
        if role == "user":
            print(f"{Colors.GREEN}  [{i}] You:{Colors.END} {content}")
        elif role == "assistant":
            print(f"{Colors.BLUE}  [{i}] Bot:{Colors.END} {content}")
    print(f"{Colors.CYAN}{'─' * 30}{Colors.END}")


def print_sources(chunks: list):
    """Print source chunks."""
    if not chunks:
        print_system("No sources available.")
        return

    print(f"\n{Colors.CYAN}{Colors.BOLD}─── Sources ───{Colors.END}")
    for i, chunk in enumerate(chunks[:5], 1):
        meta = chunk.get("metadata", chunk)
        doc = meta.get("document_id", "unknown")
        page = meta.get("page_number", 0)
        section = meta.get("section", "")[:40]
        score = chunk.get("score", 0)
        print(f"{Colors.YELLOW}  [{i}]{Colors.END} {doc} | Page {page} | {section}")
        print(f"       Score: {score:.4f} | {chunk.get('text', '')[:80]}...")
    print(f"{Colors.CYAN}{'─' * 30}{Colors.END}")


# ── Main Chat Loop ──────────────────────────────────────────────

def main():
    print_banner()

    # Initialize bot
    print_system("Loading RAG ChatBot...")
    try:
        bot = RAGChatBot()
        print_system("✅ Bot ready! Session started.")
    except Exception as e:
        print(f"{Colors.RED}❌ Failed to initialize bot: {e}{Colors.END}")
        sys.exit(1)

    session_id = None
    last_chunks = []

    while True:
        try:
            # Get user input
            user_input = input(f"\n{Colors.GREEN}You > {Colors.END}").strip()

            if not user_input:
                continue

            # ── Commands ──
            if user_input.lower() == "/quit":
                print_system("Goodbye! 👋")
                break

            elif user_input.lower() == "/new":
                session_id = bot.new_chat(session_id)
                print_system("🆕 New chat started. History cleared.")
                continue

            elif user_input.lower() == "/history":
                history = bot.get_history(session_id) if session_id else []
                print_history(history)
                continue

            elif user_input.lower() == "/sources":
                print_sources(last_chunks)
                continue

            # ── Normal Chat ──
            print_user(user_input)

            # Show thinking indicator
            print(f"{Colors.YELLOW}   🤖 Thinking...{Colors.END}", end="", flush=True)

            # Get response
            response = bot.chat(
                user_message=user_input,
                session_id=session_id,
                top_k=5,
                expand_context=True,
            )

            # Clear thinking indicator
            print("\r" + " " * 20 + "\r", end="")

            # Store session ID for continuity
            session_id = response["session_id"]
            last_chunks = response.get("chunks", [])

            # Print response
            sources = [c.get("chunk_id", "") for c in last_chunks]
            print_bot(response["answer"], sources=sources if sources else None)

            # Show session info
            print(f"{Colors.CYAN}   📋 Session: {session_id[:8]}... | History: {response['history_length']} msgs{Colors.END}")

        except KeyboardInterrupt:
            print(f"\n{Colors.YELLOW}\n⚡ Interrupted. Type /quit to exit.{Colors.END}")
        except Exception as e:
            print(f"{Colors.RED}❌ Error: {e}{Colors.END}")


if __name__ == "__main__":
    main()

#chat_memory.py
"""
Chat session memory with sliding window.
Maintains conversation history per session, keeping only the last N messages.
"""

import json
import logging
import time
import uuid
from dataclasses import dataclass, asdict
from typing import Dict, List, Optional

from config import config

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)


@dataclass
class ChatMessage:
    """Single message in a chat session."""
    role: str  # "system", "user", "assistant"
    content: str
    timestamp: float = 0.0
    metadata: Optional[Dict[str, any]] = None

    def __post_init__(self):
        if self.timestamp == 0.0:
            self.timestamp = time.time()


class ChatSession:
    """
    Manages conversation history for a single chat session.
    Implements sliding window: keeps only last `max_window_size` messages
    (excluding system prompt, which is always preserved).
    """

    def __init__(
        self,
        session_id: Optional[str] = None,
        max_window_size: int = 5,
        system_prompt: Optional[str] = None,
    ):
        self.session_id = session_id or str(uuid.uuid4())
        self.max_window_size = max_window_size
        self.messages: List[ChatMessage] = []
        self.created_at = time.time()
        self.last_active = time.time()

        # System prompt is always message 0, never evicted
        if system_prompt:
            self.messages.append(
                ChatMessage(role="system", content=system_prompt, timestamp=self.created_at)
            )

    def add_message(self, role: str, content: str, metadata: Optional[Dict] = None) -> None:
        """Add a message and apply sliding window."""
        self.messages.append(
            ChatMessage(role=role, content=content, metadata=metadata)
        )
        self.last_active = time.time()
        self._apply_sliding_window()

    def _apply_sliding_window(self) -> None:
        """
        Keep only the last `max_window_size` non-system messages.
        System prompt (index 0) is never removed.
        """
        if len(self.messages) <= 1:
            return  # Only system prompt or empty

        non_system = [m for m in self.messages if m.role != "system"]
        system = [m for m in self.messages if m.role == "system"]

        if len(non_system) > self.max_window_size:
            # Keep only last N non-system messages
            kept_non_system = non_system[-self.max_window_size:]
            self.messages = system + kept_non_system
            logger.debug(
                f"Session {self.session_id}: slid window to {len(self.messages)} messages"
            )

    def get_history(self, include_system: bool = True) -> List[Dict[str, str]]:
        """
        Return messages as list of dicts for LLM prompt.
        Format: [{"role": "user", "content": "..."}, ...]
        """
        msgs = self.messages if include_system else [m for m in self.messages if m.role != "system"]
        return [{"role": m.role, "content": m.content} for m in msgs]

    def new_chat(self, system_prompt: Optional[str] = None) -> None:
        """Clear all messages except system prompt (or set new one)."""
        old_system = self.messages[0] if self.messages and self.messages[0].role == "system" else None
        self.messages = []
        if system_prompt:
            self.messages.append(ChatMessage(role="system", content=system_prompt))
        elif old_system:
            self.messages.append(old_system)
        self.last_active = time.time()
        logger.info(f"Session {self.session_id}: new chat started")

    def to_dict(self) -> Dict:
        """Serialize for storage."""
        return {
            "session_id": self.session_id,
            "max_window_size": self.max_window_size,
            "created_at": self.created_at,
            "last_active": self.last_active,
            "messages": [asdict(m) for m in self.messages],
        }

    @classmethod
    def from_dict(cls, data: Dict) -> "ChatSession":
        """Deserialize from storage."""
        session = cls(
            session_id=data["session_id"],
            max_window_size=data.get("max_window_size", 5),
        )
        session.created_at = data.get("created_at", time.time())
        session.last_active = data.get("last_active", time.time())
        session.messages = [
            ChatMessage(**m) for m in data.get("messages", [])
        ]
        return session


class SessionManager:
    """
    Manages multiple concurrent chat sessions.
    For Lambda: sessions can be stored in DynamoDB; this is the in-memory cache.
    """

    def __init__(self, default_window_size: int = 5):
        self.sessions: Dict[str, ChatSession] = {}
        self.default_window_size = default_window_size

    def get_or_create(
        self,
        session_id: Optional[str] = None,
        system_prompt: Optional[str] = None,
    ) -> ChatSession:
        """Get existing session or create new one."""
        if session_id and session_id in self.sessions:
            return self.sessions[session_id]

        session = ChatSession(
            session_id=session_id,
            max_window_size=self.default_window_size,
            system_prompt=system_prompt,
        )
        self.sessions[session.session_id] = session
        return session

    def new_chat(self, session_id: str, system_prompt: Optional[str] = None) -> ChatSession:
        """Start a new chat in an existing session."""
        session = self.get_or_create(session_id)
        session.new_chat(system_prompt=system_prompt)
        return session

    def delete_session(self, session_id: str) -> bool:
        """Remove a session from memory."""
        if session_id in self.sessions:
            del self.sessions[session_id]
            return True
        return False

    def get_all_sessions(self) -> List[str]:
        """Return all active session IDs."""
        return list(self.sessions.keys())

#rag_chatbot.py
"""
RAG ChatBot with session memory, retrieval, and Bedrock generation.
Integrates: ChatSession + RetrievalEngine + Bedrock Nova/Claude.
"""

import json
import logging
from typing import Any, Dict, List, Optional

import boto3

from chat_memory import ChatSession, SessionManager
from config import config
from embedding_engine import EmbeddingEngine
from faiss_vector_store import FAISSVectorStore  # or OpenSearchVectorStore
from retrieval_engine import RetrievalEngine

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)


class RAGChatBot:
    """
    End-to-end RAG chatbot:
      1. Maintain session memory (sliding window)
      2. Retrieve relevant chunks from vector store
      3. Build prompt with history + context
      4. Generate answer via Bedrock (Nova/Claude)
    """

    DEFAULT_SYSTEM_PROMPT = (
        "You are a helpful assistant for ECMC student loan services. "
        "Answer questions based ONLY on the provided context. "
        "If the answer is not in the context, say 'I don't have enough information.' "
        "Always cite the document name and page number when possible."
    )

    def __init__(
        self,
        vector_store: Optional[Any] = None,
        model_id: Optional[str] = None,
        session_manager: Optional[SessionManager] = None,
    ):
        # Vector store (FAISS or OpenSearch)
        self.store = vector_store or self._load_default_store()

        # Retrieval engine
        self.retriever = RetrievalEngine(vector_store=self.store)

        # Session manager
        self.sessions = session_manager or SessionManager(default_window_size=5)

        # Bedrock generation
        self.bedrock = boto3.client("bedrock-runtime", region_name=config.AWS_REGION)
        self.model_id = model_id or "amazon.nova-pro-v1:0"  # or your preferred model

    def _load_default_store(self) -> FAISSVectorStore:
        """Load FAISS index from S3 vectorstore/ folder."""
        store = FAISSVectorStore()
        try:
            store.load_from_s3(name="ecmc_index")
        except FileNotFoundError:
            logger.warning("No saved FAISS index found. Call build_index() first.")
        return store

    def build_index(self) -> None:
        """Build FAISS index from S3 embeddings and save."""
        self.store.build_from_s3()
        self.store.save_to_s3(name="ecmc_index")

    # ------------------------------------------------------------------
    # Public API
    # ------------------------------------------------------------------

    def chat(
        self,
        user_message: str,
        session_id: Optional[str] = None,
        top_k: int = 5,
        expand_context: bool = True,
    ) -> Dict[str, Any]:
        """
        Main chat entry point.

        Args:
            user_message: User's question.
            session_id: Existing session ID, or None to create new.
            top_k: Number of chunks to retrieve.
            expand_context: Whether to fetch prev/next chunks.

        Returns:
            Dict with:
                - "session_id": session ID
                - "answer": generated answer
                - "chunks": retrieved chunks (for UI display)
                - "context_used": whether RAG context was used
        """
        # 1. Get or create session
        session = self.sessions.get_or_create(
            session_id=session_id,
            system_prompt=self.DEFAULT_SYSTEM_PROMPT,
        )

        # 2. Retrieve relevant chunks
        retrieval_result = self.retriever.retrieve(
            query=user_message,
            top_k=top_k,
            expand_context=expand_context,
        )
        context = retrieval_result["context_text"]
        chunks = retrieval_result["chunks"]

        # 3. Build messages for LLM
        # Add user message with context injected
        user_prompt = self._build_user_prompt(user_message, context)
        session.add_message("user", user_prompt)

        # 4. Call Bedrock
        messages = session.get_history(include_system=True)
        answer = self._generate(messages)

        # 5. Add assistant response to history
        session.add_message("assistant", answer, metadata={"sources": [c.get("chunk_id") for c in chunks]})

        return {
            "session_id": session.session_id,
            "answer": answer,
            "chunks": chunks,
            "context_used": bool(context.strip()),
            "history_length": len(session.messages),
        }

    def new_chat(self, session_id: Optional[str] = None) -> str:
        """
        Start a new chat. Clears history but keeps session ID.
        Returns the session ID.
        """
        if session_id:
            session = self.sessions.new_chat(session_id, system_prompt=self.DEFAULT_SYSTEM_PROMPT)
            return session.session_id
        else:
            session = self.sessions.get_or_create(system_prompt=self.DEFAULT_SYSTEM_PROMPT)
            session.new_chat()
            return session.session_id

    def get_history(self, session_id: str) -> List[Dict[str, str]]:
        """Return chat history for a session."""
        session = self.sessions.get_or_create(session_id)
        return session.get_history(include_system=False)

    # ------------------------------------------------------------------
    # Internal helpers
    # ------------------------------------------------------------------

    @staticmethod
    def _build_user_prompt(question: str, context: str) -> str:
        """Inject retrieved context into the user message."""
        return (
            f"Context:\\n{context}\\n\\n"
            f"Question: {question}\\n\\n"
            f"Please answer based on the context above."
        )

    def _generate(self, messages: List[Dict[str, str]]) -> str:
        """
        Call Bedrock for text generation.
        Supports Nova, Claude, and other Bedrock models.
        """
        try:
            # Converse API (recommended for Nova/Claude)
            response = self.bedrock.converse(
                modelId=self.model_id,
                messages=[
                    {"role": m["role"], "content": [{"text": m["content"]}]}
                    for m in messages
                ],
                inferenceConfig={"maxTokens": 1024, "temperature": 0.3},
            )
            return response["output"]["message"]["content"][0]["text"]

        except Exception as e:
            logger.error(f"Bedrock generation failed: {e}")
            # Fallback: try InvokeModel for older APIs
            return self._generate_fallback(messages)

    def _generate_fallback(self, messages: List[Dict[str, str]]) -> str:
        """Fallback using InvokeModel with native payload format."""
        # Format for Nova
        native_request = {
            "messages": [
                {"role": m["role"], "content": {"text": m["content"]}}
                for m in messages
            ],
            "inferenceConfig": {"max_new_tokens": 1024, "temperature": 0.3},
        }

        request = json.dumps(native_request)

        try:
            response = self.bedrock.invoke_model(
                modelId=self.model_id,
                body=request,
            )
            model_response = json.loads(response["body"].read())
            # Nova response format
            return model_response.get("output", {}).get("message", {}).get("content", [{}])[0].get("text", "")
        except Exception as e:
            logger.error(f"Fallback generation also failed: {e}")
            return "I apologize, but I'm unable to generate a response right now."
