Handles both RAGAS 0.1.x and 0.2.x APIs.
Uses AWS Bedrock (Nova Lite + Titan Embed) as judge LLM and embeddings.

Input Excel columns required:
  - question: The user query
  - answer: The model-generated answer
  - ground_truth: The expected/correct answer
  - contexts: Retrieved chunks (list, pipe-separated string, or JSON string)

Output Excel: original columns + metric score columns

Usage:
    python ragas_evaluate_v2.py

# ── Bedrock Setup ──────────────────────────────────────────────────────────

def setup_bedrock_llm():
    """Create Bedrock LLM for RAGAS evaluation."""
    llm = ChatBedrock(
        model_id=LLM_MODEL_ID,

    """Create Bedrock embeddings for RAGAS evaluation."""
    emb = BedrockEmbeddings(

# ── Data Loading ───────────────────────────────────────────────────────────

def load_input_data(excel_path: str) -> pd.DataFrame:
    if not os.path.exists(excel_path):
        raise FileNotFoundError(f"Input file not found: {excel_path}")
    df = pd.read_excel(excel_path)
    print(f"Loaded {len(df)} rows from {excel_path}")
    print(f"Columns: {list(df.columns)}")
    return df


def parse_contexts(raw_contexts) -> List[str]:
    """Parse contexts into list of strings."""
    if pd.isna(raw_contexts):
        return []
    if isinstance(raw_contexts, list):
        return [str(c).strip() for c in raw_contexts if c]
    text = str(raw_contexts).strip()
    if text.startswith("[") and text.endswith("]"):
        try:
            parsed = json.loads(text.replace("'", '"'))
            if isinstance(parsed, list):
                return [str(c).strip() for c in parsed if c]
        except Exception:
            pass
    if CONTEXT_SEPARATOR in text:
        return [c.strip() for c in text.split(CONTEXT_SEPARATOR) if c.strip()]
    return [text] if text else []


def prepare_ragas_dataset(df: pd.DataFrame) -> Dataset:
    required = {"question", "answer", "ground_truth"}
    missing = required - set(df.columns)
    if missing:
        raise ValueError(f"Missing required columns: {missing}. Found: {list(df.columns)}")

    if "contexts" in df.columns:
        df["contexts"] = df["contexts"].apply(parse_contexts)
    else:
        print("WARNING: 'contexts' column not found. Context metrics will score 0.")
        df["contexts"] = [[] for _ in range(len(df))]

    for col in ["question", "answer", "ground_truth"]:
        df[col] = df[col].fillna("").astype(str)

    data = {
        "question": df["question"].tolist(),
        "answer": df["answer"].tolist(),
        "contexts": df["contexts"].tolist(),
        "ground_truth": df["ground_truth"].tolist(),
    }
    return Dataset.from_dict(data)

# ── Evaluation ─────────────────────────────────────────────────────────────

def run_evaluation(ragas_dataset: Dataset, llm, embeddings) -> Dataset:
    metric_list = list(METRICS.values())
    print(f"\nRunning RAGAS evaluation with {len(metric_list)} metrics:")
    for name, metric in METRICS.items():
        print(f"  - {name}")

    # RAGAS 0.2.x uses evaluate() with llm and embeddings kwargs
    # RAGAS 0.1.x uses evaluate() with the same kwargs
    try:
        result = evaluate(
            dataset=ragas_dataset,
            metrics=metric_list,
            llm=llm,
            embeddings=embeddings,
        )
    except TypeError as e:
        # Some versions might not accept embeddings parameter
        print(f"Warning: {e}. Retrying without embeddings parameter...")
        result = evaluate(
            dataset=ragas_dataset,
            metrics=metric_list,
            llm=llm,
        )

    return result

# ── Output ─────────────────────────────────────────────────────────────────

def save_results(result_dataset: Dataset, original_df: pd.DataFrame, output_path: str) -> None:
    result_df = result_dataset.to_pandas()

    # Identify metric columns (not in original input columns)
    input_cols = {"question", "answer", "contexts", "ground_truth"}
    metric_cols = [c for c in result_df.columns if c not in input_cols]

    print(f"\nMetric scores computed: {metric_cols}")

    output_df = original_df.copy()
    for col in metric_cols:
        output_df[col] = result_df[col].values

    output_df.to_excel(output_path, index=False)
    print(f"\nSaved results to: {output_path}")
    print(f"Total rows: {len(output_df)}")

    print("\n--- Metric Summary ---")
    for col in metric_cols:
        mean_score = output_df[col].mean()
        print(f"  {col}: {mean_score:.4f} (mean)")

# ── Main ───────────────────────────────────────────────────────────────────

def main():
    print("=" * 60)
    print("ECMC RAGAS Evaluation")
    print(f"RAGAS API: {API_VERSION}")
    print(f"LLM: {LLM_MODEL_ID}")
    print(f"Embeddings: {EMBEDDING_MODEL_ID}")
    print("=" * 60)
    print()

    df = load_input_data(INPUT_FILE)
    ragas_dataset = prepare_ragas_dataset(df)
    print(f"Prepared dataset with {len(ragas_dataset)} samples")

    print("\nInitializing Bedrock models...")
    llm = setup_bedrock_llm()
    embeddings = setup_bedrock_embeddings()
    print("Models ready")

    result = run_evaluation(ragas_dataset, llm, embeddings)
    save_results(result, df, OUTPUT_FILE)

    print("\n" + "=" * 60)
    print("Evaluation complete")
    print("=" * 60)


if __name__ == "__main__":
    main()

def load_input_data(excel_path: str) -> pd.DataFrame:
    """Load evaluation data from Excel.

    Expected columns:
      - question

    if not os.path.exists(excel_path):

def parse_contexts(raw_contexts) -> List[str]:
    """Parse contexts column into list of strings.

    Handles:
      - Python list literal: "['chunk1', 'chunk2']"
      - Pipe-separated: "chunk1 | chunk2"
      - Single string: "chunk1"
      - NaN/None: returns []
    """
    if pd.isna(raw_contexts):
        return []

    if isinstance(raw_contexts, list):
        return [str(c).strip() for c in raw_contexts if c]

    text = str(raw_contexts).strip()

    # Try JSON/list literal
    if text.startswith("[") and text.endswith("]"):
        try:
            parsed = json.loads(text.replace("'", '"'))
            if isinstance(parsed, list):
                return [str(c).strip() for c in parsed if c]
        except Exception:
            pass

    # Try pipe-separated
    if CONTEXT_SEPARATOR in text:
        return [c.strip() for c in text.split(CONTEXT_SEPARATOR) if c.strip()]

    # Single context
    return [text] if text else []


def prepare_ragas_dataset(df: pd.DataFrame) -> Dataset:
    """Convert DataFrame to RAGAS Dataset format.

    RAGAS expects:
      - question: str
      - answer: str
      - contexts: List[str]
      - ground_truth: str
    """
    required = {"question", "answer", "ground_truth"}
    missing = required - set(df.columns)
    if missing:
        raise ValueError(f"Missing required columns: {missing}. Found: {list(df.columns)}")

    # Parse contexts
    if "contexts" in df.columns:
        df["contexts"] = df["contexts"].apply(parse_contexts)
    else:
        print("WARNING: 'contexts' column not found. Metrics requiring contexts will fail.")
        df["contexts"] = [[] for _ in range(len(df))]

    # Ensure all values are strings
    for col in ["question", "answer", "ground_truth"]:
        df[col] = df[col].fillna("").astype(str)

    # Convert to RAGAS Dataset
    data = {
        "question": df["question"].tolist(),
        "answer": df["answer"].tolist(),
        "contexts": df["contexts"].tolist(),
        "ground_truth": df["ground_truth"].tolist(),
    }

    return Dataset.from_dict(data)

# ── Metric Selection ───────────────────────────────────────────────────────

def get_metrics() -> List[Any]:
    """Return the list of RAGAS metrics to evaluate.

    Metrics:
      - context_precision: Measures if relevant contexts are ranked higher
      - context_recall: Measures if all ground truth info is in retrieved contexts
      - context_entity_recall: Measures entity coverage in contexts
      - noise_sensitivity: Measures robustness to irrelevant contexts
      - answer_relevancy: Measures how relevant the answer is to the question
      - faithfulness: Measures if answer is grounded in retrieved contexts
    """
    metrics = []

    # Context metrics (require contexts + ground_truth)
    metrics.append(context_precision)
    metrics.append(context_recall)

    # Context entity recall (requires contexts + ground_truth)
    try:
        metrics.append(context_entity_recall)
    except Exception as exc:
        print(f"WARNING: context_entity_recall not available in this RAGAS version: {exc}")

    # Noise sensitivity (requires contexts + ground_truth + answer)
    try:
        metrics.append(noise_sensitivity)
    except Exception as exc:
        print(f"WARNING: noise_sensitivity not available: {exc}")

    # Answer metrics
    metrics.append(answer_relevancy)
    metrics.append(faithfulness)

    return metrics

# ── Evaluation ─────────────────────────────────────────────────────────────

def run_evaluation(ragas_dataset: Dataset, metrics: List[Any], llm, embeddings) -> Dataset:
    """Run RAGAS evaluation with Bedrock models."""
    print(f"\nRunning RAGAS evaluation with {len(metrics)} metrics...")
    for m in metrics:
        print(f"  - {m.name}")

    result = evaluate(
        dataset=ragas_dataset,
        metrics=metrics,
        llm=llm,
        embeddings=embeddings,
    )

    return result

def save_results(result_dataset: Dataset, original_df: pd.DataFrame, output_path: str) -> None:
    """Merge RAGAS scores with original data and save to Excel."""
    result_df = result_dataset.to_pandas()

    # RAGAS result columns: question, answer, contexts, ground_truth, metric_1, metric_2, ...
    # We want to keep original columns and add metric scores

    # Identify metric columns (float values between 0 and 1, or NaN)
    metric_cols = []
    for col in result_df.columns:
        if col not in {"question", "answer", "contexts", "ground_truth"}:
            metric_cols.append(col)

    print(f"\nMetric scores computed: {metric_cols}")

    # Merge with original dataframe (preserve original column order + add metrics)
    output_df = original_df.copy()
    for col in metric_cols:
        output_df[col] = result_df[col].values

    # Save
    output_df.to_excel(output_path, index=False)
    print(f"\nSaved results to: {output_path}")
    print(f"Total rows: {len(output_df)}")

    # Print summary statistics
    print("\n--- Metric Summary ---")
    for col in metric_cols:
        mean_score = output_df[col].mean()
        print(f"  {col}: {mean_score:.4f} (mean)")

def read_rows(path):
    wb = load_workbook(path, data_only=True)
    ws = wb.active
    header_row = next(ws.iter_rows(min_row=1, max_row=1, values_only=True))
    headers = [(h or "").strip().lower() for h in header_row]

    q_col = headers.index("question")
    a_col = headers.index("model answer")
    g_col = headers.index("ground truth")
    c_col = headers.index("retrieved context") if "retrieved context" in headers else None

    rows = []
    for r in ws.iter_rows(min_row=2, values_only=True):
        if not r[q_col]:
            continue
        contexts = []
        if c_col is not None and r[c_col]:
            contexts = [c.strip() for c in str(r[c_col]).split("|") if c.strip()]
        rows.append({
            "question": str(r[q_col]).strip(),
            "answer": str(r[a_col]).strip() if r[a_col] else "",
            "reference": str(r[g_col]).strip() if r[g_col] else "",
            "contexts": contexts,
        })
    return rows


def write_rows(path, results):
    wb = Workbook()
    ws = wb.active
    ws.title = "ragas_results"
    ws.append([
        "question", "model_answer", "ground_truth",
        "answer_relevancy", "noise_sensitivity", "faithfulness", "context_relevance",
    ])
    for r in results:
        ws.append([
            r["question"], r["answer"], r["reference"],
            r["answer_relevancy"], r["noise_sensitivity"], r["faithfulness"], r["context_relevance"],
        ])
    wb.save(path)

"""
RAGAS-style evaluation for the ECMC RAG pipeline.

Reads an xlsx with Question / Model Answer / Ground truth columns, retrieves
context for each question through the existing retrieval + reranking
pipeline, and scores each row on six metrics using Bedrock's Nova Lite as
the judge model:

  - faithfulness
  - answer_relevancy
  - context_precision (with reference)
  - context_recall
  - context_entities_recall
  - noise_sensitivity

Formulas follow the Ragas docs:
  https://docs.ragas.io/en/stable/concepts/metrics/available_metrics/

Usage:
    python ragas_eval.py --input qa.xlsx --output scored.xlsx
"""

import argparse

import re

# NOTE: this needs to actually resolve to a model you have Bedrock access
# to in your region - if you're on a cross-region inference profile you
# may need "us.amazon.nova-lite-v1:0" instead.
MODEL_ID = "amazon.nova-lite-v1:0"
REGION = "us-east-1"
TOP_K = 5

bedrock = boto3.client("bedrock-runtime", region_name=REGION)


def call_llm(system_prompt, user_prompt, max_tokens=1024):
    resp = bedrock.converse(
        modelId=MODEL_ID,
        system=[{"text": system_prompt}],
        messages=[{"role": "user", "content": [{"text": user_prompt}]}],
        inferenceConfig={"temperature": 0, "maxTokens": max_tokens},
    )
    return resp["output"]["message"]["content"][0]["text"]


def extract_json(text):
    match = re.search(r"[\{\[].*[\}\]]", text, re.DOTALL)
    if not match:
        raise ValueError(f"no JSON found in model response: {text!r}")
    return json.loads(match.group(0))


def cosine_sim(a, b):
    a, b = np.array(a), np.array(b)
    denom = np.linalg.norm(a) * np.linalg.norm(b)
    return float(np.dot(a, b) / denom) if denom else 0.0


# ---------------------------------------------------------------------
# LLM judge calls
# ---------------------------------------------------------------------

def split_claims(text):
    system = ('Break the text into a list of individual factual claims. '
              'Respond with only JSON: {"claims": ["claim 1", "claim 2"]}')
    result = extract_json(call_llm(system, text))
    return [c for c in result.get("claims", []) if c.strip()]


def claim_supported_by_context(claim, contexts):
    context_text = "\n\n".join(contexts)
    system = ('Decide if the claim can be inferred from the context. '
              'Respond with only JSON: {"verdict": true or false}')
    user = f"CONTEXT:\n{context_text}\n\nCLAIM:\n{claim}"
    return bool(extract_json(call_llm(system, user)).get("verdict"))


def claim_correct_vs_reference(claim, reference):
    system = ('Decide if the claim is factually consistent with the reference answer. '
              'Respond with only JSON: {"verdict": true or false}')
    user = f"REFERENCE:\n{reference}\n\nCLAIM:\n{claim}"
    return bool(extract_json(call_llm(system, user)).get("verdict"))


def chunk_relevant_to_reference(chunk, reference):
    system = ('Decide if the context chunk was useful for producing the reference answer. '
              'Respond with only JSON: {"verdict": true or false}')
    user = f"REFERENCE:\n{reference}\n\nCHUNK:\n{chunk}"
    return bool(extract_json(call_llm(system, user)).get("verdict"))


def extract_entities(text):
    system = ('Extract named entities (people, places, organizations, dates, numbers, '
              'specific terms) from the text. Respond with only JSON: '
              '{"entities": ["entity 1", "entity 2"]}')
    result = extract_json(call_llm(system, text))
    return {e.strip().lower() for e in result.get("entities", []) if e.strip()}


def generate_questions_from_answer(answer, n=3):
    system = (f'Generate {n} different questions that this answer would be a good response to. '
              'Respond with only JSON: {"questions": ["q1", "q2", "q3"]}')
    result = extract_json(call_llm(system, answer))
    return [q for q in result.get("questions", []) if q.strip()]


# ---------------------------------------------------------------------
# metrics
# ---------------------------------------------------------------------

def faithfulness(answer, contexts):
    claims = split_claims(answer)
    if not claims:
        return None
    supported = sum(claim_supported_by_context(c, contexts) for c in claims)
    return supported / len(claims)


def context_recall(reference, contexts):
    claims = split_claims(reference)
    if not claims:
        return None
    supported = sum(claim_supported_by_context(c, contexts) for c in claims)
    return supported / len(claims)


def context_precision(reference, contexts):
    if not contexts:
        return None
    relevance = [chunk_relevant_to_reference(c, reference) for c in contexts]

    relevant_count = 0
    precision_sum = 0.0
    for i, is_relevant in enumerate(relevance, start=1):
        if is_relevant:
            relevant_count += 1
            precision_sum += relevant_count / i

    return precision_sum / relevant_count if relevant_count else 0.0


def context_entities_recall(reference, contexts):
    ref_entities = extract_entities(reference)
    if not ref_entities:
        return None
    ctx_entities = set()
    for c in contexts:
        ctx_entities |= extract_entities(c)
    return len(ref_entities & ctx_entities) / len(ref_entities)


def noise_sensitivity(answer, reference):
    # "relevant" mode: fraction of the answer's claims that don't hold up
    # against the reference answer.
    claims = split_claims(answer)
    if not claims:
        return None
    incorrect = sum(not claim_correct_vs_reference(c, reference) for c in claims)
    return incorrect / len(claims)


def answer_relevancy(question, answer):
    questions = generate_questions_from_answer(answer)
    if not questions:
        return None
    q_vec = embed_text(bedrock, question)
    sims = [cosine_sim(q_vec, embed_text(bedrock, gq)) for gq in questions]
    return sum(sims) / len(sims)

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--input", required=True, help="xlsx with Question / Model Answer / Ground truth columns")
    parser.add_argument("--output", default="ragas_results.xlsx")
    args = parser.parse_args()

    rows = read_rows(args.input)
    retriever = get_retriever()

    results = []
    for i, row in enumerate(rows, start=1):
        print(f"[{i}/{len(rows)}] {row['question']}")

        try:
            fused = rerank(bedrock, retriever, row["question"], top_k=TOP_K)
            contexts = [c["text"] for c in fused]

            results.append({
                "question": row["question"],
                "answer": row["answer"],
                "reference": row["reference"],
                "faithfulness": faithfulness(row["answer"], contexts),
                "answer_relevancy": answer_relevancy(row["question"], row["answer"]),
                "context_precision": context_precision(row["reference"], contexts),
                "context_recall": context_recall(row["reference"], contexts),
                "context_entities_recall": context_entities_recall(row["reference"], contexts),
                "noise_sensitivity": noise_sensitivity(row["answer"], row["reference"]),
            })
        except Exception as e:
            print(f"  FAILED: {e}")
            results.append({
                "question": row["question"], "answer": row["answer"], "reference": row["reference"],
                "faithfulness": None, "answer_relevancy": None, "context_precision": None,
                "context_recall": None, "context_entities_recall": None, "noise_sensitivity": None,
            })

    write_rows(args.output, results)
    print(f"\nwrote {len(results)} rows to {args.output}")


if __name__ == "__main__":
    main()

{"answer": "<your complete answer, the greeting/decline message, or a statement that the \
documents do not contain this information>", "chunk_id": "<the chunk_id of the ONE chunk \
that most directly supports your answer, or null if none do>"}
This section is required in every response, with no exceptions -- even if STEP 4 concluded \
the input is a greeting, unrelated, or unanswerable, still produce a valid FINAL_JSON line.
"""
########diagnose_terse_queries.py
"""
Isolates WHERE terse/fragment queries ("title 4", "back to school") and
vocabulary variants ("title IV" vs "title 4" vs "title four") actually
fail, so you know whether to fix retrieval (context problem) or the prompt
(LLM formatting problem) -- fixing the wrong one won't help.

For each test query, three layers are checked independently and reported
separately:

    1. RETRIEVAL (context layer) -- calls retriever.search() directly and
       reports how many semantic/keyword hits came back and what the top
       chunk actually is. If this comes back empty or clearly wrong for a
       given phrasing, that phrasing has a retrieval problem: the FAISS/
       BM25 search itself isn't finding the right chunk for that wording,
       no matter what the prompt says to do with it. This is the layer
       most likely to break on "title 4" vs "title IV" vs "title four" --
       BM25 does token matching (4 / IV / four are three unrelated
       tokens), and Titan embeddings can also drift on very short queries
       since there's less surrounding context to embed.

    2. FUSION (RRF) -- calls rerank() on top of that, so you can see
       whether fusion is dropping something retrieval actually found (rare,
       but distinguishes "retrieval found it but fusion ranked it out of
       top_k" from "retrieval never found it at all").

    3. GENERATION (LLM layer) -- calls Bedrock directly with
       generation_pipeline.SYSTEM_PROMPT and whatever context survived
       steps 1-2, and reports the RAW model output plus whether
       parse_model_response() succeeded or raised. If retrieval/fusion
       found perfectly good context but the model still fails to produce
       a valid FINAL_JSON section (or produces one but with an empty/
       refusal answer despite good context), that's a genuine LLM/prompt
       problem, not a context one -- exactly the failure mode rule 0 in
       the refined prompt was meant to fix.

TEST_QUERIES below covers the two reported failures plus the Title
IV/4/four vocabulary variants, so you can see in one run whether the
numeral/roman-numeral/spelled-out forms retrieve the same chunk or not.
Edit the list to add more.

Usage:
    python diagnose_terse_queries.py

To test against a DIFFERENT system prompt (e.g. the refined one from this
conversation) instead of generation_pipeline.py's current SYSTEM_PROMPT,
paste it into CUSTOM_SYSTEM_PROMPT below and it'll be used instead --
useful for A/B checking the old vs. refined prompt against the same
retrieved context without editing generation_pipeline.py itself.

Requires AWS credentials with Bedrock access, and the FAISS index already
built in S3 (embedding_pipeline.py must have been run at least once).
"""

from aws_utils import get_bedrock_client
from config import NOVA_MODEL_ID, GENERATION_TEMPERATURE, GENERATION_MAX_TOKENS, HYBRID_CANDIDATE_K, FINAL_TOP_K
from retrieval_pipeline import get_retriever
from reranking_pipeline import rerank
from generation_pipeline import SYSTEM_PROMPT, build_context_block, parse_model_response

TEST_QUERIES = [
    # The two reported failures, verbatim:
    "title 4",
    "back to school",
    # Vocabulary variants of the same underlying term -- do these all
    # retrieve the same chunk, or does phrasing change what comes back?
    "title IV",
    "title four",
    "Title IV letter",
    # A couple more terse ones from the shared test batches, for coverage:
    "Requirements: What to send in for the review",
    "I want to consolidate",
]

# Leave as None to test generation_pipeline.py's current SYSTEM_PROMPT
# as-is. Paste an alternate prompt string here to A/B test it against the
# same queries/context instead.
CUSTOM_SYSTEM_PROMPT = None


def check_retrieval(bedrock_client, retriever, query):
    results = retriever.search(bedrock_client, query, top_k=HYBRID_CANDIDATE_K)
    semantic = results["semantic"]
    keyword = results["keyword"]

    print(f"  RETRIEVAL: {len(semantic)} semantic hit(s), {len(keyword)} keyword hit(s)")
    if semantic:
        top = semantic[0]
        print(f"    top semantic: {top['chunk_id']} (score={top['semantic_score']:.4f}) -- {top['text'][:120]!r}")
    else:
        print("    top semantic: NONE")
    if keyword:
        top = keyword[0]
        print(f"    top keyword:  {top['chunk_id']} (score={top['bm25_score']:.4f}) -- {top['text'][:120]!r}")
    else:
        print("    top keyword:  NONE (BM25 found zero token overlap)")
    return results


def check_fusion(bedrock_client, retriever, query):
    fused = rerank(bedrock_client, retriever, query, top_k=FINAL_TOP_K)
    print(f"  FUSION: {len(fused)} chunk(s) survived RRF into the final context")
    for c in fused[:3]:
        print(f"    {c['chunk_id']} (rrf_score={c['rrf_score']:.5f}) -- {c['text'][:120]!r}")
    return fused


def check_generation(bedrock_client, query, fused_chunks):
    if not fused_chunks:
        print("  GENERATION: skipped -- no context to generate from (this query is purely a retrieval failure)")
        return

    system_prompt = CUSTOM_SYSTEM_PROMPT or SYSTEM_PROMPT
    context_block = build_context_block(fused_chunks)
    user_message = f"CONTEXT:\n{context_block}\n\nQUESTION:\n{query}"

    response = bedrock_client.converse(
        modelId=NOVA_MODEL_ID,
        system=[{"text": system_prompt}],
        messages=[{"role": "user", "content": [{"text": user_message}]}],
        inferenceConfig={"temperature": GENERATION_TEMPERATURE, "maxTokens": GENERATION_MAX_TOKENS},
    )
    raw_text = response["output"]["message"]["content"][0]["text"]

    print(f"  GENERATION raw output ({len(raw_text)} chars):")
    print("    " + raw_text.replace("\n", "\n    "))

    try:
        parsed = parse_model_response(raw_text)
        print(f"  GENERATION: parsed OK -- answer={parsed.get('answer', '')!r} chunk_id={parsed.get('chunk_id', parsed.get('chunk id'))!r}")
    except Exception as e:
        print(f"  GENERATION: FAILED TO PARSE -- {e}")
        print("    ^ this is an LLM/prompt problem: retrieval/fusion supplied context, "
              "but the model didn't produce a valid REASONING/FINAL_JSON response for this query.")


def main():
    bedrock_client = get_bedrock_client()
    retriever = get_retriever()

    for query in TEST_QUERIES:
        print(f"\n{'=' * 70}\nQUERY: {query!r}\n{'=' * 70}")
        check_retrieval(bedrock_client, retriever, query)
        fused_chunks = check_fusion(bedrock_client, retriever, query)
        check_generation(bedrock_client, query, fused_chunks)


if __name__ == "__main__":
    main()
########ragas_metrics.py
"""
RAGAS-style evaluation metrics -- faithfulness, answer relevancy, context
precision, and context recall -- implemented directly against this
pipeline's own Bedrock stack (Nova as judge, Titan for embeddings) rather
than the `ragas` pip package, which defaults to OpenAI/langchain wiring
this project doesn't otherwise depend on. Same REASONING/FINAL_JSON
convention as generation_pipeline.py's SYSTEM_PROMPT and eval_pipeline.py's
JUDGE_SYSTEM_PROMPT, so every judge call here is auditable the same way
(reasoning is logged, only FINAL_JSON is parsed).

Four metrics, each scored 0.0-1.0 (or None if not computable for that row --
see each function's docstring for when that happens; None is kept distinct
from 0.0 on purpose, same as eval_pipeline.py's "N/A" verdict for rows with
no expected_answer):

    faithfulness        -- of the claims the generated answer makes, what
                            fraction are actually supported by the
                            retrieved context? Catches hallucination -- an
                            answer can be fluent and on-topic and still
                            fail this if it states something the context
                            doesn't back up.
    answer_relevancy    -- does the answer actually address the question
                            asked? Measured indirectly: Nova writes a few
                            questions the given answer would suit, each is
                            embedded (Titan, same model used to build the
                            FAISS index) and compared by cosine similarity
                            against the real question's embedding. High
                            similarity means the answer is shaped like a
                            response to what was actually asked, not just
                            topically nearby.
    context_precision   -- of the chunks retrieval handed to generation,
                            are the relevant ones ranked near the top?
                            Scored as an average-precision-style number
                            over Nova's per-chunk relevance judgments, in
                            the same rank order the chunks were fused in.
    context_recall      -- does the retrieved context, all together,
                            contain what's needed to support the
                            *reference* answer (golden dataset's
                            expected_answer)? The reference is split into
                            claims and each is checked against the context.
                            Needs a ground_truth -- returns None without
                            one, same convention as eval_pipeline.py's
                            judge_answer().

Each function takes plain Python values (question / contexts list[str] /
answer / ground_truth strings) rather than a pipeline object, so they work
the same whether the caller is v1's eval_pipeline.py or a future v2 one --
pass in whichever retriever/generation output you already have. Nothing
here calls retrieval or generation itself; get `contexts` from
reranking_pipeline.rerank(...)'s fused chunk list (pull out each chunk's
"text") and `answer` from generation_pipeline.generate_answer(...).

Install:
    pip install boto3 numpy json_repair
"""

import json
import logging
import re

import numpy as np
from json_repair import repair_json

from config import NOVA_MODEL_ID, GENERATION_TEMPERATURE

logging.basicConfig(level=logging.INFO, format="%(asctime)s %(levelname)s %(message)s", datefmt="%H:%M:%S")
log = logging.getLogger(__name__)

JUDGE_MAX_TOKENS = 1024
FINAL_JSON_MARKER = "FINAL_JSON:"


def _call_nova_json(bedrock_client, system_prompt, user_message, max_tokens=JUDGE_MAX_TOKENS):
    """Shared judge call -- same REASONING/FINAL_JSON convention used
    everywhere else in this codebase. Returns the parsed FINAL_JSON
    object. Raises if nothing parseable comes back; callers don't catch
    this, so a bad judge response surfaces as a real error for that
    question rather than silently becoming a wrong 0.0 score."""
    response = bedrock_client.converse(
        modelId=NOVA_MODEL_ID,
        system=[{"text": system_prompt}],
        messages=[{"role": "user", "content": [{"text": user_message}]}],
        inferenceConfig={"temperature": GENERATION_TEMPERATURE, "maxTokens": max_tokens},
    )
    raw_text = response["output"]["message"]["content"][0]["text"]

    if FINAL_JSON_MARKER in raw_text:
        reasoning_part, _, json_part = raw_text.partition(FINAL_JSON_MARKER)
    else:
        reasoning_part, json_part = "", raw_text
    log.info("Judge reasoning: %s", reasoning_part.strip().replace("REASONING:", "").strip())

    match = re.search(r"[\{\[].*[\}\]]", json_part, re.DOTALL)
    if not match:
        raise ValueError(f"No JSON found in judge response: {raw_text!r}")
    candidate = match.group(0)

    try:
        return json.loads(candidate)
    except json.JSONDecodeError as e:
        # Nova frequently quotes the source document verbatim inside a
        # claim's text (e.g. `"claim": "the notice says "within 30 days""`)
        # -- those un-escaped inner quotes break strict JSON parsing right
        # at that point, even though the rest of the structure is fine.
        # json_repair handles this (and other common near-miss JSON --
        # trailing commas, unbalanced brackets) far more reliably than a
        # regex patch would. Logged as a warning, not silently swallowed,
        # so a genuinely broken response is still visible in the logs.
        log.warning("Strict JSON parse failed (%s) -- repairing: %r", e, candidate[:200])
        repaired = repair_json(candidate)
        return json.loads(repaired)


def _cosine_similarity(a, b):
    a, b = np.array(a), np.array(b)
    denom = np.linalg.norm(a) * np.linalg.norm(b)
    return float(np.dot(a, b) / denom) if denom else 0.0


def _format_contexts(contexts):
    return "\n\n".join(f"[{i}] {c}" for i, c in enumerate(contexts, start=1))


# --------------------------------------------------------------------------
# Faithfulness
# --------------------------------------------------------------------------

FAITHFULNESS_SYSTEM_PROMPT = """You are grading whether a generated answer is fully \
supported by the given context, for a bankruptcy/student-loan document assistant. \
Break the answer into its individual factual claims, then judge each claim against \
the context alone -- not outside knowledge, not whether the claim sounds plausible.

Respond in exactly this format, with these two labeled sections and nothing else \
before, between, or after them:

REASONING:
List each factual claim the answer makes. For each, note whether the context \
actually states or directly implies it.

FINAL_JSON:
A single-line JSON object, exactly this shape:
{"claims": [{"claim": "<the claim>", "supported": true | false}, ...]}
If the answer makes no checkable factual claims (e.g. it's a refusal like "the \
documents do not contain this information"), return {"claims": []}.
Write each claim in your own words -- paraphrase, do not copy a quoted phrase from \
the answer or context in double quotes, since that breaks JSON formatting.
"""


def faithfulness(bedrock_client, question, contexts, answer):
    """Fraction of the answer's factual claims the context actually
    supports. Returns None (not 0.0) if the answer made no checkable
    claims at all -- e.g. a refusal -- since that's a different situation
    than making claims and getting them wrong."""
    user_message = (
        f"QUESTION:\n{question}\n\n"
        f"CONTEXT:\n{_format_contexts(contexts)}\n\n"
        f"ANSWER:\n{answer}"
    )
    result = _call_nova_json(bedrock_client, FAITHFULNESS_SYSTEM_PROMPT, user_message)
    claims = result.get("claims", [])
    if not claims:
        return {"score": None, "claims": []}
    supported = sum(1 for c in claims if c.get("supported"))
    return {"score": supported / len(claims), "claims": claims}


# --------------------------------------------------------------------------
# Answer relevancy
# --------------------------------------------------------------------------

RELEVANCY_QUESTIONS_SYSTEM_PROMPT = """Given an answer, write 3 different questions \
that this answer would be a good, direct answer to. Write only questions this \
specific answer actually addresses -- don't invent questions about topics the \
answer doesn't cover.

Respond in exactly this format, with these two labeled sections and nothing else \
before, between, or after them:

REASONING:
Note what the answer is actually about, then draft each question.

FINAL_JSON:
A single-line JSON object, exactly this shape:
{"questions": ["<question 1>", "<question 2>", "<question 3>"]}
"""


def answer_relevancy(bedrock_client, embed_fn, question, answer):
    """Nova writes a few questions the answer would suit; each is embedded
    (pass in the same embed_text used to build the FAISS index -- e.g.
    embedding_pipeline.embed_text or embedding_pipeline_v2.embed_text) and
    compared to the real question by cosine similarity. High average
    similarity means the answer is actually shaped like a response to what
    was asked, not just topically nearby or padded with extra detail.
    Returns None if the answer is a refusal Nova can't derive questions
    from."""
    user_message = f"ANSWER:\n{answer}"
    result = _call_nova_json(bedrock_client, RELEVANCY_QUESTIONS_SYSTEM_PROMPT, user_message)
    generated_questions = [q for q in result.get("questions", []) if q.strip()]
    if not generated_questions:
        return {"score": None, "generated_questions": []}

    question_vector = embed_fn(bedrock_client, question)
    similarities = [
        _cosine_similarity(question_vector, embed_fn(bedrock_client, gq))
        for gq in generated_questions
    ]
    return {"score": sum(similarities) / len(similarities), "generated_questions": generated_questions}


# --------------------------------------------------------------------------
# Context precision
# --------------------------------------------------------------------------

CONTEXT_PRECISION_SYSTEM_PROMPT = """You are judging retrieved context chunks for \
relevance to a question, for a bankruptcy/student-loan document assistant. The \
chunks are numbered in the order retrieval ranked them. For each, judge only \
whether it contains information relevant to answering the question -- not whether \
it fully answers it alone.

Respond in exactly this format, with these two labeled sections and nothing else \
before, between, or after them:

REASONING:
Go through the chunks in order and note whether each is relevant to the question.

FINAL_JSON:
A single-line JSON object, exactly this shape:
{"relevance": [true | false, ...]}
The list must have exactly as many entries as there were numbered chunks, in the \
same order.
"""


def context_precision(bedrock_client, question, contexts):
    """Average-precision-style score over the ranked context list: at each
    position where a chunk is relevant, precision-up-to-that-position is
    added to a running total, which is then divided by the number of
    relevant chunks found. This rewards relevant chunks being ranked
    early, not just present somewhere in the list -- a relevant chunk
    buried last scores worse than the same chunk ranked first. Returns
    None if no context was retrieved, or if none of it was judged
    relevant (keeps "nothing relevant found" distinct from a real
    fractional score)."""
    if not contexts:
        return {"score": None, "relevance": []}

    user_message = f"QUESTION:\n{question}\n\nCHUNKS:\n{_format_contexts(contexts)}"
    result = _call_nova_json(bedrock_client, CONTEXT_PRECISION_SYSTEM_PROMPT, user_message)
    relevance = result.get("relevance", [])
    if len(relevance) != len(contexts):
        log.error("context_precision: judge returned %d relevance labels for %d chunks -- padding with False",
                   len(relevance), len(contexts))
        relevance = (relevance + [False] * len(contexts))[:len(contexts)]

    relevant_count = 0
    precision_sum = 0.0
    for i, is_relevant in enumerate(relevance, start=1):
        if is_relevant:
            relevant_count += 1
            precision_sum += relevant_count / i

    if relevant_count == 0:
        return {"score": None, "relevance": relevance}
    return {"score": precision_sum / relevant_count, "relevance": relevance}


# --------------------------------------------------------------------------
# Context recall
# --------------------------------------------------------------------------

CONTEXT_RECALL_SYSTEM_PROMPT = """You are judging whether retrieved context fully \
covers a reference answer, for a bankruptcy/student-loan document assistant. Break \
the reference answer into individual sentences/claims, then judge each one against \
the context alone: could someone verify or derive this claim from the context?

Respond in exactly this format, with these two labeled sections and nothing else \
before, between, or after them:

REASONING:
List each sentence/claim in the reference answer. For each, note whether the \
context supports it.

FINAL_JSON:
A single-line JSON object, exactly this shape:
{"claims": [{"claim": "<the claim>", "attributable": true | false}, ...]}
Write each claim in your own words -- paraphrase, do not copy a quoted phrase from \
the reference answer or context in double quotes, since that breaks JSON formatting.
"""


def context_recall(bedrock_client, contexts, ground_truth):
    """Fraction of the reference (golden dataset expected_answer)'s claims
    that the retrieved context actually supports. Unlike faithfulness
    (generated answer vs. context) and context_precision (context vs.
    question), this checks context against the *correct* answer -- so it
    measures whether retrieval found everything needed, independent of
    whether generation used it well. Returns None if there's no
    ground_truth to check against, matching eval_pipeline.py's
    judge_answer() "N/A" convention for ungraded rows. Returns 0.0 (not
    None) if there IS a ground_truth but zero context was retrieved --
    that's a real failure to measure, not an ungraded row."""
    if not ground_truth or not ground_truth.strip():
        return {"score": None, "claims": []}
    if not contexts:
        return {"score": 0.0, "claims": []}

    user_message = f"REFERENCE ANSWER:\n{ground_truth}\n\nCONTEXT:\n{_format_contexts(contexts)}"
    result = _call_nova_json(bedrock_client, CONTEXT_RECALL_SYSTEM_PROMPT, user_message)
    claims = result.get("claims", [])
    if not claims:
        return {"score": None, "claims": []}
    attributable = sum(1 for c in claims if c.get("attributable"))
    return {"score": attributable / len(claims), "claims": claims}


# --------------------------------------------------------------------------
# Convenience: run all four for one golden-dataset row
# --------------------------------------------------------------------------

def score_all(bedrock_client, embed_fn, question, contexts, answer, ground_truth=""):
    """Runs all four metrics for one question and returns a flat dict of
    scores (each 0.0-1.0 or None) plus each metric's underlying detail --
    the shape an eval file can drop straight into extra xlsx columns per
    row, alongside eval_pipeline.py's existing CORRECT/PARTIALLY_CORRECT/
    INCORRECT verdict. A handful of Nova calls per question (faithfulness,
    relevancy-questions plus up to 3 embed calls, precision, recall) on top
    of generation's own call -- this is eval tooling run offline against a
    golden dataset, not the live query path, so the extra latency/cost is
    an accepted tradeoff for staying on this project's existing Bedrock
    stack instead of pulling in the `ragas` package and its own separate
    LLM/embeddings wiring."""
    faith = faithfulness(bedrock_client, question, contexts, answer)
    relevancy = answer_relevancy(bedrock_client, embed_fn, question, answer)
    precision = context_precision(bedrock_client, question, contexts)
    recall = context_recall(bedrock_client, contexts, ground_truth)

    return {
        "faithfulness": faith["score"],
        "answer_relevancy": relevancy["score"],
        "context_precision": precision["score"],
        "context_recall": recall["score"],
        "details": {
            "faithfulness": faith,
            "answer_relevancy": relevancy,
            "context_precision": precision,
            "context_recall": recall,
        },
    }
########run_ragas_relevancy_only.py
"""
Scores an external chatbot's question/answer pairs with ONLY
answer_relevancy -- for testing a system (e.g. a teammate's Teams bot)
where you have no visibility into what context it retrieved to answer.

Deliberately does NOT import retrieval_pipeline.py, reranking_pipeline.py,

even has one you have access to). Pulling them in here would be actively

knowing what context an answer was actually grounded in, and if you don't
have that from the bot being tested, retrieving context from a different

whether the bot's answer happens to agree with a corpus it never saw.

answer_relevancy is the one RAGAS metric that genuinely doesn't need
context -- it only checks whether the answer addresses the question that
was asked (via Nova-generated candidate questions + embedding similarity
to the real question, see ragas_metrics.py). That's why it's the only one
usable here. If you get access to whatever sources/citations the Teams bot
returns per answer, come back to run_ragas_from_excel.py's approach
instead -- feeding those in as `contexts` unlocks the other three metrics
for real, still without needing this repo's own retrieval pipeline.

Expected input columns (case-insensitive, any order):
    question       (required)
    model_answer   (required)

Output: <input>_relevancy_scored.xlsx with question, model_answer, and
answer_relevancy.

Usage:
    python run_ragas_relevancy_only.py --input teams_bot_questions.xlsx

Requires AWS credentials with Bedrock access to EMBEDDING_MODEL_ID and
NOVA_MODEL_ID (config.py) -- this still calls Bedrock directly (Nova to
generate candidate questions, Titan to embed them), it just doesn't touch
S3, FAISS, or this repo's chunk index to do it.

Install:
    pip install boto3 python-dotenv numpy openpyxl json_repair
"""

import argparse
import json
import os

from openpyxl import load_workbook, Workbook
from openpyxl.styles import Font, Alignment

from aws_utils import get_bedrock_client
from config import EMBEDDING_MODEL_ID, EMBEDDING_DIMENSIONS
import ragas_metrics

FONT_NAME = "Arial"
HEADER_FONT = Font(name=FONT_NAME, bold=True)
BODY_FONT = Font(name=FONT_NAME)


def embed_text(bedrock_client, text):
    """Minimal, self-contained copy of embedding_pipeline.py's embed_text
    -- duplicated here on purpose rather than imported, so this file has
    zero dependency on embedding_pipeline.py (which also pulls in faiss
    for the index-building side you don't need for this)."""
    body = json.dumps({
        "inputText": text,
        "dimensions": EMBEDDING_DIMENSIONS,
        "normalize": True,
    })
    response = bedrock_client.invoke_model(
        modelId=EMBEDDING_MODEL_ID,
        body=body,
        contentType="application/json",
        accept="application/json",
    )
    payload = json.loads(response["body"].read())
    return payload["embedding"]


def read_rows(input_path):
    wb = load_workbook(input_path, data_only=True)
    ws = wb.active

    header_row = next(ws.iter_rows(min_row=1, max_row=1, values_only=True))
    headers = [(h or "").strip().lower() for h in header_row]

    if "question" not in headers:
        raise ValueError(f"No 'question' column found in {input_path}. Headers found: {header_row}")
    if "model_answer" not in headers:
        raise ValueError(f"No 'model_answer' column found in {input_path}. Headers found: {header_row}")
    question_col = headers.index("question")
    answer_col = headers.index("model_answer")

    rows = []
    for row in ws.iter_rows(min_row=2, values_only=True):
        if row[question_col] is None:
            continue
        rows.append({
            "question": str(row[question_col]).strip(),
            "model_answer": str(row[answer_col] or "").strip(),
        })
    return rows


def score_rows(bedrock_client, rows):
    results = []
    for i, row in enumerate(rows, start=1):
        print(f"[{i}/{len(rows)}] {row['question']}")
        relevancy = ragas_metrics.answer_relevancy(
            bedrock_client, embed_text, row["question"], row["model_answer"]
        )
        results.append({**row, "answer_relevancy": relevancy["score"]})
    return results


def write_output(results, output_path):
    wb = Workbook()
    ws = wb.active
    ws.title = "Answer Relevancy"

    headers = ["Question", "Model Answer", "Answer Relevancy"]
    ws.append(headers)
    for cell in ws[1]:

"""
RAGAS-style evaluation metrics -- faithfulness, answer relevancy, context
precision, and context recall -- implemented directly against this
pipeline's own Bedrock stack (Nova as judge, Titan for embeddings) rather
than the `ragas` pip package, which defaults to OpenAI/langchain wiring
this project doesn't otherwise depend on. Same REASONING/FINAL_JSON
convention as generation_pipeline.py's SYSTEM_PROMPT and eval_pipeline.py's
JUDGE_SYSTEM_PROMPT, so every judge call here is auditable the same way
(reasoning is logged, only FINAL_JSON is parsed).

Four metrics, each scored 0.0-1.0 (or None if not computable for that row --
see each function's docstring for when that happens; None is kept distinct
from 0.0 on purpose, same as eval_pipeline.py's "N/A" verdict for rows with
no expected_answer):

    faithfulness        -- of the claims the generated answer makes, what
                            fraction are actually supported by the
                            retrieved context? Catches hallucination -- an
                            answer can be fluent and on-topic and still
                            fail this if it states something the context
                            doesn't back up.
    answer_relevancy    -- does the answer actually address the question
                            asked? Measured indirectly: Nova writes a few
                            questions the given answer would suit, each is
                            embedded (Titan, same model used to build the
                            FAISS index) and compared by cosine similarity
                            against the real question's embedding. High
                            similarity means the answer is shaped like a
                            response to what was actually asked, not just
                            topically nearby.
    context_precision   -- of the chunks retrieval handed to generation,
                            are the relevant ones ranked near the top?
                            Scored as an average-precision-style number
                            over Nova's per-chunk relevance judgments, in
                            the same rank order the chunks were fused in.
    context_recall      -- does the retrieved context, all together,
                            contain what's needed to support the
                            *reference* answer (golden dataset's
                            expected_answer)? The reference is split into
                            claims and each is checked against the context.
                            Needs a ground_truth -- returns None without
                            one, same convention as eval_pipeline.py's
                            judge_answer().

Each function takes plain Python values (question / contexts list[str] /
answer / ground_truth strings) rather than a pipeline object, so they work
the same whether the caller is v1's eval_pipeline.py or a future v2 one --
pass in whichever retriever/generation output you already have. Nothing
here calls retrieval or generation itself; get `contexts` from
reranking_pipeline.rerank(...)'s fused chunk list (pull out each chunk's
"text") and `answer` from generation_pipeline.generate_answer(...).

Install:
    pip install boto3 numpy
"""

import json
import logging
import re

import numpy as np

from config import NOVA_MODEL_ID, GENERATION_TEMPERATURE

logging.basicConfig(level=logging.INFO, format="%(asctime)s %(levelname)s %(message)s", datefmt="%H:%M:%S")
log = logging.getLogger(__name__)

JUDGE_MAX_TOKENS = 1024
FINAL_JSON_MARKER = "FINAL_JSON:"


def _call_nova_json(bedrock_client, system_prompt, user_message, max_tokens=JUDGE_MAX_TOKENS):
    """Shared judge call -- same REASONING/FINAL_JSON convention used
    everywhere else in this codebase. Returns the parsed FINAL_JSON
    object. Raises if nothing parseable comes back; callers don't catch
    this, so a bad judge response surfaces as a real error for that
    question rather than silently becoming a wrong 0.0 score."""
    response = bedrock_client.converse(
        modelId=NOVA_MODEL_ID,
        system=[{"text": system_prompt}],
        messages=[{"role": "user", "content": [{"text": user_message}]}],
        inferenceConfig={"temperature": GENERATION_TEMPERATURE, "maxTokens": max_tokens},
    )
    raw_text = response["output"]["message"]["content"][0]["text"]

    if FINAL_JSON_MARKER in raw_text:
        reasoning_part, _, json_part = raw_text.partition(FINAL_JSON_MARKER)
    else:
        reasoning_part, json_part = "", raw_text
    log.info("Judge reasoning: %s", reasoning_part.strip().replace("REASONING:", "").strip())

    match = re.search(r"[\{\[].*[\}\]]", json_part, re.DOTALL)
    if not match:
        raise ValueError(f"No JSON found in judge response: {raw_text!r}")
    return json.loads(match.group(0))


def _cosine_similarity(a, b):
    a, b = np.array(a), np.array(b)
    denom = np.linalg.norm(a) * np.linalg.norm(b)
    return float(np.dot(a, b) / denom) if denom else 0.0


def _format_contexts(contexts):
    return "\n\n".join(f"[{i}] {c}" for i, c in enumerate(contexts, start=1))


# --------------------------------------------------------------------------
# Faithfulness
# --------------------------------------------------------------------------

FAITHFULNESS_SYSTEM_PROMPT = """You are grading whether a generated answer is fully \
supported by the given context, for a bankruptcy/student-loan document assistant. \
Break the answer into its individual factual claims, then judge each claim against \
the context alone -- not outside knowledge, not whether the claim sounds plausible.

Respond in exactly this format, with these two labeled sections and nothing else \
before, between, or after them:

REASONING:
List each factual claim the answer makes. For each, note whether the context \
actually states or directly implies it.

FINAL_JSON:
A single-line JSON object, exactly this shape:
{"claims": [{"claim": "<the claim>", "supported": true | false}, ...]}
If the answer makes no checkable factual claims (e.g. it's a refusal like "the \
documents do not contain this information"), return {"claims": []}.
"""


def faithfulness(bedrock_client, question, contexts, answer):
    """Fraction of the answer's factual claims the context actually
    supports. Returns None (not 0.0) if the answer made no checkable
    claims at all -- e.g. a refusal -- since that's a different situation
    than making claims and getting them wrong."""
    user_message = (
        f"QUESTION:\n{question}\n\n"
        f"CONTEXT:\n{_format_contexts(contexts)}\n\n"
        f"ANSWER:\n{answer}"
    )
    result = _call_nova_json(bedrock_client, FAITHFULNESS_SYSTEM_PROMPT, user_message)
    claims = result.get("claims", [])
    if not claims:
        return {"score": None, "claims": []}
    supported = sum(1 for c in claims if c.get("supported"))
    return {"score": supported / len(claims), "claims": claims}


# --------------------------------------------------------------------------
# Answer relevancy
# --------------------------------------------------------------------------

RELEVANCY_QUESTIONS_SYSTEM_PROMPT = """Given an answer, write 3 different questions \
that this answer would be a good, direct answer to. Write only questions this \
specific answer actually addresses -- don't invent questions about topics the \
answer doesn't cover.

Respond in exactly this format, with these two labeled sections and nothing else \
before, between, or after them:

REASONING:
Note what the answer is actually about, then draft each question.

FINAL_JSON:
A single-line JSON object, exactly this shape:
{"questions": ["<question 1>", "<question 2>", "<question 3>"]}
"""


def answer_relevancy(bedrock_client, embed_fn, question, answer):
    """Nova writes a few questions the answer would suit; each is embedded
    (pass in the same embed_text used to build the FAISS index -- e.g.
    embedding_pipeline.embed_text or embedding_pipeline_v2.embed_text) and
    compared to the real question by cosine similarity. High average
    similarity means the answer is actually shaped like a response to what
    was asked, not just topically nearby or padded with extra detail.
    Returns None if the answer is a refusal Nova can't derive questions
    from."""
    user_message = f"ANSWER:\n{answer}"
    result = _call_nova_json(bedrock_client, RELEVANCY_QUESTIONS_SYSTEM_PROMPT, user_message)
    generated_questions = [q for q in result.get("questions", []) if q.strip()]
    if not generated_questions:
        return {"score": None, "generated_questions": []}

    question_vector = embed_fn(bedrock_client, question)
    similarities = [
        _cosine_similarity(question_vector, embed_fn(bedrock_client, gq))
        for gq in generated_questions
    ]
    return {"score": sum(similarities) / len(similarities), "generated_questions": generated_questions}


# --------------------------------------------------------------------------
# Context precision
# --------------------------------------------------------------------------

CONTEXT_PRECISION_SYSTEM_PROMPT = """You are judging retrieved context chunks for \
relevance to a question, for a bankruptcy/student-loan document assistant. The \
chunks are numbered in the order retrieval ranked them. For each, judge only \
whether it contains information relevant to answering the question -- not whether \
it fully answers it alone.

Respond in exactly this format, with these two labeled sections and nothing else \
before, between, or after them:

REASONING:
Go through the chunks in order and note whether each is relevant to the question.

FINAL_JSON:
A single-line JSON object, exactly this shape:
{"relevance": [true | false, ...]}
The list must have exactly as many entries as there were numbered chunks, in the \
same order.
"""


def context_precision(bedrock_client, question, contexts):
    """Average-precision-style score over the ranked context list: at each
    position where a chunk is relevant, precision-up-to-that-position is
    added to a running total, which is then divided by the number of
    relevant chunks found. This rewards relevant chunks being ranked
    early, not just present somewhere in the list -- a relevant chunk
    buried last scores worse than the same chunk ranked first. Returns
    None if no context was retrieved, or if none of it was judged
    relevant (keeps "nothing relevant found" distinct from a real
    fractional score)."""
    if not contexts:
        return {"score": None, "relevance": []}

    user_message = f"QUESTION:\n{question}\n\nCHUNKS:\n{_format_contexts(contexts)}"
    result = _call_nova_json(bedrock_client, CONTEXT_PRECISION_SYSTEM_PROMPT, user_message)
    relevance = result.get("relevance", [])
    if len(relevance) != len(contexts):
        log.error("context_precision: judge returned %d relevance labels for %d chunks -- padding with False",
                   len(relevance), len(contexts))
        relevance = (relevance + [False] * len(contexts))[:len(contexts)]

    relevant_count = 0
    precision_sum = 0.0
    for i, is_relevant in enumerate(relevance, start=1):
        if is_relevant:
            relevant_count += 1
            precision_sum += relevant_count / i

    if relevant_count == 0:
        return {"score": None, "relevance": relevance}
    return {"score": precision_sum / relevant_count, "relevance": relevance}


# --------------------------------------------------------------------------
# Context recall
# --------------------------------------------------------------------------

CONTEXT_RECALL_SYSTEM_PROMPT = """You are judging whether retrieved context fully \
covers a reference answer, for a bankruptcy/student-loan document assistant. Break \
the reference answer into individual sentences/claims, then judge each one against \
the context alone: could someone verify or derive this claim from the context?

Respond in exactly this format, with these two labeled sections and nothing else \
before, between, or after them:

REASONING:
List each sentence/claim in the reference answer. For each, note whether the \
context supports it.

FINAL_JSON:
A single-line JSON object, exactly this shape:
{"claims": [{"claim": "<the claim>", "attributable": true | false}, ...]}
"""


def context_recall(bedrock_client, contexts, ground_truth):
    """Fraction of the reference (golden dataset expected_answer)'s claims
    that the retrieved context actually supports. Unlike faithfulness
    (generated answer vs. context) and context_precision (context vs.
    question), this checks context against the *correct* answer -- so it
    measures whether retrieval found everything needed, independent of
    whether generation used it well. Returns None if there's no
    ground_truth to check against, matching eval_pipeline.py's
    judge_answer() "N/A" convention for ungraded rows. Returns 0.0 (not
    None) if there IS a ground_truth but zero context was retrieved --
    that's a real failure to measure, not an ungraded row."""
    if not ground_truth or not ground_truth.strip():
        return {"score": None, "claims": []}
    if not contexts:
        return {"score": 0.0, "claims": []}

    user_message = f"REFERENCE ANSWER:\n{ground_truth}\n\nCONTEXT:\n{_format_contexts(contexts)}"
    result = _call_nova_json(bedrock_client, CONTEXT_RECALL_SYSTEM_PROMPT, user_message)
    claims = result.get("claims", [])
    if not claims:
        return {"score": None, "claims": []}
    attributable = sum(1 for c in claims if c.get("attributable"))
    return {"score": attributable / len(claims), "claims": claims}


# --------------------------------------------------------------------------
# Convenience: run all four for one golden-dataset row
# --------------------------------------------------------------------------

def score_all(bedrock_client, embed_fn, question, contexts, answer, ground_truth=""):
    """Runs all four metrics for one question and returns a flat dict of
    scores (each 0.0-1.0 or None) plus each metric's underlying detail --
    the shape an eval file can drop straight into extra xlsx columns per
    row, alongside eval_pipeline.py's existing CORRECT/PARTIALLY_CORRECT/
    INCORRECT verdict. A handful of Nova calls per question (faithfulness,
    relevancy-questions plus up to 3 embed calls, precision, recall) on top
    of generation's own call -- this is eval tooling run offline against a
    golden dataset, not the live query path, so the extra latency/cost is
    an accepted tradeoff for staying on this project's existing Bedrock
    stack instead of pulling in the `ragas` package and its own separate
    LLM/embeddings wiring."""
    faith = faithfulness(bedrock_client, question, contexts, answer)
    relevancy = answer_relevancy(bedrock_client, embed_fn, question, answer)
    precision = context_precision(bedrock_client, question, contexts)
    recall = context_recall(bedrock_client, contexts, ground_truth)

    return {
        "faithfulness": faith["score"],
        "answer_relevancy": relevancy["score"],
        "context_precision": precision["score"],
        "context_recall": recall["score"],
        "details": {
            "faithfulness": faith,
            "answer_relevancy": relevancy,
            "context_precision": precision,
            "context_recall": recall,
        },
    }

GROUND_TRUTH_HEADER_ALIASES = {"expected_answer", "ground_truth", "reference_answer"}


def read_rows(input_path):
    """Reads the input xlsx, matching headers case-insensitively so
    "Question"/"question"/"QUESTION" etc. all work. Returns a list of
    {"question": ..., "model_answer": ..., "ground_truth": ...} dicts --
    ground_truth is "" if no matching column exists in this file."""
    wb = load_workbook(input_path, data_only=True)
    ws = wb.active

    header_row = next(ws.iter_rows(min_row=1, max_row=1, values_only=True))
    headers = [(h or "").strip().lower() for h in header_row]

    if "question" not in headers:
        raise ValueError(f"No 'question' column found in {input_path}. Headers found: {header_row}")
    if "model_answer" not in headers:
        raise ValueError(f"No 'model_answer' column found in {input_path}. Headers found: {header_row}")
    question_col = headers.index("question")
    answer_col = headers.index("model_answer")

    ground_truth_col = next((headers.index(a) for a in GROUND_TRUTH_HEADER_ALIASES if a in headers), None)
    if ground_truth_col is None:
        print("No expected_answer/ground_truth/reference_answer column found -- "
              "context_recall will be None for every row (the other three metrics still run).")

    rows = []
    for row in ws.iter_rows(min_row=2, values_only=True):
        if row[question_col] is None:
            continue
        rows.append({
            "question": str(row[question_col]).strip(),
            "model_answer": str(row[answer_col] or "").strip(),
            "ground_truth": str(row[ground_truth_col]).strip()
                            if ground_truth_col is not None and row[ground_truth_col] else "",
        })
    return rows


def score_rows(bedrock_client, retriever, rows):
    results = []
    for i, row in enumerate(rows, start=1):
        print(f"[{i}/{len(rows)}] {row['question']}")

        # Re-run retrieval+rerank for this question to get the context
        # chunks -- faithfulness/context_precision/context_recall all need
        # them, and the Excel file only has the question and the answer,
        # not what was retrieved when that answer was originally produced.
        fused_chunks = rerank(bedrock_client, retriever, row["question"])
        contexts = [c["text"] for c in fused_chunks]

        scores = ragas_metrics.score_all(
            bedrock_client, embed_text, row["question"], contexts,
            row["model_answer"], row["ground_truth"],
        )

        results.append({
            **row,
            "contexts_used": len(contexts),
            "faithfulness": scores["faithfulness"],
            "answer_relevancy": scores["answer_relevancy"],
            "context_precision": scores["context_precision"],
            "context_recall": scores["context_recall"],
        })
    return results


def write_output(results, output_path):
    wb = Workbook()
    ws = wb.active

    headers = ["Question", "Model Answer", "Ground Truth", "Contexts Used",

    ws.append(headers)
    for cell in ws[1]:
        cell.font = HEADER_FONT

    for r in results:
        ws.append([
            r["question"], r["model_answer"], r["ground_truth"], r["contexts_used"],
            r["faithfulness"], r["answer_relevancy"], r["context_precision"], r["context_recall"],
        ])

    for row in ws.iter_rows(min_row=2):
        for cell in row:
            cell.font = BODY_FONT
            cell.alignment = Alignment(wrap_text=True, vertical="top")

    widths = {"A": 35, "B": 45, "C": 45, "D": 14, "E": 14, "F": 16, "G": 16, "H": 14}
    for col, width in widths.items():
        ws.column_dimensions[col].width = width

    wb.save(output_path)
    print(f"\nSaved {len(results)} scored row(s) to {output_path}")

def check_raw_textract_json(s3_client, doc_id):

        if kvs:
            pages_with_kv += 1
            total_kv += len(kvs)

    print(f"  {key}")
    print(f"  total key_value pairs across all pages: {total_kv} (on {pages_with_kv} page(s))")
    if total_kv == 0:
        print("  PROBLEM (or just a form-less document): Textract's raw output for this")
        print("  document has zero key_values on every page. If FEATURE_TYPES already")
        print("  includes FORMS (layer 1 passed), this means Textract itself didn't detect")
        print("  any key-value pairs on this specific document -- worth opening the PDF and")
        print("  checking whether it actually has recognizable label/value form fields, or")
        print("  re-running Textract on just this doc to rule out a stale/incomplete JSON")
        print("  from an earlier partial run (remember: textract skips already-processed docs).")
    return total_kv


def check_chunks(s3_client, doc_id):
    """Layer 3: did chunking_pipeline.py turn those key_values (if any)
    into a form_fields chunk."""
    key = f"{CHUNKS_PREFIX}/{doc_id}/{doc_id}_chunks.json"
    try:
        chunks = read_json_from_s3(s3_client, S3_BUCKET, key)
    except Exception as e:
        print(f"  Could not read {key}: {e}")
        return None

    form_chunks = [c for c in chunks if c["element_type"] == "form_fields"]
    print(f"  {key}")
    print(f"  form_fields chunks: {len(form_chunks)}")
    for c in form_chunks[:2]:
        print(f"    sample: {c['text'][:200]}")
    return len(form_chunks)


def diagnose_one(s3_client, doc_id):
    print(f"\n--- {doc_id} ---")
    print("Layer 2: raw Textract JSON output")
    total_kv = check_raw_textract_json(s3_client, doc_id)

    print("\nLayer 3: chunking output")
    total_form_chunks = check_chunks(s3_client, doc_id)

    if total_kv and not total_form_chunks:
        print("\n  MISMATCH: Textract found key_values but chunking produced 0 form_fields")
        print("  chunks for this doc -- that WOULD be a real chunking bug. Send me this")
        print("  doc_id and I'll dig into build_chunks_from_json() specifically for it.")
    elif total_kv and total_form_chunks:
        print("\n  OK: key_values found by Textract made it through chunking correctly.")


def main():

#chuncking
"""
v2 -- Chunks Textract output (written by textract_sequential_pipeline.py
v2) into retrieval-ready chunks for embedding, now consuming the new
"text_blocks" field (title/section-header-tagged paragraphs) that v2's
Textract stage added.

The actual fix this file makes over v1: v1's chunker built each body-text
chunk from page["lines"] with zero heading context -- a chunk pulled out
in isolation had no signal about what topic/section it was even from,
which both the AWS Textract Layout team's own benchmark and the AWS
layout-aware-RAG sample repo point to as a real accuracy cost. Here, every
body-text chunk has its title/section header prepended directly into the
text that gets embedded, so the chunk is self-describing even out of
context.

Tables and form key-values are UNCHANGED from v1 -- they still come from
Textractor's dedicated Table/KeyValue extraction (page["tables"] /
page["key_values"]), not from the LAYOUT_TABLE/LAYOUT_KEY_VALUE blocks, and
they are NOT tagged with a heading here either, since v2's Textract stage
deliberately didn't correlate them with which section they fall under
(see extract_text_blocks() in textract_sequential_pipeline.py v2). That's
a known scoping limit, not an oversight -- worth revisiting later if
tables/forms turn out to need the same treatment.

Same CHUNK_SOURCE_FORMAT toggle as v1 (json / md / txt) is kept for
consistency with config.py, but only "json" gets the new heading-aware
treatment -- "md"/"txt" fall back to the same flat build_chunks_from_text()
v1 used, since text_blocks only exists in the JSON output.

Single entrypoint: lambda_handler(event, context). Expects event =
{"source_key": "<s3 key of one Textract output file>"} -- one invocation
chunks one document, same as v1.

No CLI/argparse here on purpose -- for local/manual testing, call
process_document(...) / lambda_handler({"source_key": "..."}) directly.

Install:
    pip install boto3 python-dotenv
"""

import logging

from aws_utils import get_s3_client, id_from_key, read_json_from_s3, read_text_from_s3, write_json_to_s3
from config import (
    S3_BUCKET,
    CHUNKS_PREFIX,
    CHUNK_CHAR_SIZE,
    CHUNK_CHAR_OVERLAP,
    CHUNK_SOURCE_FORMAT,
    SOURCE_FORMAT_SETTINGS,
)

logging.basicConfig(level=logging.INFO, format="%(asctime)s %(levelname)s %(message)s", datefmt="%H:%M:%S")
log = logging.getLogger(__name__)

if CHUNK_SOURCE_FORMAT not in SOURCE_FORMAT_SETTINGS:
    raise ValueError(
        f"config.CHUNK_SOURCE_FORMAT={CHUNK_SOURCE_FORMAT!r} is not one of {list(SOURCE_FORMAT_SETTINGS)}"
    )

SOURCE_PREFIX_FOR_CHUNKING = SOURCE_FORMAT_SETTINGS[CHUNK_SOURCE_FORMAT]["prefix"]
DOC_ID_SUFFIX = SOURCE_FORMAT_SETTINGS[CHUNK_SOURCE_FORMAT]["suffix"]


def split_into_chunks(text, size, overlap):
    """Greedy paragraph-packing splitter -- unchanged from v1. Deterministic,
    character-count based, no tokenizer dependency, no model calls."""
    text = text.strip()
    if not text:
        return []
    if len(text) <= size:
        return [text]

    paragraphs = [p.strip() for p in text.split("\n\n") if p.strip()]
    chunks = []
    current = ""

    for para in paragraphs:
        candidate = f"{current}\n\n{para}".strip() if current else para

        if len(candidate) <= size:
            current = candidate
            continue

        if current:
            chunks.append(current)
            carry = current[-overlap:] if overlap else ""
            current = f"{carry}\n\n{para}".strip() if carry else para
            if len(current) <= size:
                continue

        remainder = current if current else para
        current = ""
        step = max(size - overlap, 1)
        for i in range(0, len(remainder), step):
            piece = remainder[i:i + size]
            if len(piece) <= size:
                chunks.append(piece)

    if current:
        chunks.append(current)

    return chunks


def group_text_blocks(text_blocks):
    """Groups consecutive text_blocks that share the same (title,
    section_header) pair -- text_blocks is already in true reading order
    (see extract_text_blocks() in textract_sequential_pipeline.py v2), so
    this just merges adjacent paragraphs under the same heading into one
    unit before size-based chunking, the same way v1 treated a whole page
    as one candidate unit before splitting -- just scoped to one heading's
    worth of paragraphs instead of a whole page."""
    groups = []
    current_key = None
    current_texts = []

    for block in text_blocks:
        key = (block.get("title"), block.get("section_header"))
        text = (block.get("text") or "").strip()
        if not text:
            continue

        if key != current_key and current_texts:
            groups.append({"title": current_key[0], "section_header": current_key[1],
                            "text": "\n\n".join(current_texts)})
            current_texts = []

        current_key = key
        current_texts.append(text)

    if current_texts:
        groups.append({"title": current_key[0], "section_header": current_key[1],
                        "text": "\n\n".join(current_texts)})

    return groups


def build_heading_prefix(title, section_header):
    """Builds the header text prepended to every chunk under this heading:

        [Title: Borrower's Authorization to Disclose Information]
        [Section: Section A: Borrower Info]

    Missing pieces are omitted, not printed as "None" -- plenty of ECMC's
    shorter documents won't have a detected title or section header at all.
    Returns "" if neither is present, so callers can skip prepending
    entirely rather than adding a blank line for nothing.
    """
    lines = []
    if title:
        lines.append(f"[Title: {title}]")
    if section_header:
        lines.append(f"[Section: {section_header}]")
    return "\n".join(lines)


def build_chunks_from_json(doc_id, source, doc_json):
    """Walks the page-grouped Textract JSON and emits chunk records.
    Chunk numbering is sequential across the whole document, in the
    document's own page order -- not per-page-restarted -- so chunk_id
    stays unique and stable, same as v1.
    """
    chunks = []
    seq = 0

    def next_chunk_id():
        nonlocal seq
        chunk_id = f"{doc_id}_chunk_{seq:03d}"
        seq += 1
        return chunk_id

    for page_entry in doc_json.get("content", []):
        page_num = page_entry.get("page")

        # Body text: text_blocks grouped by heading, chunked to
        # CHUNK_CHAR_SIZE, with the heading prepended to every resulting
        # chunk -- this is the actual fix over v1. Every chunk here is
        # self-describing even pulled out of context, because its own
        # embedded text carries what section it's from.
        for group in group_text_blocks(page_entry.get("text_blocks", [])):
            heading_prefix = build_heading_prefix(group["title"], group["section_header"])
            for piece in split_into_chunks(group["text"], CHUNK_CHAR_SIZE, CHUNK_CHAR_OVERLAP):
                chunk_text = f"{heading_prefix}\n\n{piece}" if heading_prefix else piece
                chunks.append({
                    "chunk_id": next_chunk_id(),
                    "doc_id": doc_id,
                    "source": source,
                    "page": page_num,
                    "element_type": "text",
                    "title": group["title"],
                    "section_header": group["section_header"],
                    "text": chunk_text,
                })

        # Tables: unchanged from v1 -- always one chunk each, never split,
        # not tagged with a heading (see module docstring for why).
        for table_text in page_entry.get("tables", []):
            table_text = table_text.strip()
            if not table_text:
                continue
            chunks.append({
                "chunk_id": next_chunk_id(),
                "doc_id": doc_id,
                "source": source,
                "page": page_num,
                "element_type": "table",
                "title": None,
                "section_header": None,
                "text": table_text,
            })

        # Form key-values: unchanged from v1 -- combined into one
        # supplementary chunk per page, not tagged with a heading.
        key_values = page_entry.get("key_values", [])
        if key_values:
            kv_text = "\n".join(
                f"{kv.get('key', '').strip()}: {kv.get('value', '').strip()}"
                for kv in key_values
                if kv.get("key") or kv.get("value")
            )
            if kv_text.strip():
                chunks.append({
                    "chunk_id": next_chunk_id(),
                    "doc_id": doc_id,
                    "source": source,
                    "page": page_num,
                    "element_type": "form_fields",
                    "title": None,
                    "section_header": None,
                    "text": kv_text,
                })

    return chunks


def build_chunks_from_text(doc_id, source, text):
    """Unchanged from v1 -- used when CHUNK_SOURCE_FORMAT is "md" or "txt".
    Both are a single flat string with no text_blocks/heading structure to
    walk (that only exists in the JSON output), so this just splits the
    whole document and tags every chunk generically."""
    chunks = []
    seq = 0

    def next_chunk_id():
        nonlocal seq
        chunk_id = f"{doc_id}_chunk_{seq:03d}"
        seq += 1
        return chunk_id

    for piece in split_into_chunks(text, CHUNK_CHAR_SIZE, CHUNK_CHAR_OVERLAP):
        chunks.append({
            "chunk_id": next_chunk_id(),
            "doc_id": doc_id,
            "source": source,
            "page": None,
            "element_type": "text",
            "title": None,
            "section_header": None,
            "text": piece,
        })

    return chunks


def process_document(s3_client, source_key):
    doc_id = id_from_key(source_key, DOC_ID_SUFFIX)
    log.info("Chunking %s (%s) [source format: %s]", source_key, doc_id, CHUNK_SOURCE_FORMAT)

    source = f"{doc_id}.pdf"

    if CHUNK_SOURCE_FORMAT == "json":
        doc_json = read_json_from_s3(s3_client, S3_BUCKET, source_key)
        chunks = build_chunks_from_json(doc_id, source, doc_json)

    elif CHUNK_SOURCE_FORMAT == "md":
        text = read_text_from_s3(s3_client, S3_BUCKET, source_key)
        chunks = build_chunks_from_text(doc_id, source, text)

    elif CHUNK_SOURCE_FORMAT == "txt":
        text = read_text_from_s3(s3_client, S3_BUCKET, source_key)
        chunks = build_chunks_from_text(doc_id, source, text)

    else:
        raise ValueError(f"Unhandled CHUNK_SOURCE_FORMAT: {CHUNK_SOURCE_FORMAT!r}")

    log.info("%s -> %d chunks", doc_id, len(chunks))

    out_key = f"{CHUNKS_PREFIX}/{doc_id}/{doc_id}_chunks.json"
    write_json_to_s3(s3_client, S3_BUCKET, out_key, chunks)
    log.info("Saved %s", out_key)
    return {"doc_id": doc_id, "chunk_count": len(chunks), "chunks_key": out_key}


def lambda_handler(event, context=None):
    """event = {"source_key": "<s3 key of one Textract output file>"}.
    Chunks exactly that one document and returns a small JSON-serializable
    summary -- intended to be one state in a Step Functions Map over
    Textract output."""
    s3_client = get_s3_client()
    result = process_document(s3_client, event["source_key"])
    return {"status": "ok", **result}

"""
v2 -- Extracts text, tables, form fields, AND layout structure (title /
section header context) from PDFs stored in S3 using AWS Textract, via the
amazon-textract-textractor library.

This is the v1 file (textract_sequential_pipeline.py) with exactly one
change: build_json() now also walks each page's LAYOUT_TITLE /
LAYOUT_SECTION_HEADER / LAYOUT_TEXT blocks, in true reading order, so every
paragraph of body text is tagged with the title/section header it actually
falls under. v1 was already requesting TextractFeatures.LAYOUT (see
FEATURE_TYPES in config.py) -- Textract was already detecting this
structure, v1's build_json() just never captured it. Everything else
(process_document, lambda_handler, table/key-value extraction) is
unchanged from v1.

Why this matters: per AWS's own benchmark of the LAYOUT feature
(https://aws.amazon.com/blogs/machine-learning/amazon-textracts-new-layout-feature-introduces-efficiencies-in-general-purpose-and-generative-ai-document-processing-tasks/),
giving an LLM layout-aware, heading-tagged text instead of plain
page-linearized text improved their in-context Q&A precision by ~50% on
average (ROUGE) and produced measurable accuracy gains on entity
extraction (ANLS) across two academic datasets. This file is step one of
applying that: capturing the heading structure. chunking_pipeline.py (v2)
is step two: actually prepending that heading to each chunk.

Verified against the installed amazon-textract-textractor==1.9.2 source
(not just the AWS blog's example) -- textractor.entities.page.Page.layouts
returns an EntityList[Layout] of every LAYOUT_* block on the page;
textractor.entities.layout.Layout exposes .reading_order (int) and .text;
textractor.entities.page_layout.PageLayout is the alternative
type-filtered accessor (page.page_layout.titles, .section_headers, etc.)
used in AWS's own sample notebook, but page.layouts + manual sorting by
reading_order is used here instead so title/section-header blocks can be
correlated with the LAYOUT_TEXT blocks that follow them in one single pass
-- PageLayout's per-type lists don't preserve that ordering relationship
across types.

Writes, per document (same three files as v1, JSON content section has
one new field per page -- see build_json() below):
    - {doc_id}.json  -> text-only extraction: lines, tables, key-value
                        pairs, AND (new) text_blocks -- grouped by page.
    - {doc_id}.md    -> document.to_markdown() -- unchanged from v1.
    - {doc_id}.txt   -> document.text -- unchanged from v1.

Single entrypoint: lambda_handler(event, context). Expects event =
{"key": "<s3 key of one source pdf>"} -- one invocation processes one
document. See v1's module docstring for the Step Functions Map / S3
trigger notes -- unchanged here.

Install:
    pip install amazon-textract-textractor boto3 python-dotenv
"""

import logging

from textractor.data.constants import TextractFeatures

from aws_utils import get_s3_client, get_textractor, id_from_key, write_json_to_s3, write_text_to_s3
from config import (
    S3_BUCKET,
    OUTPUT_JSON_PREFIX,
    OUTPUT_MD_PREFIX,
    OUTPUT_TEXT_PREFIX,
    FEATURE_TYPES,
)

logging.basicConfig(level=logging.INFO, format="%(asctime)s %(levelname)s %(message)s", datefmt="%H:%M:%S")
log = logging.getLogger(__name__)

FEATURE_MAP = {
    "LAYOUT": TextractFeatures.LAYOUT,
    "TABLES": TextractFeatures.TABLES,
    "FORMS": TextractFeatures.FORMS,
    "SIGNATURES": TextractFeatures.SIGNATURES,
}
TEXTRACT_FEATURES = [FEATURE_MAP[f] for f in FEATURE_TYPES if f in FEATURE_MAP]

DOC_ID_SUFFIX = ".pdf"

# LAYOUT block types that update the "current heading" state as we walk a
# page in reading order, rather than becoming a chunk-worthy text_block
# themselves.
TITLE_LAYOUT_TYPE = "LAYOUT_TITLE"
SECTION_HEADER_LAYOUT_TYPE = "LAYOUT_SECTION_HEADER"
TEXT_LAYOUT_TYPE = "LAYOUT_TEXT"


def extract_text_blocks(page):
    """Walks every LAYOUT_* block on this page in true reading order
    (page.layouts, sorted by .reading_order -- NOT pre-sorted by the
    library, confirmed against the installed textractor source) and
    returns a list of {"title", "section_header", "text"} dicts, one per
    LAYOUT_TEXT block, carrying whichever title/section header most
    recently preceded it on the page.

    Tables, key-values, and lists are intentionally NOT handled here --
    they keep using the dedicated Table/KeyValue extraction in build_json()
    below, which is more reliable than a LAYOUT_TABLE/LAYOUT_KEY_VALUE
    block's own generic text linearization. LAYOUT_HEADER / LAYOUT_FOOTER /
    LAYOUT_PAGE_NUMBER / LAYOUT_FIGURE are page furniture, not content --
    skipped entirely, matching v1's behavior of never chunking these.
    """
    current_title = None
    current_section_header = None
    text_blocks = []

    for layout_block in sorted(page.layouts, key=lambda b: b.reading_order):
        block_text = (layout_block.text or "").strip()
        if not block_text:
            continue

        if layout_block.layout_type == TITLE_LAYOUT_TYPE:
            current_title = block_text
        elif layout_block.layout_type == SECTION_HEADER_LAYOUT_TYPE:
            current_section_header = block_text
        elif layout_block.layout_type == TEXT_LAYOUT_TYPE:
            text_blocks.append({
                "title": current_title,
                "section_header": current_section_header,
                "text": block_text,
            })

    return text_blocks


def build_json(doc_id, document, pages):
    """Text-only extraction — lines, tables, key-value pairs, and (new)
    title/section-header-tagged text_blocks, grouped by page. No Geometry/
    BoundingBox data, which is what document.response (the raw Textract
    response) would otherwise be full of.

    Grouped by page (rather than one flat list across the whole document)
    so that downstream chunking can attach a page number to every chunk —
    needed for citations later. Page number comes from enumerate() over
    document.pages, not from a Page attribute, since that's guaranteed
    correct regardless of what the library calls it internally.

    "lines"/"tables"/"key_values" are unchanged from v1, kept for backward
    compatibility with v1's chunker if ever needed for comparison.
    "text_blocks" is new -- this is what chunking_pipeline.py (v2) reads
    from to prepend heading context to each chunk.
    """
    pages_out = []
    for page_num, page in enumerate(document.pages, start=1):
        key_values = []
        for kv in page.key_values:
            key_text = " ".join(w.text for w in kv.key) if kv.key else ""
            value_text = kv.value.text if kv.value else ""
            key_values.append({"key": key_text, "value": value_text})

        pages_out.append({
            "page": page_num,
            "lines": [line.text for line in page.lines],
            "tables": [table.text for table in page.tables],
            "key_values": key_values,
            "text_blocks": extract_text_blocks(page),
        })

    return {
        "doc_id": doc_id,
        "pages": pages,
        "content": pages_out,
    }


def process_document(s3_client, extractor, key):
    doc_id = id_from_key(key, DOC_ID_SUFFIX)
    log.info("Processing %s (%s)", key, doc_id)

    document = extractor.start_document_analysis(
        file_source=f"s3://{S3_BUCKET}/{key}",
        features=TEXTRACT_FEATURES,
        save_image=False,
    )

    # start_document_analysis returns a LazyDocument. Accessing any property
    # below triggers Textractor to poll the async Textract job and block
    # until it completes — no manual polling loop needed.
    log.info("Waiting on Textract job for %s...", doc_id)
    text = document.text
    markdown = document.to_markdown()
    n_tables = len(document.tables)
    n_key_values = len(document.key_values)
    pages = len(document.pages)
    log.info("Textract done for %s: %s pages, %d tables, %d key-value pairs",
             doc_id, pages, n_tables, n_key_values)

    write_json_to_s3(s3_client, S3_BUCKET, f"{OUTPUT_JSON_PREFIX}/{doc_id}/{doc_id}.json",
                      build_json(doc_id, document, pages))
    write_text_to_s3(s3_client, S3_BUCKET, f"{OUTPUT_MD_PREFIX}/{doc_id}/{doc_id}.md", markdown)
    write_text_to_s3(s3_client, S3_BUCKET, f"{OUTPUT_TEXT_PREFIX}/{doc_id}/{doc_id}.txt", text)

    log.info("Saved json/md/txt for %s", doc_id)
    return {"doc_id": doc_id, "pages": pages, "tables": n_tables, "key_values": n_key_values}


def lambda_handler(event, context=None):
    """event = {"key": "<s3 key of one source pdf>"}. Processes exactly
    that one document and returns a small JSON-serializable summary --
    intended to be one state in a Step Functions Map over source PDFs."""
    key = event["key"]
    s3_client = get_s3_client()
    extractor = get_textractor()
    result = process_document(s3_client, extractor, key)
    return {"status": "ok", **result}
