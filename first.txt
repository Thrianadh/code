"""
Step 8 -- reranking via Reciprocal Rank Fusion (RRF).

Takes the two independent ranked lists produced by Step 7
(HybridRetriever.search -> "semantic" and "keyword") and combines them into
one fused ranking:

    rrf_score(chunk) = sum, over each list the chunk appears in, of
                        1 / (RRF_K + rank_in_that_list)

RRF instead of a separate reranker model: there is no AWS-native
cross-encoder reranker (unlike, say, Cohere Rerank), so combining the two
rank lists with a fixed formula is the deterministic substitute -- no
extra model call, no extra latency, no extra AWS service to provision.

A chunk that only appears in one of the two lists (matched semantically
but not by keyword, or vice versa) still gets a score from whichever list
it's in -- it isn't penalized to zero for missing from the other.

Usage (standalone, for testing):
    python reranking_pipeline.py --query "what is the deadline to respond to this notice?"

Used as a module by the answer-generation stage (Step 9), which needs the
final fused top-k chunks as the context to answer from.
"""

import argparse
import json
import logging

import boto3

from config import AWS_REGION, RRF_K, FINAL_TOP_K, HYBRID_CANDIDATE_K
from retrieval_pipeline import HybridRetriever

logging.basicConfig(level=logging.INFO, format="%(asctime)s %(levelname)s %(message)s", datefmt="%H:%M:%S")
log = logging.getLogger(__name__)


def reciprocal_rank_fusion(semantic_results, keyword_results, k=RRF_K, top_k=FINAL_TOP_K):
    """Merges two ranked chunk lists into one, by chunk_id, using RRF.
    Returns the fused list sorted by rrf_score descending, truncated to
    top_k."""
    fused = {}

    for chunk in semantic_results:
        chunk_id = chunk["chunk_id"]
        entry = fused.setdefault(chunk_id, dict(chunk))
        entry["rrf_score"] = entry.get("rrf_score", 0.0) + 1.0 / (k + chunk["semantic_rank"])

    for chunk in keyword_results:
        chunk_id = chunk["chunk_id"]
        if chunk_id in fused:
            fused[chunk_id]["bm25_rank"] = chunk["bm25_rank"]
            fused[chunk_id]["bm25_score"] = chunk["bm25_score"]
        else:
            fused[chunk_id] = dict(chunk)
        fused[chunk_id]["rrf_score"] = fused[chunk_id].get("rrf_score", 0.0) + 1.0 / (k + chunk["bm25_rank"])

    ranked = sorted(fused.values(), key=lambda c: c["rrf_score"], reverse=True)
    return ranked[:top_k]


def rerank(bedrock_client, retriever, query, candidate_k=None, top_k=None):
    """Runs Step 7's hybrid search, then fuses the two lists with RRF."""
    candidate_k = candidate_k or HYBRID_CANDIDATE_K
    top_k = top_k or FINAL_TOP_K

    results = retriever.search(bedrock_client, query, top_k=candidate_k)
    return reciprocal_rank_fusion(results["semantic"], results["keyword"], k=RRF_K, top_k=top_k)


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--query", required=True, help="Query text to retrieve and rerank chunks for")
    parser.add_argument("--top-k", type=int, default=FINAL_TOP_K)
    args = parser.parse_args()

    session = boto3.Session(region_name=AWS_REGION)
    s3_client = session.client("s3")
    bedrock_client = session.client("bedrock-runtime")

    retriever = HybridRetriever(s3_client)
    final_chunks = rerank(bedrock_client, retriever, args.query, top_k=args.top_k)

    print(json.dumps(final_chunks, indent=2, default=str))
    log.info("Query %r -> %d fused chunks after RRF", args.query, len(final_chunks))


if __name__ == "__main__":
    main()
