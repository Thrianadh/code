"""
Runs a fixed list of test questions through the full RAG pipeline
(retrieval -> RRF fusion -> Nova generation, Steps 7-9) and records each
result and its latency into an .xlsx file. This is a manual test harness
for exactly this kind of question -- "does the pipeline answer correctly
and cite the right chunk, and how long does it take" -- run across a
batch, rather than one at a time via generation_pipeline.py's CLI.

Two ways to give it questions -- pick whichever is easier in your
environment:

    1. Edit the QUESTIONS list below, directly in this file, and just run:
           python eval_pipeline.py
       This is the simplest option if creating/managing a separate file is
       awkward in your environment.

    2. Put one question per line in a plain text file and pass it in:
           python eval_pipeline.py --questions-file questions.txt
       A --questions-file argument always takes priority over the QUESTIONS
       list below, when given.

Output: an .xlsx file with two sheets:
    - Results  -> one row per question: question, answer, chunk_id,
                  source, excerpt, has_citation, latency_seconds
    - Summary  -> total questions, how many got a citation vs. were
                  refused (Step 10's hallucination control), and
                  average/min/max latency -- all as formulas over the
                  Results sheet, so they recalculate if a row is edited.

Install:
    pip install boto3 python-dotenv openpyxl

Usage:
    python eval_pipeline.py
    python eval_pipeline.py --questions-file questions.txt --output eval_results.xlsx
"""

import argparse
import logging
import time

import boto3
from openpyxl import Workbook
from openpyxl.styles import Font, Alignment

from config import AWS_REGION
from retrieval_pipeline import HybridRetriever
from generation_pipeline import generate_answer

logging.basicConfig(level=logging.INFO, format="%(asctime)s %(levelname)s %(message)s", datefmt="%H:%M:%S")
log = logging.getLogger(__name__)

FONT_NAME = "Arial"
HEADER_FONT = Font(name=FONT_NAME, bold=True)
BODY_FONT = Font(name=FONT_NAME)

# Option 1: paste your questions here, one string per line, and run this
# script with no arguments. Only used if --questions-file is not given (or
# not found). Leave empty to require --questions-file instead.
QUESTIONS = [
    # "What is the deadline to respond to this notice?",
    # "Who is the borrower listed on the authorization form?",
]


def load_questions(path):
    questions = []
    with open(path, "r", encoding="utf-8") as f:
        for line in f:
            line = line.strip()
            if line:
                questions.append(line)
    return questions


def run_questions(bedrock_client, retriever, questions):
    """Calls Step 9's generate_answer() once per question, timing each call
    individually. A failure on one question is recorded as an error row
    and does not stop the rest of the batch."""
    rows = []
    for i, question in enumerate(questions, start=1):
        log.info("[%d/%d] %s", i, len(questions), question)
        start = time.perf_counter()
        error = None
        try:
            result = generate_answer(bedrock_client, retriever, question)
        except Exception as e:
            log.error("Question %d failed: %s", i, e)
            result = {"answer": None, "citation": None}
            error = str(e)
        elapsed = time.perf_counter() - start

        citation = result.get("citation")
        rows.append({
            "question": question,
            "answer": f"ERROR: {error}" if error else result.get("answer"),
            "chunk_id": citation["chunk_id"] if citation else "",
            "source": citation["source"] if citation else "",
            "excerpt": citation["excerpt"] if citation else "",
            "has_citation": "Yes" if citation else "No",
            "latency_seconds": round(elapsed, 3),
        })
    return rows


def write_xlsx(rows, output_path):
    wb = Workbook()
    ws = wb.active
    ws.title = "Results"

    headers = ["#", "Question", "Answer", "Chunk ID", "Source", "Excerpt", "Has Citation", "Latency (s)"]
    ws.append(headers)
    for cell in ws[1]:
        cell.font = HEADER_FONT

    for i, row in enumerate(rows, start=1):
        ws.append([
            i, row["question"], row["answer"], row["chunk_id"],
            row["source"], row["excerpt"], row["has_citation"], row["latency_seconds"],
        ])

    for row in ws.iter_rows(min_row=2):
        for cell in row:
            cell.font = BODY_FONT
            cell.alignment = Alignment(wrap_text=True, vertical="top")

    widths = {"A": 4, "B": 40, "C": 45, "D": 22, "E": 20, "F": 45, "G": 12, "H": 12}
    for col, width in widths.items():
        ws.column_dimensions[col].width = width

    last_row = 1 + len(rows)

    summary = wb.create_sheet("Summary")
    summary_rows = [
        ("Metric", "Value"),
        ("Total questions", f"=COUNTA(Results!B2:B{last_row})"),
        ("Answered with citation", f'=COUNTIF(Results!G2:G{last_row},"Yes")'),
        ("Refused (no citation)", f'=COUNTIF(Results!G2:G{last_row},"No")'),
        ("Average latency (s)", f"=AVERAGE(Results!H2:H{last_row})"),
        ("Min latency (s)", f"=MIN(Results!H2:H{last_row})"),
        ("Max latency (s)", f"=MAX(Results!H2:H{last_row})"),
    ]
    for r in summary_rows:
        summary.append(r)
    for cell in summary[1]:
        cell.font = HEADER_FONT
    for row in summary.iter_rows(min_row=2):
        for cell in row:
            cell.font = BODY_FONT
    summary.column_dimensions["A"].width = 24
    summary.column_dimensions["B"].width = 16

    wb.save(output_path)


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--questions-file", default=None,
                         help="Optional: one question per line. If omitted (or the file "
                              "doesn't exist), the QUESTIONS list in this file is used instead.")
    parser.add_argument("--output", default="eval_results.xlsx")
    args = parser.parse_args()

    if args.questions_file:
        try:
            questions = load_questions(args.questions_file)
        except FileNotFoundError:
            log.error("--questions-file %s not found", args.questions_file)
            return
    else:
        questions = [q.strip() for q in QUESTIONS if q.strip()]

    if not questions:
        log.error(
            "No questions to run. Either paste them into the QUESTIONS list "
            "near the top of eval_pipeline.py, or pass --questions-file."
        )
        return

    log.info("Loaded %d questions", len(questions))

    session = boto3.Session(region_name=AWS_REGION)
    s3_client = session.client("s3")
    bedrock_client = session.client("bedrock-runtime")

    retriever = HybridRetriever(s3_client)
    rows = run_questions(bedrock_client, retriever, questions)

    write_xlsx(rows, args.output)
    log.info("Wrote %s (%d rows)", args.output, len(rows))


if __name__ == "__main__":
    main()
