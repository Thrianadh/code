"""
v2 -- Extracts text, tables, form fields, AND layout structure (title /
section header context) from PDFs stored in S3 using AWS Textract, via the
amazon-textract-textractor library.

This is the v1 file (textract_sequential_pipeline.py) with exactly one
change: build_json() now also walks each page's LAYOUT_TITLE /
LAYOUT_SECTION_HEADER / LAYOUT_TEXT blocks, in true reading order, so every
paragraph of body text is tagged with the title/section header it actually
falls under. v1 was already requesting TextractFeatures.LAYOUT (see
FEATURE_TYPES in config.py) -- Textract was already detecting this
structure, v1's build_json() just never captured it. Everything else
(process_document, lambda_handler, table/key-value extraction) is
unchanged from v1.

Why this matters: per AWS's own benchmark of the LAYOUT feature
(https://aws.amazon.com/blogs/machine-learning/amazon-textracts-new-layout-feature-introduces-efficiencies-in-general-purpose-and-generative-ai-document-processing-tasks/),
giving an LLM layout-aware, heading-tagged text instead of plain
page-linearized text improved their in-context Q&A precision by ~50% on
average (ROUGE) and produced measurable accuracy gains on entity
extraction (ANLS) across two academic datasets. This file is step one of
applying that: capturing the heading structure. chunking_pipeline.py (v2)
is step two: actually prepending that heading to each chunk.

Verified against the installed amazon-textract-textractor==1.9.2 source
(not just the AWS blog's example) -- textractor.entities.page.Page.layouts
returns an EntityList[Layout] of every LAYOUT_* block on the page;
textractor.entities.layout.Layout exposes .reading_order (int) and .text;
textractor.entities.page_layout.PageLayout is the alternative
type-filtered accessor (page.page_layout.titles, .section_headers, etc.)
used in AWS's own sample notebook, but page.layouts + manual sorting by
reading_order is used here instead so title/section-header blocks can be
correlated with the LAYOUT_TEXT blocks that follow them in one single pass
-- PageLayout's per-type lists don't preserve that ordering relationship
across types.

Writes, per document (same three files as v1, JSON content section has
one new field per page -- see build_json() below):
    - {doc_id}.json  -> text-only extraction: lines, tables, key-value
                        pairs, AND (new) text_blocks -- grouped by page.
    - {doc_id}.md    -> document.to_markdown() -- unchanged from v1.
    - {doc_id}.txt   -> document.text -- unchanged from v1.

Single entrypoint: lambda_handler(event, context). Expects event =
{"key": "<s3 key of one source pdf>"} -- one invocation processes one
document. See v1's module docstring for the Step Functions Map / S3
trigger notes -- unchanged here.

Install:
    pip install amazon-textract-textractor boto3 python-dotenv
"""

import logging

from textractor.data.constants import TextractFeatures

from aws_utils import get_s3_client, get_textractor, id_from_key, write_json_to_s3, write_text_to_s3
from config import (
    S3_BUCKET,
    OUTPUT_JSON_PREFIX,
    OUTPUT_MD_PREFIX,
    OUTPUT_TEXT_PREFIX,
    FEATURE_TYPES,
)

logging.basicConfig(level=logging.INFO, format="%(asctime)s %(levelname)s %(message)s", datefmt="%H:%M:%S")
log = logging.getLogger(__name__)

FEATURE_MAP = {
    "LAYOUT": TextractFeatures.LAYOUT,
    "TABLES": TextractFeatures.TABLES,
    "FORMS": TextractFeatures.FORMS,
    "SIGNATURES": TextractFeatures.SIGNATURES,
}
TEXTRACT_FEATURES = [FEATURE_MAP[f] for f in FEATURE_TYPES if f in FEATURE_MAP]

DOC_ID_SUFFIX = ".pdf"

# LAYOUT block types that update the "current heading" state as we walk a
# page in reading order, rather than becoming a chunk-worthy text_block
# themselves.
TITLE_LAYOUT_TYPE = "LAYOUT_TITLE"
SECTION_HEADER_LAYOUT_TYPE = "LAYOUT_SECTION_HEADER"
TEXT_LAYOUT_TYPE = "LAYOUT_TEXT"


def extract_text_blocks(page):
    """Walks every LAYOUT_* block on this page in true reading order
    (page.layouts, sorted by .reading_order -- NOT pre-sorted by the
    library, confirmed against the installed textractor source) and
    returns a list of {"title", "section_header", "text"} dicts, one per
    LAYOUT_TEXT block, carrying whichever title/section header most
    recently preceded it on the page.

    Tables, key-values, and lists are intentionally NOT handled here --
    they keep using the dedicated Table/KeyValue extraction in build_json()
    below, which is more reliable than a LAYOUT_TABLE/LAYOUT_KEY_VALUE
    block's own generic text linearization. LAYOUT_HEADER / LAYOUT_FOOTER /
    LAYOUT_PAGE_NUMBER / LAYOUT_FIGURE are page furniture, not content --
    skipped entirely, matching v1's behavior of never chunking these.
    """
    current_title = None
    current_section_header = None
    text_blocks = []

    for layout_block in sorted(page.layouts, key=lambda b: b.reading_order):
        block_text = (layout_block.text or "").strip()
        if not block_text:
            continue

        if layout_block.layout_type == TITLE_LAYOUT_TYPE:
            current_title = block_text
        elif layout_block.layout_type == SECTION_HEADER_LAYOUT_TYPE:
            current_section_header = block_text
        elif layout_block.layout_type == TEXT_LAYOUT_TYPE:
            text_blocks.append({
                "title": current_title,
                "section_header": current_section_header,
                "text": block_text,
            })

    return text_blocks


def build_json(doc_id, document, pages):
    """Text-only extraction — lines, tables, key-value pairs, and (new)
    title/section-header-tagged text_blocks, grouped by page. No Geometry/
    BoundingBox data, which is what document.response (the raw Textract
    response) would otherwise be full of.

    Grouped by page (rather than one flat list across the whole document)
    so that downstream chunking can attach a page number to every chunk —
    needed for citations later. Page number comes from enumerate() over
    document.pages, not from a Page attribute, since that's guaranteed
    correct regardless of what the library calls it internally.

    "lines"/"tables"/"key_values" are unchanged from v1, kept for backward
    compatibility with v1's chunker if ever needed for comparison.
    "text_blocks" is new -- this is what chunking_pipeline.py (v2) reads
    from to prepend heading context to each chunk.
    """
    pages_out = []
    for page_num, page in enumerate(document.pages, start=1):
        key_values = []
        for kv in page.key_values:
            key_text = " ".join(w.text for w in kv.key) if kv.key else ""
            value_text = kv.value.text if kv.value else ""
            key_values.append({"key": key_text, "value": value_text})

        pages_out.append({
            "page": page_num,
            "lines": [line.text for line in page.lines],
            "tables": [table.text for table in page.tables],
            "key_values": key_values,
            "text_blocks": extract_text_blocks(page),
        })

    return {
        "doc_id": doc_id,
        "pages": pages,
        "content": pages_out,
    }


def process_document(s3_client, extractor, key):
    doc_id = id_from_key(key, DOC_ID_SUFFIX)
    log.info("Processing %s (%s)", key, doc_id)

    document = extractor.start_document_analysis(
        file_source=f"s3://{S3_BUCKET}/{key}",
        features=TEXTRACT_FEATURES,
        save_image=False,
    )

    # start_document_analysis returns a LazyDocument. Accessing any property
    # below triggers Textractor to poll the async Textract job and block
    # until it completes — no manual polling loop needed.
    log.info("Waiting on Textract job for %s...", doc_id)
    text = document.text
    markdown = document.to_markdown()
    n_tables = len(document.tables)
    n_key_values = len(document.key_values)
    pages = len(document.pages)
    log.info("Textract done for %s: %s pages, %d tables, %d key-value pairs",
             doc_id, pages, n_tables, n_key_values)

    write_json_to_s3(s3_client, S3_BUCKET, f"{OUTPUT_JSON_PREFIX}/{doc_id}/{doc_id}.json",
                      build_json(doc_id, document, pages))
    write_text_to_s3(s3_client, S3_BUCKET, f"{OUTPUT_MD_PREFIX}/{doc_id}/{doc_id}.md", markdown)
    write_text_to_s3(s3_client, S3_BUCKET, f"{OUTPUT_TEXT_PREFIX}/{doc_id}/{doc_id}.txt", text)

    log.info("Saved json/md/txt for %s", doc_id)
    return {"doc_id": doc_id, "pages": pages, "tables": n_tables, "key_values": n_key_values}


def lambda_handler(event, context=None):
    """event = {"key": "<s3 key of one source pdf>"}. Processes exactly
    that one document and returns a small JSON-serializable summary --
    intended to be one state in a Step Functions Map over source PDFs."""
    key = event["key"]
    s3_client = get_s3_client()
    extractor = get_textractor()
    result = process_document(s3_client, extractor, key)
    return {"status": "ok", **result}

#   "txt"  -> flat plain text, no structure at all

CHUNK_SOURCE_FORMAT = "json"

# Maps CHUNK_SOURCE_FORMAT -> which S3 prefix/file extension
# chunking_pipeline.py reads from. Defined here, next to the three
# OUTPUT_*_PREFIX values above, rather than in chunking_pipeline.py, so

SOURCE_FORMAT_SETTINGS = {
    "json": {"prefix": OUTPUT_JSON_PREFIX, "suffix": ".json"},
    "md":   {"prefix": OUTPUT_MD_PREFIX,   "suffix": ".md"},
    "txt":  {"prefix": OUTPUT_TEXT_PREFIX, "suffix": ".txt"},
}
