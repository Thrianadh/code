# ---------------------------------------------------------------------------
# Dataclasses for structured output
# ---------------------------------------------------------------------------

@dataclass
class TextElement:
    """Represents a single text element with metadata."""
    text: str
    block_type: str
    page: int
    confidence: float
    bbox: Dict[str, float]
    block_id: str
    is_heading: bool = False
    heading_level: int = 0


@dataclass
class TableCell:
    """Represents a table cell."""
    text: str
    row_index: int
    column_index: int
    row_span: int = 1
    column_span: int = 1
    is_header: bool = False
    confidence: float = 0.0


@dataclass
class TableElement:
    """Represents a parsed table."""
    cells: List[TableCell] = field(default_factory=list)
    page: int = 1
    bbox: Dict[str, float] = field(default_factory=dict)
    block_id: str = ""
    column_count: int = 0


@dataclass
class ParsedDocument:
    """Structured document output."""
    elements: List[Any] = field(default_factory=list)
    plain_text: str = ""
    markdown_text: str = ""
    metadata: Dict[str, Any] = field(default_factory=dict)

    def to_dict(self) -> Dict[str, Any]:
        """Serialize to a plain dict (JSON-friendly)."""
        return {
            "metadata": self.metadata,
            "plain_text": self.plain_text,
            "markdown_text": self.markdown_text,
            "elements": [
                {
                    "type": type(e).__name__,
                    **asdict(e),
                }
                for e in self.elements
            ],
        }

class S3Manager:
    """Handles all S3 operations (upload, download, list, validate, put_string)."""

    def __init__(self, bucket_name: str, region: str = "us-east-1"):
        self.bucket_name = bucket_name
        self.s3_client = boto3.client("s3", region_name=region)

    def upload_file(
        self,
        local_file_path: str,
        s3_key: str,
        extra_args: Optional[Dict[str, Any]] = None,
    ) -> bool:
        try:
            self.s3_client.upload_file(
                local_file_path,
                self.bucket_name,
                s3_key,
                ExtraArgs=extra_args or {},
            )
            logger.info(f"Uploaded {local_file_path} → s3://{self.bucket_name}/{s3_key}")
            return True
        except (ClientError, BotoCoreError) as e:
            logger.error(f"Failed to upload {local_file_path}: {e}")
            return False

    def download_file(self, s3_key: str, local_file_path: str) -> bool:
        try:
            self.s3_client.download_file(self.bucket_name, s3_key, local_file_path)
            logger.info(f"Downloaded s3://{self.bucket_name}/{s3_key} → {local_file_path}")
            return True
        except (ClientError, BotoCoreError) as e:
            logger.error(f"Failed to download {s3_key}: {e}")
            return False

    def put_string(self, s3_key: str, content: str, content_type: str = "text/plain; charset=utf-8") -> bool:
        """Save a string directly to S3."""
        try:
            self.s3_client.put_object(
                Bucket=self.bucket_name,
                Key=s3_key,
                Body=content.encode("utf-8"),
                ContentType=content_type,
            )
            logger.info(f"Saved string → s3://{self.bucket_name}/{s3_key}")
            return True
        except (ClientError, BotoCoreError) as e:
            logger.error(f"Failed to save string to {s3_key}: {e}")
            return False

    def get_string(self, s3_key: str) -> Optional[str]:
        """Read a string directly from S3."""
        try:
            response = self.s3_client.get_object(Bucket=self.bucket_name, Key=s3_key)
            return response["Body"].read().decode("utf-8")
        except (ClientError, BotoCoreError) as e:
            logger.error(f"Failed to read string from {s3_key}: {e}")
            return None

    def list_objects(self, prefix: str = "") -> List[str]:
        try:
            paginator = self.s3_client.get_paginator("list_objects_v2")
            objects = []
            for page in paginator.paginate(Bucket=self.bucket_name, Prefix=prefix):
                objects.extend([obj["Key"] for obj in page.get("Contents", [])])
            return objects
        except (ClientError, BotoCoreError) as e:
            logger.error(f"Failed to list objects: {e}")
            return []

    def validate_path(self, s3_key: str) -> bool:
        try:
            self.s3_client.head_object(Bucket=self.bucket_name, Key=s3_key)
            return True
        except (ClientError, BotoCoreError):
            return False

class TextractManager:
    """Handles Amazon Textract operations with enhanced post-processing."""

    def __init__(self, region: str = "us-east-1"):
        self.textract_client = boto3.client("textract", region_name=region)

    def start_document_analysis(
        self,
        s3_bucket: str,
        s3_key: str,
        features: Optional[List[str]] = None,
        notification_channel: Optional[Dict[str, str]] = None,
    ) -> str:
        """
        Start a Textract document analysis job.

        Args:
            s3_bucket: S3 bucket name containing the document.
            s3_key: S3 key of the document to analyze.
            features: List of Textract features (FORMS, TABLES, LAYOUT).
            notification_channel: Optional SNS topic ARN for job completion.
                Example: {"SNSTopicArn": "arn:aws:sns:...", "RoleArn": "arn:aws:iam::..."}

        Returns:
            str: Job ID for tracking.
        """
        features = features or ["TABLES", "FORMS", "LAYOUT"]
        try:
            kwargs = {
                "DocumentLocation": {
                    "S3Object": {"Bucket": s3_bucket, "Name": s3_key}
                },
                "FeatureTypes": features,
            }
            if notification_channel:
                kwargs["NotificationChannel"] = notification_channel

            response = self.textract_client.start_document_analysis(**kwargs)
            job_id = response["JobId"]
            logger.info(f"Started Textract job {job_id} for s3://{s3_bucket}/{s3_key}")
            return job_id
        except (ClientError, BotoCoreError) as e:
            logger.error(f"Failed to start Textract job for {s3_key}: {e}")
            raise

    def get_job_status(self, job_id: str) -> str:
        try:
            response = self.textract_client.get_document_analysis(JobId=job_id)
            return response["JobStatus"]
        except (ClientError, BotoCoreError) as e:
            logger.error(f"Failed to get job status for {job_id}: {e}")
            raise

    def wait_for_job_completion(
        self, job_id: str, max_retries: int = 60, sleep_sec: int = 5
    ) -> bool:
        import time
        for attempt in range(max_retries):
            status = self.get_job_status(job_id)
            if status == "SUCCEEDED":
                logger.info(f"Textract job {job_id} succeeded.")
                return True
            elif status == "FAILED":
                logger.error(f"Textract job {job_id} failed.")
                return False
            logger.info(f"Job {job_id} status: {status}. Retry {attempt + 1}/{max_retries}...")
            time.sleep(sleep_sec)
        logger.error(f"Textract job {job_id} timed out after {max_retries} retries.")
        return False

    def get_job_results(self, job_id: str) -> List[Dict[str, Any]]:
        """Retrieve ALL results from a completed Textract job with pagination."""
        blocks = []
        next_token = None
        try:
            while True:
                kwargs = {"JobId": job_id}
                if next_token:
                    kwargs["NextToken"] = next_token
                response = self.textract_client.get_document_analysis(**kwargs)
                blocks.extend(response.get("Blocks", []))
                next_token = response.get("NextToken")
                if not next_token:
                    break
            logger.info(f"Retrieved {len(blocks)} blocks for job {job_id}")
            return blocks
        except (ClientError, BotoCoreError) as e:
            logger.error(f"Failed to get results for job {job_id}: {e}")
            raise

    # -----------------------------------------------------------------------
    # Enhanced Post-Processing
    # -----------------------------------------------------------------------

    def parse_document(self, blocks: List[Dict[str, Any]], document_id: str = "") -> ParsedDocument:
        """
        Parse Textract blocks into a structured document with proper reading order,
        paragraph grouping, heading detection, and table reconstruction.
        """
        block_map = {b["Id"]: b for b in blocks}
        blocks_by_page = defaultdict(list)
        for block in blocks:
            page = block.get("Page", 1)
            blocks_by_page[page].append(block)

        elements = []
        for page_num in sorted(blocks_by_page.keys()):
            page_blocks_list = blocks_by_page[page_num]
            lines = [b for b in page_blocks_list if b["BlockType"] == "LINE"]
            tables = [b for b in page_blocks_list if b["BlockType"] == "TABLE"]

            ordered_lines = self._reconstruct_reading_order(lines)
            heading_ids = self._detect_headings(ordered_lines)
            paragraphs = self._group_paragraphs(ordered_lines, heading_ids)

            table_elements = []
            for table_block in tables:
                table_elem = self._parse_table_enhanced(table_block, block_map)
                table_elem.page = page_num
                table_elements.append(table_elem)

            page_elements = self._merge_elements(paragraphs, table_elements, page_num)
            elements.extend(page_elements)

        plain_text = self._generate_plain_text(elements)
        markdown_text = self._generate_markdown(elements, document_id=document_id)

        metadata = {
            "document_id": document_id,
            "total_pages": max(blocks_by_page.keys()) if blocks_by_page else 0,
            "total_blocks": len(blocks),
            "total_lines": len([b for b in blocks if b["BlockType"] == "LINE"]),
            "total_tables": len([b for b in blocks if b["BlockType"] == "TABLE"]),
        }

        return ParsedDocument(
            elements=elements,
            plain_text=plain_text,
            markdown_text=markdown_text,
            metadata=metadata,
        )

    def _reconstruct_reading_order(self, lines: List[Dict[str, Any]]) -> List[Dict[str, Any]]:
        if not lines:
            return []
        lines_sorted = sorted(
            lines,
            key=lambda x: x.get("Geometry", {}).get("BoundingBox", {}).get("Top", 0),
        )
        bands = []
        current_band = [lines_sorted[0]]
        band_threshold = 0.02

        for line in lines_sorted[1:]:
            prev_top = current_band[-1].get("Geometry", {}).get("BoundingBox", {}).get("Top", 0)
            curr_top = line.get("Geometry", {}).get("BoundingBox", {}).get("Top", 0)
            if abs(curr_top - prev_top) <= band_threshold:
                current_band.append(line)
            else:
                bands.append(
                    sorted(current_band, key=lambda x: x.get("Geometry", {}).get("BoundingBox", {}).get("Left", 0))
                )
                current_band = [line]
        if current_band:
            bands.append(
                sorted(current_band, key=lambda x: x.get("Geometry", {}).get("BoundingBox", {}).get("Left", 0))
            )

        ordered = []
        for band in bands:
            ordered.extend(band)
        return ordered

    def _detect_headings(self, lines: List[Dict[str, Any]]) -> Set[str]:
        heading_ids = set()
        if not lines:
            return heading_ids

        heights = []
        for line in lines:
            bbox = line.get("Geometry", {}).get("BoundingBox", {})
            heights.append(bbox.get("Height", 0))
        avg_height = sum(heights) / len(heights) if heights else 0.01

        for i, line in enumerate(lines):
            text = line.get("Text", "").strip()
            if not text:
                continue
            bbox = line.get("Geometry", {}).get("BoundingBox", {})
            height = bbox.get("Height", 0)

            is_short = len(text) < 100
            is_tall = height > avg_height * 1.3
            is_all_caps = text.isupper() and len(text) > 3
            is_title_case = text.istitle() and len(text.split()) <= 10

            is_prominent = False
            if i == 0:
                is_prominent = True
            elif i > 0:
                prev_bbox = lines[i - 1].get("Geometry", {}).get("BoundingBox", {})
                curr_top = bbox.get("Top", 0)
                prev_bottom = prev_bbox.get("Top", 0) + prev_bbox.get("Height", 0)
                if curr_top - prev_bottom > avg_height * 2:
                    is_prominent = True

            score = sum([is_short, is_tall, is_all_caps, is_title_case, is_prominent])
            if score >= 3 or (is_all_caps and is_short) or (is_tall and is_prominent):
                heading_ids.add(line["Id"])
        return heading_ids

    def _group_paragraphs(
        self, lines: List[Dict[str, Any]], heading_ids: Set[str]
    ) -> List[TextElement]:
        if not lines:
            return []

        paragraphs = []
        current_para_lines = [lines[0]]

        for i in range(1, len(lines)):
            prev_line = lines[i - 1]
            curr_line = lines[i]

            prev_bbox = prev_line.get("Geometry", {}).get("BoundingBox", {})
            curr_bbox = curr_line.get("Geometry", {}).get("BoundingBox", {})

            prev_bottom = prev_bbox.get("Top", 0) + prev_bbox.get("Height", 0)
            curr_top = curr_bbox.get("Top", 0)
            vertical_gap = curr_top - prev_bottom

            prev_left = prev_bbox.get("Left", 0)
            curr_left = curr_bbox.get("Left", 0)
            horizontal_diff = abs(curr_left - prev_left)

            line_height = prev_bbox.get("Height", 0.01)
            para_gap_threshold = line_height * 1.8

            is_new_para = (
                vertical_gap > para_gap_threshold or horizontal_diff > 0.05
            )

            # Also break if current line is a heading
            if curr_line["Id"] in heading_ids:
                is_new_para = True

            if is_new_para:
                para_text = " ".join(l.get("Text", "") for l in current_para_lines)
                first_line = current_para_lines[0]
                bbox = first_line.get("Geometry", {}).get("BoundingBox", {})
                is_heading = first_line["Id"] in heading_ids

                paragraphs.append(TextElement(
                    text=para_text,
                    block_type="HEADING" if is_heading else "PARAGRAPH",
                    page=first_line.get("Page", 1),
                    confidence=min(l.get("Confidence", 99) for l in current_para_lines),
                    bbox=bbox,
                    block_id=first_line["Id"],
                    is_heading=is_heading,
                    heading_level=1 if is_heading else 0,
                ))
                current_para_lines = [curr_line]
            else:
                current_para_lines.append(curr_line)

        if current_para_lines:
            para_text = " ".join(l.get("Text", "") for l in current_para_lines)
            first_line = current_para_lines[0]
            bbox = first_line.get("Geometry", {}).get("BoundingBox", {})
            is_heading = first_line["Id"] in heading_ids
            paragraphs.append(TextElement(
                text=para_text,
                block_type="HEADING" if is_heading else "PARAGRAPH",
                page=first_line.get("Page", 1),
                confidence=min(l.get("Confidence", 99) for l in current_para_lines),
                bbox=bbox,
                block_id=first_line["Id"],
                is_heading=is_heading,
                heading_level=1 if is_heading else 0,
            ))

        return paragraphs

    def _parse_table_enhanced(
        self, table_block: Dict[str, Any], block_map: Dict[str, Dict[str, Any]]
    ) -> TableElement:
        cells = []
        cell_blocks = []
        for rel in table_block.get("Relationships", []):
            if rel["Type"] == "CHILD":
                for cell_id in rel["Ids"]:
                    cell = block_map.get(cell_id)
                    if cell and cell["BlockType"] == "CELL":
                        cell_blocks.append(cell)

        max_col = 0
        for cell in cell_blocks:
            row_idx = cell.get("RowIndex", 0)
            col_idx = cell.get("ColumnIndex", 0)
            row_span = cell.get("RowSpan", 1)
            col_span = cell.get("ColumnSpan", 1)

            cell_texts = []
            for rel in cell.get("Relationships", []):
                if rel["Type"] == "CHILD":
                    for child_id in rel["Ids"]:
                        child = block_map.get(child_id)
                        if child and child["BlockType"] in ("WORD", "LINE", "SELECTION_ELEMENT"):
                            if child["BlockType"] == "SELECTION_ELEMENT":
                                status = "[x]" if child.get("SelectionStatus") == "SELECTED" else "[ ]"
                                cell_texts.append(status)
                            else:
                                cell_texts.append(child.get("Text", ""))

            text = " ".join(cell_texts).strip()
            max_col = max(max_col, col_idx + col_span - 1)
            is_header = row_idx == 1

            cells.append(TableCell(
                text=text,
                row_index=row_idx,
                column_index=col_idx,
                row_span=row_span,
                column_span=col_span,
                is_header=is_header,
                confidence=cell.get("Confidence", 0),
            ))

        return TableElement(
            cells=cells,
            page=table_block.get("Page", 1),
            bbox=table_block.get("Geometry", {}).get("BoundingBox", {}),
            block_id=table_block["Id"],
            column_count=max_col,
        )

    def _merge_elements(
        self, paragraphs: List[TextElement], tables: List[TableElement], page_num: int
    ) -> List[Any]:
        all_elements = []
        for para in paragraphs:
            all_elements.append(("paragraph", para.bbox.get("Top", 0), para))
        for table in tables:
            all_elements.append(("table", table.bbox.get("Top", 0), table))
        all_elements.sort(key=lambda x: x[1])
        return [elem for _, _, elem in all_elements]

    def _generate_plain_text(self, elements: List[Any]) -> str:
        parts = []
        for elem in elements:
            if isinstance(elem, TextElement):
                parts.append(elem.text)
            elif isinstance(elem, TableElement):
                parts.append("[TABLE]")
        return "\n\n".join(parts)

    def _generate_markdown(self, elements: List[Any], document_id: str = "") -> str:
        """Generate Markdown with YAML frontmatter and HTML page markers."""
        lines = []
        # YAML frontmatter
        lines.append("---")
        lines.append(f'document_id: "{document_id}"')
        lines.append(f"total_pages: {max((e.page for e in elements if hasattr(e, 'page')), default=0)}")
        lines.append("---")
        lines.append("")

        current_page = 0
        for elem in elements:
            page = getattr(elem, "page", 1)
            if page != current_page:
                lines.append(f"<!-- page: {page} -->")
                current_page = page

            if isinstance(elem, TextElement):
                if elem.is_heading:
                    level = min(elem.heading_level + 1, 6)  # # or ##
                    lines.append(f"{'#' * level} {elem.text}")
                else:
                    lines.append(elem.text)
            elif isinstance(elem, TableElement):
                lines.append(self._table_to_markdown(elem))
            lines.append("")

        return "\n".join(lines)

    def _table_to_markdown(self, table: TableElement) -> str:
        if not table.cells:
            return ""
        rows = defaultdict(list)
        for cell in table.cells:
            rows[cell.row_index].append(cell)

        md_lines = []
        for row_idx in sorted(rows.keys()):
            row_cells = sorted(rows[row_idx], key=lambda c: c.column_index)
            cell_texts = [c.text.replace("|", "\|").replace("\n", " ") for c in row_cells]
            md_lines.append("| " + " | ".join(cell_texts) + " |")
            if row_idx == 1:
                md_lines.append("|" + "|".join([" --- "] * len(row_cells)) + "|")
        return "\n".join(md_lines)

    @staticmethod
    def parse_blocks(blocks: List[Dict[str, Any]]) -> Tuple[str, str]:
        manager = TextractManager()
        doc = manager.parse_document(blocks)
        return doc.plain_text, doc.markdown_text

"""
Main pipeline: Discover all documents in S3, run Textract analysis,
post-process with enhanced parsing, and save structured outputs back to S3.
Supports both sequential and parallel execution.
"""

from config import config

    level=logging.INFO,
    format="%(asctime)s - %(name)s - %(levelname)s - %(message)s",

logger = logging.getLogger(__name__)


class TextractPipeline:
    """End-to-end pipeline for batch Textract processing."""

    def __init__(self):
        self.s3 = S3Manager(
            bucket_name=config.S3_BUCKET_NAME,
            region=config.AWS_REGION,
        )
        self.textract = TextractManager(region=config.AWS_REGION)

    def discover_documents(self, prefix: Optional[str] = None) -> List[str]:
        """List all supported documents under the source S3 prefix."""
        prefix = prefix or config.S3_SOURCE_PREFIX
        all_keys = self.s3.list_objects(prefix=prefix)
        docs = [
            key for key in all_keys
