#image_extractor.py
"""Extract figure images from documents and generate captions via Nova Lite multimodal.

Pipeline:
  1. Read Textract output to find LAYOUT_FIGURE blocks with geometry
  2. Download original source file from S3
  3. Crop image region using bounding box (PIL for images, optional for PDFs)
  4. Send cropped image to Nova Lite (multimodal Converse API) for caption
  5. Store:
     - cropped images in S3: textract_output/{doc_id}/images/
     - image_captions.json in 02_json/{doc_id}/

Nova Lite prompt (from config.paths.PROMPTS.IMAGE_CAPTION):
  "Describe this figure or chart and its key data points in 2-3 sentences."

Note: For PDF source files, image cropping requires additional libraries
(e.g., PyMuPDF). If unavailable, the extractor logs a warning and stores
placeholder metadata for manual processing.
"""

import io
import json
import os
from typing import Any, Dict, List, Optional, Tuple

from botocore.exceptions import ClientError

from config.settings import load_settings
from utils.aws_clients import get_bedrock_runtime_client, get_s3_client
from utils.s3_utils import download_json, upload_json, download_bytes
from utils.logging_config import get_logger, log_operation
from utils.decorators import retry, timing

logger = get_logger("image_extractor")


class ImageExtractor:
    """Extract and caption figures detected by Textract LAYOUT analysis."""

    def __init__(self):
        self.settings = load_settings()
        self.bucket = self.settings.S3_BUCKET
        self.bedrock = get_bedrock_runtime_client()
        self.s3 = get_s3_client()
        self.chat_model = self.settings.BEDROCK_CHAT_MODEL_ID
        self.prompt = self.settings.prompts.IMAGE_CAPTION

    # ── Public API ─────────────────────────────────────────────────────────

    @timing()
    def extract_and_caption(self, doc_id: str, source_s3_key: str) -> str:
        """Extract all figures from a document and caption with Nova Lite.

        Args:
            doc_id: Document identifier.
            source_s3_key: S3 key of the original source file (PDF or image).

        Returns:
            S3 key of image_captions.json.
        """
        # Read Textract output
        textract_key = self.settings.s3_paths.textract_doc_path(doc_id) + "layout.json"
        textract_data = download_json(self.bucket, textract_key)

        if not textract_data or "Blocks" not in textract_data:
            raise ValueError(f"Invalid Textract data for {doc_id}")

        blocks = textract_data["Blocks"]

        # Find LAYOUT_FIGURE blocks
        figure_blocks = [
            b for b in blocks
            if b.get("BlockType") == "LAYOUT_FIGURE"
        ]

        if not figure_blocks:
            logger.info("No figures found in document", extra={"doc_id": doc_id})
            captions_output = {"doc_id": doc_id, "figures": []}
            captions_key = (
                self.settings.s3_paths.json_doc_path(doc_id) + "image_captions.json"
            )
            upload_json(self.bucket, captions_key, captions_output)
            return captions_key

        # Determine source file type
        file_ext = os.path.splitext(source_s3_key)[1].lower()
        is_pdf = file_ext == ".pdf"
        is_image = file_ext in (".png", ".jpg", ".jpeg", ".gif", ".bmp", ".tiff")

        # Download source file to local temp
        source_bytes = download_bytes(self.bucket, source_s3_key)

        captions_output = {"doc_id": doc_id, "figures": []}

        for i, figure_block in enumerate(figure_blocks, start=1):
            figure_id = f"{doc_id}_figure_{i:03d}"
            page = figure_block.get("Page", 1)
            bbox = figure_block.get("Geometry", {}).get("BoundingBox", {})

            figure_record = {
                "figure_id": figure_id,
                "textract_block_id": figure_block["Id"],
                "page_number": page,
                "bounding_box": bbox,
                "confidence": figure_block.get("Confidence", 0.0),
            }

            # Attempt to crop and caption
            cropped_bytes = None
            if is_image:
                cropped_bytes = self._crop_from_image(source_bytes, bbox)
            elif is_pdf:
                cropped_bytes = self._crop_from_pdf(source_bytes, page, bbox)
            else:
                logger.warning(
                    "Unsupported source type for image cropping",
                    extra={"doc_id": doc_id, "figure_id": figure_id, "ext": file_ext},
                )

            if cropped_bytes:
                # Save cropped image to S3
                image_key = (
                    self.settings.s3_paths.textract_doc_path(doc_id)
                    + f"images/{figure_id}.png"
                )
                self.s3.put_object(
                    Bucket=self.bucket,
                    Key=image_key,
                    Body=cropped_bytes,
                    ContentType="image/png",
                )
                figure_record["s3_image_key"] = image_key

                # Caption with Nova Lite multimodal (with graceful fallback)
                caption = self._caption_image_safe(cropped_bytes)
                figure_record["caption"] = caption

                logger.info(
                    "Figure captioned",
                    extra={
                        "doc_id": doc_id,
                        "figure_id": figure_id,
                        "page": page,
                        "image_key": image_key,
                        "caption_length": len(caption),
                    },
                )
            else:
                figure_record["caption"] = (
                    f"[Figure on page {page} — image extraction not available]"
                )
                figure_record["s3_image_key"] = None
                logger.warning(
                    "Could not extract figure image",
                    extra={"doc_id": doc_id, "figure_id": figure_id, "page": page},
                )

            captions_output["figures"].append(figure_record)

        # Upload captions
        captions_key = (
            self.settings.s3_paths.json_doc_path(doc_id) + "image_captions.json"
        )
        upload_json(self.bucket, captions_key, captions_output)

        logger.info(
            "Image extraction complete",
            extra={
                "doc_id": doc_id,
                "figure_count": len(captions_output["figures"]),
                "captioned": sum(
                    1 for f in captions_output["figures"]
                    if f.get("caption") and "not available" not in f["caption"]
                ),
            },
        )

        return captions_key

    # ── Internal: Image Cropping ───────────────────────────────────────────

    def _crop_from_image(self, image_bytes: bytes, bbox: Dict[str, float]) -> Optional[bytes]:
        """Crop a region from an image using PIL.

        Args:
            image_bytes: Raw image bytes (PNG/JPG).
            bbox: Textract BoundingBox dict with Top, Left, Width, Height (0-1 normalized).

        Returns:
            Cropped PNG bytes, or None if PIL unavailable or error.
        """
        try:
            from PIL import Image
        except ImportError:
            logger.warning("PIL not available, skipping image crop")
            return None

        try:
            img = Image.open(io.BytesIO(image_bytes))
            width, height = img.size

            # Convert normalized bbox to pixel coordinates
            left = int(bbox.get("Left", 0) * width)
            top = int(bbox.get("Top", 0) * height)
            right = int((bbox.get("Left", 0) + bbox.get("Width", 0)) * width)
            bottom = int((bbox.get("Top", 0) + bbox.get("Height", 0)) * height)

            # Ensure valid coordinates
            left = max(0, left)
            top = max(0, top)
            right = min(width, right)
            bottom = min(height, bottom)

            if right <= left or bottom <= top:
                logger.warning("Invalid crop coordinates", extra={"bbox": bbox})
                return None

            cropped = img.crop((left, top, right, bottom))

            output = io.BytesIO()
            cropped.save(output, format="PNG")
            return output.getvalue()

        except Exception as exc:
            logger.warning("Image crop failed", extra={"error": str(exc)})
            return None

    def _crop_from_pdf(
        self,
        pdf_bytes: bytes,
        page_number: int,
        bbox: Dict[str, float],
    ) -> Optional[bytes]:
        """Crop a region from a PDF page.

        Requires PyMuPDF (fitz). If unavailable, returns None.

        Args:
            pdf_bytes: Raw PDF bytes.
            page_number: 1-based page number.
            bbox: Textract BoundingBox dict (0-1 normalized).

        Returns:
            Cropped PNG bytes, or None if library unavailable.
        """
        try:
            import fitz  # PyMuPDF
        except ImportError:
            logger.warning("PyMuPDF not available, skipping PDF image crop")
            return None

        try:
            doc = fitz.open(stream=pdf_bytes, filetype="pdf")
            if page_number < 1 or page_number > len(doc):
                logger.warning(
                    "Page number out of range",
                    extra={"page": page_number, "total_pages": len(doc)},
                )
                return None

            page = doc[page_number - 1]  # 0-based index
            rect = page.rect

            # Convert normalized bbox to page coordinates
            x0 = bbox.get("Left", 0) * rect.width
            y0 = bbox.get("Top", 0) * rect.height
            x1 = (bbox.get("Left", 0) + bbox.get("Width", 0)) * rect.width
            y1 = (bbox.get("Top", 0) + bbox.get("Height", 0)) * rect.height

            clip_rect = fitz.Rect(x0, y0, x1, y1)
            pix = page.get_pixmap(clip=clip_rect, dpi=150)

            return pix.tobytes("png")

        except Exception as exc:
            logger.warning("PDF image crop failed", extra={"error": str(exc)})
            return None

    # ── Internal: Nova Lite Multimodal Caption (SAFE) ──────────────────────

    def _caption_image_safe(self, image_bytes: bytes) -> str:
        """Send image to Nova Lite multimodal API for caption with validation.

        Args:
            image_bytes: PNG image bytes.

        Returns:
            Generated caption string, or placeholder if API fails.
        """
        # Validate image is not empty
        if not image_bytes or len(image_bytes) < 100:
            logger.warning("Image too small or empty, skipping caption")
            return "[Image too small or empty — caption unavailable]"

        try:
            return self._caption_image(image_bytes)
        except ClientError as exc:
            error_code = exc.response["Error"]["Code"]
            error_msg = exc.response["Error"].get("Message", str(exc))
            logger.warning(
                "Image captioning failed via API",
                extra={
                    "error_code": error_code,
                    "error_message": error_msg,
                },
            )
            return f"[Image caption unavailable — {error_code}]"
        except Exception as exc:
            logger.warning(
                "Image captioning failed unexpectedly",
                extra={"error": str(exc)},
            )
            return "[Image caption unavailable — unexpected error]"

    @retry(max_attempts=2, backoff_seconds=2, exceptions=(ClientError,))
    def _caption_image(self, image_bytes: bytes) -> str:
        """Send image to Nova Lite multimodal API for caption.

        IMPORTANT: Pass raw bytes directly. Boto3 handles base64 encoding internally.
        Do NOT base64-encode the image yourself.

        Args:
            image_bytes: PNG image bytes (raw, NOT base64).

        Returns:
            Generated caption string.
        """
        response = self.bedrock.converse(
            modelId=self.chat_model,
            messages=[
                {
                    "role": "user",
                    "content": [
                        {
                            "image": {
                                "format": "png",
                                "source": {"bytes": image_bytes},  # RAW BYTES
                            }
                        },
                        {"text": self.prompt},
                    ],
                }
            ],
            inferenceConfig={
                "maxTokens": 300,
                "temperature": 0.3,
            },
        )

        return response["output"]["message"]["content"][0]["text"].strip()


# ── Convenience Function ───────────────────────────────────────────────────

def extract_and_caption_images(doc_id: str, source_s3_key: str) -> str:
    """One-shot function to extract and caption images for a document.

    Args:
        doc_id: Document identifier.
        source_s3_key: S3 key of the original source file.

    Returns:
        S3 key of image_captions.json.
    """
    extractor = ImageExtractor()
    return extractor.extract_and_caption(doc_id, source_s3_key)

#table_extractor.py
"""Extract tables from Textract output, convert to markdown, and enrich with Nova Lite.

Pipeline:
  1. Read raw Textract JSON + reading_order.json
  2. Extract TABLE blocks with cell geometry
  3. Convert each table to pipe-delimited markdown
  4. Extract surrounding text context (before/after blocks)
  5. Send table + context to Nova Lite for semantic caption
  6. Store:
     - tables.json: Markdown representations + metadata
     - table_enrichments.json: Nova-generated captions

Handles fallback output from detect_document_text (no TABLE blocks) by
writing empty table files without crashing.

Nova Lite prompt (from config.paths.PROMPTS.TABLE_ENRICHMENT):
  "Describe what this table represents, its column meanings, and key values."
"""

import json
from typing import Any, Dict, List, Optional, Tuple

from config.settings import load_settings
from utils.aws_clients import get_bedrock_runtime_client
from utils.s3_utils import download_json, upload_json
from utils.text_utils import (
    textract_table_to_markdown,
    extract_context_around_table,
    estimate_token_count,
)
from utils.logging_config import get_logger, log_operation
from utils.decorators import retry, timing

logger = get_logger("table_extractor")


class TableExtractor:
    """Extract and enrich tables from Textract output."""

    def __init__(self):
        self.settings = load_settings()
        self.bucket = self.settings.S3_BUCKET
        self.bedrock = get_bedrock_runtime_client()
        self.chat_model = self.settings.BEDROCK_CHAT_MODEL_ID
        self.prompt = self.settings.prompts.TABLE_ENRICHMENT

    # ── Public API ─────────────────────────────────────────────────────────

    @timing()
    def extract_and_enrich(self, doc_id: str) -> Tuple[str, str]:
        """Extract all tables from a document and enrich with Nova Lite captions.

        Args:
            doc_id: Document identifier.

        Returns:
            Tuple of (tables_s3_key, enrichments_s3_key).
        """
        # Read inputs
        textract_key = self.settings.s3_paths.textract_doc_path(doc_id) + "layout.json"
        reading_key = self.settings.s3_paths.json_doc_path(doc_id) + "reading_order.json"

        textract_data = download_json(self.bucket, textract_key)
        reading_data = download_json(self.bucket, reading_key)

        if not textract_data or "Blocks" not in textract_data:
            raise ValueError(f"Invalid Textract data for {doc_id}")

        blocks = textract_data["Blocks"]
        blocks_by_id = {b["Id"]: b for b in blocks}
        reading_blocks = reading_data.get("blocks", [])

        # Check if fallback was used (detect_document_text has no TABLE blocks)
        fallback_used = reading_data.get("fallback_used", False)

        # Find all TABLE blocks
        table_blocks = [b for b in blocks if b.get("BlockType") == "TABLE"]

        if not table_blocks:
            logger.info(
                "No tables found in document",
                extra={"doc_id": doc_id, "fallback_used": fallback_used},
            )
            # Write empty files for consistency (downstream expects these keys)
            tables_output = {"doc_id": doc_id, "tables": [], "fallback_used": fallback_used}
            enrichments_output = {"doc_id": doc_id, "enrichments": [], "fallback_used": fallback_used}

            tables_key = self.settings.s3_paths.json_doc_path(doc_id) + "tables.json"
            enrichments_key = (
                self.settings.s3_paths.json_doc_path(doc_id) + "table_enrichments.json"
            )
            upload_json(self.bucket, tables_key, tables_output)
            upload_json(self.bucket, enrichments_key, enrichments_output)
            return tables_key, enrichments_key

        tables_output = {"doc_id": doc_id, "tables": [], "fallback_used": fallback_used}
        enrichments_output = {"doc_id": doc_id, "enrichments": [], "fallback_used": fallback_used}

        for i, table_block in enumerate(table_blocks, start=1):
            table_id = f"{doc_id}_table_{i:03d}"

            # Convert to markdown
            markdown = textract_table_to_markdown(table_block, blocks_by_id)

            # Extract context from reading order
            context = self._extract_table_context(
                table_block["Id"], reading_blocks, blocks_by_id
            )

            # Get page number from table block
            page = table_block.get("Page", 1)

            # Build table record
            table_record = {
                "table_id": table_id,
                "textract_block_id": table_block["Id"],
                "page_number": page,
                "markdown": markdown,
                "row_count": self._count_rows(table_block),
                "column_count": self._count_columns(table_block),
                "context": context,
            }
            tables_output["tables"].append(table_record)

            # Enrich with Nova Lite (with graceful fallback)
            caption = self._enrich_table_safe(table_record)

            enrichment_record = {
                "table_id": table_id,
                "caption": caption,
                "page_number": page,
            }
            enrichments_output["enrichments"].append(enrichment_record)

            logger.info(
                "Table enriched",
                extra={
                    "doc_id": doc_id,
                    "table_id": table_id,
                    "page": page,
                    "rows": table_record["row_count"],
                    "cols": table_record["column_count"],
                },
            )

        # Upload outputs
        tables_key = self.settings.s3_paths.json_doc_path(doc_id) + "tables.json"
        enrichments_key = (
            self.settings.s3_paths.json_doc_path(doc_id) + "table_enrichments.json"
        )

        upload_json(self.bucket, tables_key, tables_output)
        upload_json(self.bucket, enrichments_key, enrichments_output)

        logger.info(
            "Table extraction complete",
            extra={
                "doc_id": doc_id,
                "table_count": len(tables_output["tables"]),
            },
        )

        return tables_key, enrichments_key

    # ── Internal: Context Extraction ───────────────────────────────────────

    def _extract_table_context(
        self,
        table_block_id: str,
        reading_blocks: List[Dict[str, Any]],
        blocks_by_id: Dict[str, Dict[str, Any]],
    ) -> str:
        """Extract text blocks immediately before and after a table.

        Uses reading_order blocks to find position, then collects
        LAYOUT_TEXT and LAYOUT_SECTION_HEADER blocks up to token limit.
        """
        # Find table position in reading order
        table_idx = -1
        for i, block in enumerate(reading_blocks):
            if block.get("block_id") == table_block_id:
                table_idx = i
                break

        if table_idx == -1:
            return ""

        max_tokens = self.settings.chunking.TABLE_CONTEXT_TOKENS
        half_tokens = max_tokens // 2

        before_parts = []
        after_parts = []
        before_tokens = 0
        after_tokens = 0

        # Collect before table
        for i in range(table_idx - 1, -1, -1):
            block = reading_blocks[i]
            if block["block_type"] in ("LAYOUT_TEXT", "LAYOUT_SECTION_HEADER"):
                text = block.get("text", "")
                tokens = estimate_token_count(text)
                if before_tokens + tokens > half_tokens:
                    break
                before_parts.insert(0, text)
                before_tokens += tokens

        # Collect after table
        for i in range(table_idx + 1, len(reading_blocks)):
            block = reading_blocks[i]
            if block["block_type"] in ("LAYOUT_TEXT", "LAYOUT_SECTION_HEADER"):
                text = block.get("text", "")
                tokens = estimate_token_count(text)
                if after_tokens + tokens > half_tokens:
                    break
                after_parts.append(text)
                after_tokens += tokens

        return " ".join(before_parts + after_parts)

    # ── Internal: Nova Lite Enrichment (SAFE) ────────────────────────────

    def _enrich_table_safe(self, table_record: Dict[str, Any]) -> str:
        """Send table to Nova Lite with validation and fallback."""
        markdown = table_record.get("markdown", "")
        if not markdown or len(markdown) < 10:
            return "[Table too small or empty — enrichment unavailable]"

        try:
            return self._enrich_table(table_record)
        except Exception as exc:
            logger.warning(
                "Table enrichment failed, using placeholder",
                extra={"error": str(exc)},
            )
            return "[Table enrichment unavailable — could not process]"

    @retry(max_attempts=2, backoff_seconds=2, exceptions=(Exception,))
    def _enrich_table(self, table_record: Dict[str, Any]) -> str:
        """Send table markdown + context to Nova Lite for semantic caption.

        Args:
            table_record: Dict with 'markdown', 'context', 'table_id'.

        Returns:
            Nova Lite generated caption string.
        """
        prompt = self._build_enrichment_prompt(table_record)

        response = self.bedrock.converse(
            modelId=self.chat_model,
            messages=[
                {
                    "role": "user",
                    "content": [{"text": prompt}],
                }
            ],
            inferenceConfig={
                "maxTokens": 500,
                "temperature": 0.3,
            },
        )

        output = response["output"]["message"]["content"][0]["text"]
        return output.strip()

    def _build_enrichment_prompt(self, table_record: Dict[str, Any]) -> str:
        """Build the prompt for table enrichment."""
        markdown = table_record.get("markdown", "")
        context = table_record.get("context", "")

        prompt_parts = [self.settings.prompts.TABLE_ENRICHMENT]

        if context:
            prompt_parts.append(f"\n\nContext from the document:\n{context}")

        prompt_parts.append(f"\n\nTable (in markdown format):\n\n{markdown}")

        return "\n".join(prompt_parts)

#layout_parser.py
"""Parse Textract raw JSON into structured reading order and document metadata.

Enforces sequential document flow:
  1. Group blocks by page number
  2. Sort by vertical position (top-to-bottom)
  3. Within same Y-position, sort by horizontal (left-to-right)
  4. Assign reading_order integers
  5. Identify section boundaries at LAYOUT_TITLE / LAYOUT_SECTION_HEADER

Handles fallback output from detect_document_text (which produces LINE blocks
instead of LAYOUT blocks) by creating synthetic layout blocks.

Outputs:
  - reading_order.json: Ordered blocks with geometry, page, sequence
  - document_metadata.json: Doc-level info (title, pages, source)
"""

import json
from typing import Any, Dict, List, Optional, Tuple

from config.settings import load_settings
from utils.s3_utils import download_json, upload_json
from utils.logging_config import get_logger, log_operation
from utils.decorators import timing

logger = get_logger("layout_parser")


class LayoutParser:
    """Parse Textract layout into sequential reading order."""

    def __init__(self):
        self.settings = load_settings()
        self.bucket = self.settings.S3_BUCKET

    # ── Public API ─────────────────────────────────────────────────────────

    @timing()
    def parse_document(self, doc_id: str) -> Tuple[str, str]:
        """Parse Textract output into reading_order.json and document_metadata.json.

        Args:
            doc_id: Document identifier (e.g., 'doc_001').

        Returns:
            Tuple of (reading_order_s3_key, metadata_s3_key).
        """
        # Read raw Textract JSON from S3
        textract_key = self.settings.s3_paths.textract_doc_path(doc_id) + "layout.json"
        raw_data = download_json(self.bucket, textract_key)

        if not raw_data or "Blocks" not in raw_data:
            raise ValueError(f"Invalid Textract data for {doc_id}: missing 'Blocks'")

        blocks = raw_data["Blocks"]
        doc_meta = raw_data.get("DocumentMetadata", {})
        fallback_used = raw_data.get("_fallback_used", False)

        if fallback_used:
            logger.warning(
                "Fallback output detected (detect_document_text), "
                "using LINE blocks for reading order",
                extra={"doc_id": doc_id},
            )

        # Build block lookup for relationship resolution
        blocks_by_id = {b["Id"]: b for b in blocks}

        # Extract layout blocks (text, headers, tables, figures)
        layout_blocks = self._extract_layout_blocks(blocks, blocks_by_id, fallback_used)

        # Sort by page → top → left
        sorted_blocks = self._sort_by_reading_order(layout_blocks)

        # Assign sequential reading_order integers
        for i, block in enumerate(sorted_blocks, start=1):
            block["reading_order"] = i

        # Identify section boundaries
        sections = self._identify_sections(sorted_blocks)

        # Build outputs
        reading_order = {
            "doc_id": doc_id,
            "total_pages": doc_meta.get("Pages", 0),
            "total_blocks": len(sorted_blocks),
            "fallback_used": fallback_used,
            "blocks": sorted_blocks,
            "sections": sections,
        }

        metadata = {
            "doc_id": doc_id,
            "total_pages": doc_meta.get("Pages", 0),
            "total_blocks": len(blocks),
            "layout_blocks": len(sorted_blocks),
            "sections_count": len(sections),
            "fallback_used": fallback_used,
            "source_s3_key": textract_key,
        }

        # Upload to S3
        reading_key = self.settings.s3_paths.json_doc_path(doc_id) + "reading_order.json"
        metadata_key = self.settings.s3_paths.json_doc_path(doc_id) + "document_metadata.json"

        upload_json(self.bucket, reading_key, reading_order)
        upload_json(self.bucket, metadata_key, metadata)

        logger.info(
            "Layout parsed",
            extra={
                "doc_id": doc_id,
                "blocks": len(sorted_blocks),
                "sections": len(sections),
                "pages": doc_meta.get("Pages", 0),
                "fallback_used": fallback_used,
            },
        )

        return reading_key, metadata_key

    # ── Internal: Block Extraction ─────────────────────────────────────────

    def _extract_layout_blocks(
        self,
        blocks: List[Dict[str, Any]],
        blocks_by_id: Dict[str, Dict[str, Any]],
        fallback_used: bool = False,
    ) -> List[Dict[str, Any]]:
        """Extract relevant layout blocks with resolved text and geometry.

        For analyze_document output: uses LAYOUT_* block types.
        For detect_document_text fallback: uses LINE blocks as synthetic layout.
        """
        layout_types = {
            "LAYOUT_TEXT",
            "LAYOUT_TITLE",
            "LAYOUT_SECTION_HEADER",
            "LAYOUT_TABLE",
            "LAYOUT_FIGURE",
            "LAYOUT_LIST",
            "LAYOUT_PAGE_NUMBER",
            "LAYOUT_FOOTER",
            "LAYOUT_HEADER",
        }

        result = []
        for block in blocks:
            block_type = block.get("BlockType", "")

            if block_type in layout_types:
                # Standard analyze_document output
                result.append(self._build_layout_record(block, blocks_by_id))
            elif fallback_used and block_type == "LINE":
                # Fallback: create synthetic layout block from LINE
                result.append(self._build_synthetic_layout_record(block))

        # If still no blocks (very rare), try WORD blocks
        if not result and fallback_used:
            logger.warning(
                "No LINE blocks found, falling back to WORD blocks",
                extra={"doc_id": "unknown"},
            )
            for block in blocks:
                if block.get("BlockType") == "WORD":
                    result.append(self._build_synthetic_layout_record(block))

        return result

    def _build_layout_record(
        self,
        block: Dict[str, Any],
        blocks_by_id: Dict[str, Dict[str, Any]],
    ) -> Dict[str, Any]:
        """Build a layout record from a standard Textract LAYOUT block."""
        geometry = block.get("Geometry", {})
        bbox = geometry.get("BoundingBox", {})
        text = self._resolve_text(block, blocks_by_id)
        page = block.get("Page", 1)

        return {
            "block_id": block["Id"],
            "block_type": block["BlockType"],
            "page_number": page,
            "text": text,
            "geometry": {
                "top": bbox.get("Top", 0.0),
                "left": bbox.get("Left", 0.0),
                "width": bbox.get("Width", 0.0),
                "height": bbox.get("Height", 0.0),
            },
            "confidence": block.get("Confidence", 0.0),
        }

    def _build_synthetic_layout_record(self, block: Dict[str, Any]) -> Dict[str, Any]:
        """Build a synthetic layout record from a LINE or WORD block.

        Used when detect_document_text fallback is active (no LAYOUT blocks).
        """
        geometry = block.get("Geometry", {})
        bbox = geometry.get("BoundingBox", {})
        text = block.get("Text", "")
        page = block.get("Page", 1)

        # Heuristic: classify as title if short and all caps
        block_type = "LAYOUT_TEXT"
        if text and len(text) < 100 and text.isupper():
            block_type = "LAYOUT_SECTION_HEADER"
        elif text and len(text) < 50 and text.endswith(":"):
            block_type = "LAYOUT_SECTION_HEADER"

        return {
            "block_id": block["Id"],
            "block_type": block_type,
            "page_number": page,
            "text": text,
            "geometry": {
                "top": bbox.get("Top", 0.0),
                "left": bbox.get("Left", 0.0),
                "width": bbox.get("Width", 0.0),
                "height": bbox.get("Height", 0.0),
            },
            "confidence": block.get("Confidence", 0.0),
            "synthetic": True,
        }

    def _resolve_text(
        self,
        block: Dict[str, Any],
        blocks_by_id: Dict[str, Dict[str, Any]],
    ) -> str:
        """Resolve full text from a block by concatenating child WORD blocks."""
        text_parts = []
        for rel in block.get("Relationships", []):
            if rel.get("Type") == "CHILD":
                for child_id in rel.get("Ids", []):
                    child = blocks_by_id.get(child_id)
                    if child and child.get("BlockType") == "WORD":
                        text_parts.append(child.get("Text", ""))
        return " ".join(text_parts).strip()

    # ── Internal: Sorting ──────────────────────────────────────────────────

    def _sort_by_reading_order(
        self,
        blocks: List[Dict[str, Any]],
    ) -> List[Dict[str, Any]]:
        """Sort blocks by page number, then vertical position, then horizontal."""
        return sorted(
            blocks,
            key=lambda b: (
                b["page_number"],
                b["geometry"]["top"],
                b["geometry"]["left"],
            ),
        )

    # ── Internal: Section Identification ───────────────────────────────────

    def _identify_sections(
        self,
        sorted_blocks: List[Dict[str, Any]],
    ) -> List[Dict[str, Any]]:
        """Identify document sections bounded by LAYOUT_TITLE or LAYOUT_SECTION_HEADER.

        Returns list of sections with start_block_id, end_block_id, heading, page_range.
        """
        sections = []
        current_section = None

        for block in sorted_blocks:
            block_type = block["block_type"]
            block_id = block["block_id"]
            page = block["page_number"]
            text = block["text"]

            # Start new section at title or section header
            if block_type in ("LAYOUT_TITLE", "LAYOUT_SECTION_HEADER"):
                # Close previous section
                if current_section:
                    current_section["end_block_id"] = sorted_blocks[
                        sorted_blocks.index(block) - 1
                    ]["block_id"]
                    sections.append(current_section)

                current_section = {
                    "section_id": f"section_{len(sections) + 1:03d}",
                    "heading": text,
                    "heading_block_id": block_id,
                    "start_block_id": block_id,
                    "end_block_id": None,
                    "page_start": page,
                    "page_end": page,
                }
            elif current_section:
                # Update page range
                current_section["page_end"] = max(current_section["page_end"], page)

        # Close final section
        if current_section and sorted_blocks:
            current_section["end_block_id"] = sorted_blocks[-1]["block_id"]
            sections.append(current_section)

        # If no sections found (fallback case), create one default section
        if not sections and sorted_blocks:
            sections.append({
                "section_id": "section_001",
                "heading": "Document Content",
                "heading_block_id": sorted_blocks[0]["block_id"],
                "start_block_id": sorted_blocks[0]["block_id"],
                "end_block_id": sorted_blocks[-1]["block_id"],
                "page_start": sorted_blocks[0]["page_number"],
                "page_end": sorted_blocks[-1]["page_number"],
            })

        return sections

#textract_processor.py
"""AWS Textract document analysis with polling and S3 persistence.

Handles:
  - analyze_document (synchronous, <10 pages) or start_document_analysis (async, >10 pages)
  - Pagination of Textract block responses
  - Polling with configurable interval and max attempts
  - Raw JSON upload to S3 for downstream parsing
  - Fallback to detect_document_text for unsupported PDF formats
"""

import json
import time
from typing import Any, Dict, List, Optional

import boto3
from botocore.exceptions import ClientError

from config.settings import load_settings
from utils.aws_clients import get_textract_client, get_s3_client
from utils.s3_utils import upload_json, check_exists
from utils.logging_config import get_logger, log_operation
from utils.decorators import retry, timing

logger = get_logger("textract_processor")


class TextractProcessor:
    """Orchestrates Textract document analysis and S3 persistence."""

    def __init__(self):
        self.settings = load_settings()
        self.textract = get_textract_client()
        self.s3 = get_s3_client()
        self.bucket = self.settings.S3_BUCKET
        self.features = list(self.settings.textract.FEATURES)
        self.poll_interval = self.settings.textract.POLLING_INTERVAL_SECONDS
        self.max_polls = self.settings.textract.MAX_POLLING_ATTEMPTS

    # ── Public API ─────────────────────────────────────────────────────────

    @timing()
    def process_document(self, doc_id: str, s3_input_key: str) -> str:
        """Analyze a document via Textract and store raw output in S3.

        Args:
            doc_id: Document identifier (e.g., 'doc_001').
            s3_input_key: S3 key of the source file (e.g., 'ecmc_org/source_files/pdfs/doc_001.pdf').

        Returns:
            S3 key where raw Textract JSON was saved.

        Raises:
            RuntimeError: If Textract job fails or polling exceeds max attempts.
            ClientError: On AWS API errors.
        """
        output_key = self.settings.s3_paths.textract_doc_path(doc_id) + "layout.json"

        # Skip if already processed
        if check_exists(self.bucket, output_key):
            logger.info(
                "Textract output already exists, skipping",
                extra={"doc_id": doc_id, "s3_key": output_key},
            )
            return output_key

        logger.info(
            "Starting Textract analysis",
            extra={"doc_id": doc_id, "s3_input_key": s3_input_key, "features": self.features},
        )

        # Determine sync vs async based on file size (heuristic: >10 pages or >5MB → async)
        file_size = self._get_s3_file_size(s3_input_key)
        if file_size and file_size > 5 * 1024 * 1024:
            raw_response = self._analyze_async(s3_input_key)
        else:
            raw_response = self._analyze_sync(s3_input_key)

        # Upload raw response to S3
        upload_json(self.bucket, output_key, raw_response)

        logger.info(
            "Textract analysis complete",
            extra={
                "doc_id": doc_id,
                "s3_output_key": output_key,
                "block_count": len(raw_response.get("Blocks", [])),
                "fallback_used": raw_response.get("_fallback_used", False),
            },
        )
        return output_key

    # ── Internal: Synchronous Analysis ───────────────────────────────────

    @retry(max_attempts=3, backoff_seconds=2, exceptions=(ClientError,))
    def _analyze_sync(self, s3_input_key: str) -> Dict[str, Any]:
        """Call analyze_document (synchronous, best for <10 pages).

        Falls back to detect_document_text if the document format is unsupported
        (e.g., OneNote-exported PDFs, image-based PDFs without proper structure).
        """
        try:
            response = self.textract.analyze_document(
                Document={
                    "S3Object": {"Bucket": self.bucket, "Name": s3_input_key}
                },
                FeatureTypes=self.features,
            )
            return self._flatten_response(response)
        except ClientError as exc:
            error_code = exc.response["Error"]["Code"]
            if error_code == "UnsupportedDocumentException":
                logger.warning(
                    "AnalyzeDocument unsupported format, falling back to DetectDocumentText",
                    extra={"s3_key": s3_input_key, "error": str(exc)},
                )
                # Fallback: basic OCR without tables/layout/forms
                response = self.textract.detect_document_text(
                    Document={
                        "S3Object": {"Bucket": self.bucket, "Name": s3_input_key}
                    }
                )
                result = self._flatten_response(response)
                result["_fallback_used"] = True
                result["_fallback_type"] = "detect_document_text"
                result["_original_error"] = str(exc)
                return result
            raise

    # ── Internal: Asynchronous Analysis ────────────────────────────────────

    @retry(max_attempts=3, backoff_seconds=2, exceptions=(ClientError,))
    def _analyze_async(self, s3_input_key: str) -> Dict[str, Any]:
        """Call start_document_analysis + poll (best for >10 pages)."""
        start_resp = self.textract.start_document_analysis(
            DocumentLocation={
                "S3Object": {"Bucket": self.bucket, "Name": s3_input_key}
            },
            FeatureTypes=self.features,
        )
        job_id = start_resp["JobId"]
        logger.info(
            "Async Textract job started",
            extra={"job_id": job_id, "s3_input_key": s3_input_key},
        )

        # Poll until complete
        for attempt in range(1, self.max_polls + 1):
            status_resp = self.textract.get_document_analysis(JobId=job_id)
            status = status_resp["JobStatus"]

            if status == "SUCCEEDED":
                logger.info(
                    "Async Textract job succeeded",
                    extra={"job_id": job_id, "attempts": attempt},
                )
                return self._paginate_async_results(job_id)

            if status == "FAILED":
                error_msg = status_resp.get("StatusMessage", "Unknown error")
                logger.error(
                    "Async Textract job failed",
                    extra={"job_id": job_id, "error": error_msg},
                )
                raise RuntimeError(f"Textract job {job_id} failed: {error_msg}")

            logger.info(
                "Textract job in progress, polling...",
                extra={"job_id": job_id, "attempt": attempt, "status": status},
            )
            time.sleep(self.poll_interval)

        raise RuntimeError(
            f"Textract job {job_id} did not complete within "
            f"{self.max_polls * self.poll_interval} seconds"
        )

    # ── Internal: Pagination ───────────────────────────────────────────────

    def _paginate_async_results(self, job_id: str) -> Dict[str, Any]:
        """Paginate through all blocks of an async Textract job."""
        all_blocks: List[Dict[str, Any]] = []
        next_token: Optional[str] = None
        page_count = 0

        while True:
            kwargs = {"JobId": job_id}
            if next_token:
                kwargs["NextToken"] = next_token

            page = self.textract.get_document_analysis(**kwargs)
            page_count += 1
            blocks = page.get("Blocks", [])
            all_blocks.extend(blocks)

            next_token = page.get("NextToken")
            if not next_token:
                break

        logger.info(
            "Paginated Textract results",
            extra={"job_id": job_id, "pages": page_count, "total_blocks": len(all_blocks)},
        )

        # Return a structure compatible with analyze_document response
        return {
            "DocumentMetadata": page.get("DocumentMetadata", {}),
            "Blocks": all_blocks,
            "JobId": job_id,
        }

    # ── Internal: Helpers ──────────────────────────────────────────────────

    def _get_s3_file_size(self, key: str) -> Optional[int]:
        """Return S3 object size in bytes, or None if not found."""
        try:
            resp = self.s3.head_object(Bucket=self.bucket, Key=key)
            return resp["ContentLength"]
        except ClientError:
            return None

    def _flatten_response(self, response: Dict[str, Any]) -> Dict[str, Any]:
        """Ensure consistent structure between sync and async responses."""
        return {
            "DocumentMetadata": response.get("DocumentMetadata", {}),
            "Blocks": response.get("Blocks", []),
        }


# ── Convenience Function ───────────────────────────────────────────────────

def process_document(doc_id: str, s3_input_key: str) -> str:
    """One-shot function to process a single document.

    Args:
        doc_id: Document identifier.
        s3_input_key: S3 key of source PDF/HTML.

    Returns:
        S3 key of stored Textract output.
    """
    processor = TextractProcessor()
    return processor.process_document(doc_id, s3_input_key)

#!/usr/bin/env python3
"""CLI entry point: Ingest all PDFs from S3 source_files through Textract pipeline.

Usage:
    # Process all PDFs in source_files/
    python scripts/01_ingest.py

    # Process specific PDF
    python scripts/01_ingest.py --source-key ecmc_org/source_files/borrower_guide.pdf

    # Dry run (discover files but do not process)
    python scripts/01_ingest.py --dry-run

Flow:
    1. List all PDFs in s3://{bucket}/ecmc_org/source_files/
    2. Assign sequential doc_ids (doc_001, doc_002, ...)
    3. For each PDF:
       a. Textract analysis → s3://.../textract_output/{doc_id}/layout.json
       b. Layout parsing → s3://.../ECMC_RAG/02_json/{doc_id}/reading_order.json
       c. Table extraction → s3://.../ECMC_RAG/02_json/{doc_id}/tables.json
       d. Image extraction → s3://.../ECMC_RAG/02_json/{doc_id}/image_captions.json
    4. Write manifest.json with doc_id → source_key mapping

Prerequisites:
    - .env file with AWS_REGION, S3_BUCKET, BEDROCK_CHAT_MODEL_ID, BEDROCK_EMBED_MODEL_ID
    - AWS credentials configured (IAM role or ~/.aws/credentials)
    - pip install -r requirements.txt
"""

import argparse
import json
import sys
from typing import Dict, List, Optional, Set

from config.settings import load_settings
from utils.s3_utils import list_objects, upload_json, check_exists
from utils.text_utils import generate_doc_id

def discover_pdfs(bucket: str, prefix: str) -> List[str]:
    """List all PDF files under the source_files prefix.

    Args:
        bucket: S3 bucket name.
        prefix: S3 key prefix (e.g., 'ecmc_org/source_files/').

    Returns:
        List of S3 keys ending with .pdf.
    """
    all_keys = list_objects(bucket, prefix)
    pdf_keys = [k for k in all_keys if k.lower().endswith(".pdf")]
    logger.info(
        "Discovered PDFs",
        extra={"bucket": bucket, "prefix": prefix, "count": len(pdf_keys)},
    )
    return sorted(pdf_keys)


def load_existing_manifest(bucket: str, manifest_key: str) -> Dict[str, str]:
    """Load existing manifest to avoid re-processing and maintain doc_id continuity.

    Args:
        bucket: S3 bucket name.
        manifest_key: S3 key of manifest.json.

    Returns:
        Dict mapping doc_id → source_s3_key.
    """
    from utils.s3_utils import download_json
    try:
        data = download_json(bucket, manifest_key)
        if data and "documents" in data:
            return {d["doc_id"]: d["source_key"] for d in data["documents"]}
    except Exception:
        pass
    return {}


def process_single_document(
    doc_id: str,
    source_key: str,
    bucket: str,
    settings,
    skip_existing: bool = True,
) -> Dict[str, str]:
    """Run full ingestion pipeline for one PDF.

    Args:
        doc_id: Assigned document identifier.
        source_key: S3 key of source PDF.
        bucket: S3 bucket name.
        settings: Settings instance.
        skip_existing: If True, skip steps where output already exists in S3.

    Returns:
        Dict with output S3 keys for each step.
    """
    outputs = {
        "doc_id": doc_id,
        "source_key": source_key,
        "textract_output": None,
        "reading_order": None,
        "document_metadata": None,
        "tables": None,
        "table_enrichments": None,
        "image_captions": None,
    }

    # ── Step 1: Textract ─────────────────────────────────────────────────
    textract_key = settings.s3_paths.textract_doc_path(doc_id) + "layout.json"

    if skip_existing and check_exists(bucket, textract_key):
        logger.info(
            "Textract output exists, skipping",
            extra={"doc_id": doc_id, "s3_key": textract_key},
        )
        outputs["textract_output"] = textract_key
    else:
        logger.info(
            "Running Textract",
            extra={"doc_id": doc_id, "source_key": source_key},
        )
        outputs["textract_output"] = run_textract(doc_id, source_key)

    # ── Step 2: Layout Parsing ───────────────────────────────────────────
    reading_key = settings.s3_paths.json_doc_path(doc_id) + "reading_order.json"

    if skip_existing and check_exists(bucket, reading_key):
        logger.info(
            "Reading order exists, skipping",
            extra={"doc_id": doc_id, "s3_key": reading_key},
        )
        outputs["reading_order"] = reading_key
    else:
        logger.info(
            "Parsing layout",
            extra={"doc_id": doc_id},
        )
        reading_key, meta_key = parse_layout(doc_id)
        outputs["reading_order"] = reading_key
        outputs["document_metadata"] = meta_key

    # ── Step 3: Table Extraction ─────────────────────────────────────────
    tables_key = settings.s3_paths.json_doc_path(doc_id) + "tables.json"

    if skip_existing and check_exists(bucket, tables_key):
        logger.info(
            "Tables exist, skipping",
            extra={"doc_id": doc_id, "s3_key": tables_key},
        )
        outputs["tables"] = tables_key
    else:
        logger.info(
            "Extracting tables",
            extra={"doc_id": doc_id},
        )
        tables_key, enrich_key = extract_and_enrich_tables(doc_id)
        outputs["tables"] = tables_key
        outputs["table_enrichments"] = enrich_key

    # ── Step 4: Image Extraction ─────────────────────────────────────────
    captions_key = settings.s3_paths.json_doc_path(doc_id) + "image_captions.json"

    if skip_existing and check_exists(bucket, captions_key):
        logger.info(
            "Image captions exist, skipping",
            extra={"doc_id": doc_id, "s3_key": captions_key},
        )
        outputs["image_captions"] = captions_key
    else:
        logger.info(
            "Extracting images",
            extra={"doc_id": doc_id},
        )
        outputs["image_captions"] = extract_and_caption_images(doc_id, source_key)

    return outputs


def main():
    parser = argparse.ArgumentParser(
        description="Ingest PDFs from S3 through Textract pipeline"
    )
    parser.add_argument(
        "--source-key",
        type=str,
        default=None,
        help="Process a specific S3 key (e.g., ecmc_org/source_files/doc_001.pdf). "
             "If omitted, processes all PDFs in source_files/.",
    )
    parser.add_argument(
        "--dry-run",
        action="store_true",
        help="List files that would be processed without running Textract.",
    )
    parser.add_argument(
        "--force",
        action="store_true",
        help="Re-process even if outputs already exist in S3.",
    )
    parser.add_argument(
        "--max-docs",
        type=int,
        default=None,
        help="Limit number of documents to process (for testing).",
    )

    args = parser.parse_args()

    # Load configuration
    settings = load_settings()
    bucket = settings.S3_BUCKET
    source_prefix = settings.s3_paths.SOURCE_FILES

    logger.info(
        "Ingestion started",
        extra={
            "bucket": bucket,
            "source_prefix": source_prefix,
            "dry_run": args.dry_run,
            "force": args.force,
        },
    )

    # Discover PDFs
    if args.source_key:
        pdf_keys = [args.source_key]
    else:
        pdf_keys = discover_pdfs(bucket, source_prefix)

    if not pdf_keys:
        logger.error("No PDFs found", extra={"prefix": source_prefix})
        sys.exit(1)

    # Load manifest for doc_id continuity
    manifest_key = settings.s3_paths.PROCESSED_BASE + "manifest.json"
    existing = load_existing_manifest(bucket, manifest_key)
    existing_ids = set(existing.keys())

    if args.dry_run:
        print(f"\nDry run: Would process {len(pdf_keys)} PDFs")
        for key in pdf_keys[:args.max_docs or len(pdf_keys)]:
            doc_id = generate_doc_id(key.split("/")[-1], existing_ids)
            print(f"  {doc_id}: {key}")
        sys.exit(0)

    # Process documents
    results = []
    skip_existing = not args.force

    for i, source_key in enumerate(pdf_keys):
        if args.max_docs and i >= args.max_docs:
            logger.info(
                "Max documents reached",
                extra={"max_docs": args.max_docs, "processed": i},
            )
            break

        # Generate or reuse doc_id
        filename = source_key.split("/")[-1]
        doc_id = generate_doc_id(filename, existing_ids)
        existing_ids.add(doc_id)

        logger.info(
            f"Processing [{i+1}/{len(pdf_keys)}]",
            extra={"doc_id": doc_id, "source_key": source_key},
        )

        try:
            outputs = process_single_document(
                doc_id=doc_id,
                source_key=source_key,
                bucket=bucket,
                settings=settings,
                skip_existing=skip_existing,
            )
            results.append({
                "doc_id": doc_id,
                "source_key": source_key,
                "status": "success",
                "outputs": outputs,
            })
        except Exception as exc:
            logger.error(
                "Document processing failed",
                extra={
                    "doc_id": doc_id,
                    "source_key": source_key,
                    "error": str(exc),
                },
            )
            results.append({
                "doc_id": doc_id,
                "source_key": source_key,
                "status": "failed",
                "error": str(exc),
            })

    # Update manifest
    manifest = {
        "bucket": bucket,
        "source_prefix": source_prefix,
        "total_documents": len(results),
        "successful": sum(1 for r in results if r["status"] == "success"),
        "failed": sum(1 for r in results if r["status"] == "failed"),
        "documents": results,
    }
    upload_json(bucket, manifest_key, manifest)

    logger.info(
        "Ingestion complete",
        extra={
            "total": manifest["total_documents"],
            "successful": manifest["successful"],
            "failed": manifest["failed"],
            "manifest_key": manifest_key,
        },
    )

    # Print summary
    print(f"\n{'='*60}")
    print(f"Ingestion Complete")
    print(f"{'='*60}")
    print(f"Total:     {manifest['total_documents']}")
    print(f"Success:   {manifest['successful']}")
    print(f"Failed:    {manifest['failed']}")
    print(f"Manifest:  s3://{bucket}/{manifest_key}")
    print(f"{'='*60}")

    if manifest["failed"] > 0:
        sys.exit(1)


if __name__ == "__main__":
    main()

from typing import Optional

from config.settings import load_settings

# Module-level singleton cache
_bedrock_runtime: Optional[boto3.client] = None
_textract: Optional[boto3.client] = None
_s3: Optional[boto3.client] = None


def _get_region() -> str:
    """Fetch region from Settings singleton."""
    return load_settings().AWS_REGION


def get_bedrock_runtime_client() -> boto3.client:
    """Return singleton Bedrock Runtime client.

    Used for: Converse API (Nova Lite), InvokeModel (Titan Embed).
    Retry mode: adaptive (handles throttling automatically).

    global _bedrock_runtime
    if _bedrock_runtime is None:
        _bedrock_runtime = boto3.client(
            service_name="bedrock-runtime",
            region_name=_get_region(),
            config=Config(

                max_pool_connections=25,
            ),
        )
    return _bedrock_runtime

    """Return singleton Textract client.

    Used for: analyze_document, start_document_analysis, get_document_analysis.

    global _textract
    if _textract is None:
        _textract = boto3.client(
            service_name="textract",
            region_name=_get_region(),
            config=Config(
                retries={"max_attempts": 3, "mode": "standard"},
            ),
        )
    return _textract

"""All S3 read/write operations with centralized error handling.

Every S3 interaction in the pipeline goes through this module.
Provides format-aware upload/download (JSON, text, markdown, bytes)
and utility functions (list, exists, prefix normalization).
"""

import json
import logging
from io import BytesIO
from typing import Any, Dict, List, Optional

import boto3
from botocore.exceptions import ClientError

from .aws_clients import get_s3_client
from .logging_config import get_logger

logger = get_logger("s3_utils")


# ── Upload Operations ──────────────────────────────────────────────────────

def upload_json(bucket: str, key: str, data: Dict[str, Any]) -> None:
    """Serialize dict to JSON bytes and upload to S3.

    Args:
        bucket: S3 bucket name.
        key: S3 object key (should end with .json).
        data: Python dictionary to serialize.

    Raises:
        ClientError: On S3 upload failure.
    """
    body = json.dumps(data, indent=2, ensure_ascii=False).encode("utf-8")
    upload_bytes(bucket, key, body, content_type="application/json")
    logger.info("Uploaded JSON", extra={"bucket": bucket, "key": key, "bytes": len(body)})


def upload_text(bucket: str, key: str, text: str) -> None:
    """Upload plain text string to S3.

    Args:
        bucket: S3 bucket name.
        key: S3 object key (should end with .txt).
        text: UTF-8 string content.
    """
    body = text.encode("utf-8")
    upload_bytes(bucket, key, body, content_type="text/plain; charset=utf-8")
    logger.info("Uploaded text", extra={"bucket": bucket, "key": key, "bytes": len(body)})


def upload_markdown(bucket: str, key: str, markdown: str) -> None:
    """Upload markdown string to S3 with validation.

    Args:
        bucket: S3 bucket name.
        key: S3 object key (must end with .md).
        markdown: Markdown-formatted string.

    Raises:
        ValueError: If key does not end with .md.
    """
    if not key.endswith(".md"):
        raise ValueError(f"Markdown upload requires .md extension. Got: {key}")
    body = markdown.encode("utf-8")
    upload_bytes(bucket, key, body, content_type="text/markdown; charset=utf-8")
    logger.info("Uploaded markdown", extra={"bucket": bucket, "key": key, "bytes": len(body)})


def upload_bytes(
    bucket: str,
    key: str,
    data: bytes,
    content_type: str = "application/octet-stream",
) -> None:
    """Upload raw bytes to S3.

    Args:
        bucket: S3 bucket name.
        key: S3 object key.
        data: Raw bytes payload.
        content_type: MIME type for the object.

    Raises:
        ClientError: On S3 upload failure.
    """
    s3 = get_s3_client()
    try:
        s3.put_object(
            Bucket=bucket,
            Key=key,
            Body=data,
            ContentType=content_type,
        )
    except ClientError as exc:
        logger.error(
            "S3 upload failed",
            extra={"bucket": bucket, "key": key, "error": str(exc)},
        )
        raise


# ── Download Operations ──────────────────────────────────────────────────────

def download_json(bucket: str, key: str) -> Dict[str, Any]:
    """Download S3 object and deserialize JSON.

    Args:
        bucket: S3 bucket name.
        key: S3 object key.

    Returns:
        Python dictionary. Returns empty dict if key does not exist (404).

    Raises:
        ClientError: On non-404 S3 errors (permissions, network).
    """
    try:
        body = download_bytes(bucket, key)
        return json.loads(body.decode("utf-8"))
    except ClientError as exc:
        error_code = exc.response["Error"]["Code"]
        if error_code == "NoSuchKey":
            logger.warning("S3 key not found, returning empty dict", extra={"bucket": bucket, "key": key})
            return {}
        raise


def download_text(bucket: str, key: str) -> str:
    """Download S3 object as UTF-8 string.

    Args:
        bucket: S3 bucket name.
        key: S3 object key.

    Returns:
        UTF-8 decoded string.

    Raises:
        ClientError: On S3 download failure.
    """
    body = download_bytes(bucket, key)
    return body.decode("utf-8")


def download_bytes(bucket: str, key: str) -> bytes:
    """Download S3 object as raw bytes.

    Args:
        bucket: S3 bucket name.
        key: S3 object key.

    Returns:
        Raw bytes payload.

    Raises:
        ClientError: On S3 download failure (including 404).
    """
    s3 = get_s3_client()
    try:
        response = s3.get_object(Bucket=bucket, Key=key)
        body = response["Body"].read()
        logger.info(
            "Downloaded bytes",
            extra={"bucket": bucket, "key": key, "bytes": len(body)},
        )
        return body
    except ClientError as exc:
        logger.error(
            "S3 download failed",
            extra={"bucket": bucket, "key": key, "error": str(exc)},
        )
        raise


# ── Listing & Existence Operations ─────────────────────────────────────────

def list_objects(bucket: str, prefix: str) -> List[str]:
    """List all object keys under a prefix.

    Handles pagination automatically (S3 returns max 1000 per call).

    Args:
        bucket: S3 bucket name.
        prefix: S3 key prefix (e.g., 'ecmc_org/ECMC_RAG/01_markdown/').

    Returns:
        List of full object keys.
    """
    s3 = get_s3_client()
    keys: List[str] = []
    paginator = s3.get_paginator("list_objects_v2")

    for page in paginator.paginate(Bucket=bucket, Prefix=prefix):
        for obj in page.get("Contents", []):
            keys.append(obj["Key"])

    logger.info(
        "Listed S3 objects",
        extra={"bucket": bucket, "prefix": prefix, "count": len(keys)},
    )
    return keys


def check_exists(bucket: str, key: str) -> bool:
    """Check if an S3 object exists using HeadObject.

    Args:
        bucket: S3 bucket name.
        key: S3 object key.

    Returns:
        True if object exists, False otherwise.
    """
    s3 = get_s3_client()
    try:
        s3.head_object(Bucket=bucket, Key=key)
        return True
    except ClientError as exc:
        error_code = exc.response["Error"]["Code"]
        if error_code == "404":
            return False
        logger.error(
            "S3 head_object failed",
            extra={"bucket": bucket, "key": key, "error": str(exc)},
        )
        raise


def ensure_prefix(prefix: str) -> str:
    """Ensure prefix ends with '/' for safe concatenation.

    Args:
        prefix: S3 key prefix string.

    Returns:
        Prefix with trailing slash.
    """
    return prefix if prefix.endswith("/") else prefix + "/"

"""Text processing utilities for chunking, ID generation, and table conversion.

All token counting is approximate (whitespace-based) for fast, dependency-free
chunking decisions. Exact tokenization happens inside Bedrock.
"""

import re
import uuid
from typing import Dict, List, Optional, Set, Tuple

from config.paths import CHUNKING


# ── Token Estimation ───────────────────────────────────────────────────────

def estimate_token_count(text: str) -> int:
    """Approximate token count using whitespace-based heuristic.

    Rule of thumb: 1 token ≈ 0.75 English words.
    This is fast, dependency-free, and accurate within ±15% for chunking.

    Args:
        text: Input text string.

    Returns:
        Estimated integer token count.

    Example:
        >>> estimate_token_count("Income-driven repayment plans calculate.")
        6
    """
    if not text or not text.strip():
        return 0
    # Split on whitespace to get word count
    word_count = len(text.split())
    # Apply 0.75 tokens-per-word ratio and round up
    return max(1, int(word_count * 0.75))


# ── ID Generation ──────────────────────────────────────────────────────────

def generate_doc_id(filename: str, existing_ids: Optional[Set[str]] = None) -> str:
    """Generate sequential doc_id from filename.

    Format: doc_001, doc_002, ...
    If existing_ids provided, finds next available sequence number.

    Args:
        filename: Original file name (e.g., 'Federal_Student_Aid_Guide_2024.pdf').
        existing_ids: Set of already-assigned doc_ids. If None, generates doc_001.

    Returns:
        New unique doc_id string.

    Example:
        >>> generate_doc_id("guide.pdf")
        'doc_001'
        >>> generate_doc_id("guide2.pdf", {"doc_001"})
        'doc_002'
    """
    existing_ids = existing_ids or set()
    if not existing_ids:
        return "doc_001"

    # Extract numeric suffixes from existing doc_ids
    max_num = 0
    for doc_id in existing_ids:
        match = re.match(r"doc_(\d{3})", doc_id)
        if match:
            max_num = max(max_num, int(match.group(1)))

    next_num = max_num + 1
    return f"doc_{next_num:03d}"


def generate_parent_id(doc_id: str, sequence: int) -> str:
    """Generate parent chunk ID.

    Format: {doc_id}_parent_{sequence:03d}

    Args:
        doc_id: Document identifier (e.g., 'doc_001').
        sequence: Parent sequence number within document (1-indexed).

    Returns:
        Parent chunk ID string.
    """
    return f"{doc_id}_parent_{sequence:03d}"


def generate_child_id(doc_id: str, parent_sequence: int, child_sequence: int) -> str:
    """Generate child chunk ID.

    Format: {doc_id}_child_{parent_sequence:03d}_{child_sequence:03d}

    Returns:

    """

# ── Table Conversion ───────────────────────────────────────────────────────

def textract_table_to_markdown(
    table_block: Dict[str, Any],
    blocks_by_id: Dict[str, Dict[str, Any]],
) -> str:
    """Convert an AWS Textract TABLE block to pipe-delimited markdown.

    Handles merged cells by filling empty cells with placeholder.
    Detects header row from WORD block confidence or first row.

    Args:
        table_block: Textract Block with BlockType="TABLE".
        blocks_by_id: Dict mapping all block IDs to their block objects.

    Returns:
        Markdown table string.

    Example Output:
        | Income Range | Family Size 1 | Family Size 2 |
        | --- | --- | --- |
        | $0 - $20,000 | $0 | $0 |
    """
    # Extract relationships: TABLE → CELL blocks
    cells = []
    for rel in table_block.get("Relationships", []):
        if rel.get("Type") == "CHILD":
            for cell_id in rel.get("Ids", []):
                cell = blocks_by_id.get(cell_id)
                if cell and cell.get("BlockType") == "CELL":
                    cells.append(cell)

    if not cells:
        return ""

    # Build grid: row_index -> {col_index -> text}
    grid: Dict[int, Dict[int, str]] = {}
    max_row = 0
    max_col = 0

    for cell in cells:
        row = cell.get("RowIndex", 1)
        col = cell.get("ColumnIndex", 1)
        # Extract text from cell's child WORD blocks
        cell_text = ""
        for rel in cell.get("Relationships", []):
            if rel.get("Type") == "CHILD":
                for word_id in rel.get("Ids", []):
                    word = blocks_by_id.get(word_id)
                    if word and word.get("BlockType") == "WORD":
                        cell_text += word.get("Text", "") + " "
        cell_text = cell_text.strip()

        if row not in grid:
            grid[row] = {}
        grid[row][col] = cell_text
        max_row = max(max_row, row)
        max_col = max(max_col, col)

    # Build markdown rows
    rows: List[str] = []
    for r in range(1, max_row + 1):
        cols = [grid.get(r, {}).get(c, "") for c in range(1, max_col + 1)]
        rows.append("| " + " | ".join(cols) + " |")

    # Insert header separator after first row (assume first row is header)
    if len(rows) >= 1:
        separator = "| " + " | ".join(["---"] * max_col) + " |"
        rows.insert(1, separator)

    return "\n".join(rows)


# ── Markdown Cleaning ──────────────────────────────────────────────────────

def clean_markdown(text: str) -> str:
    """Normalize markdown text for consistent formatting.

    Operations:
      - Collapse multiple blank lines to single blank line
      - Strip trailing whitespace from each line
      - Ensure single newline between paragraphs
      - Standardize heading levels (no space between # and text)

    Args:
        text: Raw markdown string.

    Returns:
        Cleaned markdown string.
    """
    # Normalize line endings
    text = text.replace("\r\n", "\n").replace("\r", "\n")

    # Strip trailing whitespace per line
    lines = [line.rstrip() for line in text.split("\n")]

    # Collapse multiple blank lines
    cleaned_lines = []
    prev_blank = False
    for line in lines:
        is_blank = line == ""
        if is_blank and prev_blank:
            continue
        cleaned_lines.append(line)
        prev_blank = is_blank

    # Fix heading spacing: "#  Title" → "# Title"
    result_lines = []
    for line in cleaned_lines:
        match = re.match(r"^(#{1,6})\s+(.*)", line)
        if match:
            result_lines.append(f"{match.group(1)} {match.group(2).strip()}")
        else:
            result_lines.append(line)

    return "\n".join(result_lines)


# ── Smart Truncation ───────────────────────────────────────────────────────

def truncate_at_boundary(
    text: str,
    max_tokens: int,
    boundary: str = "sentence",
) -> str:
    """Truncate text at sentence or paragraph boundary without mid-sentence cuts.

    Args:
        text: Input text to truncate.
        max_tokens: Maximum estimated tokens allowed.
        boundary: "sentence" or "paragraph". Default is sentence.

    Returns:
        Truncated text ending at a natural boundary.

    Raises:
        ValueError: If boundary is not 'sentence' or 'paragraph'.
    """
    if boundary not in ("sentence", "paragraph"):
        raise ValueError(f"boundary must be 'sentence' or 'paragraph', got {boundary}")

    if estimate_token_count(text) <= max_tokens:
        return text

    if boundary == "sentence":
        # Split on sentence endings (.!? followed by space or newline)
        sentences = re.split(r"(?<=[.!?])\s+", text)
    else:
        sentences = text.split("\n\n")

    result = []
    current_tokens = 0
    for segment in sentences:
        seg_tokens = estimate_token_count(segment)
        if current_tokens + seg_tokens > max_tokens:
            break
        result.append(segment)
        current_tokens += seg_tokens

    return " ".join(result) if boundary == "sentence" else "\n\n".join(result)


# ── Context Extraction ─────────────────────────────────────────────────────

def extract_context_around_table(
    reading_order: List[Dict[str, Any]],
    table_block_id: str,
    max_tokens: int = 200,
) -> str:
    """Extract text blocks immediately before and after a table.

    Used for table enrichment: provides Nova Lite with surrounding context
    to generate accurate captions.

    Args:
        reading_order: List of blocks sorted by reading order.
        table_block_id: The block ID of the TABLE block.
        max_tokens: Maximum tokens of context to extract (split before/after).

    Returns:
        Concatenated surrounding text.
    """
    # Find table position in reading order
    table_idx = -1
    for i, block in enumerate(reading_order):
        if block.get("block_id") == table_block_id:
            table_idx = i
            break

    if table_idx == -1:
        return ""

    half_tokens = max_tokens // 2
    before_text = []
    after_text = []

    # Collect text before table (reverse order, stop when token limit reached)
    before_tokens = 0
    for i in range(table_idx - 1, -1, -1):
        block = reading_order[i]
        if block.get("block_type") in ("LAYOUT_TEXT", "LAYOUT_SECTION_HEADER"):
            text = block.get("text", "")
            tokens = estimate_token_count(text)
            if before_tokens + tokens > half_tokens:
                break
            before_text.insert(0, text)
            before_tokens += tokens

    # Collect text after table
    after_tokens = 0
    for i in range(table_idx + 1, len(reading_order)):
        block = reading_order[i]
        if block.get("block_type") in ("LAYOUT_TEXT", "LAYOUT_SECTION_HEADER"):
            text = block.get("text", "")
            tokens = estimate_token_count(text)
            if after_tokens + tokens > half_tokens:
                break
            after_text.append(text)
            after_tokens += tokens

    return " ".join(before_text + after_text)

"""Structured JSON logging for production observability.

All modules use get_logger() to emit JSON-formatted logs.
CloudWatch automatically parses these for metric filters and dashboards.
"""

import json
import logging
import sys
from datetime import datetime, timezone
from typing import Any, Dict, Optional


class JSONFormatter(logging.Formatter):
    """Custom formatter that outputs log records as JSON lines.

    Format:
        {
          "timestamp": "2026-08-01T10:30:00Z",
          "level": "INFO",
          "module": "textract_processor",
          "function": "analyze_document",
          "message": "Job started",
          "doc_id": "doc_001",
          "duration_ms": 4500,
          "status": "success"
        }
    """

    def format(self, record: logging.LogRecord) -> str:
        log_entry: Dict[str, Any] = {
            "timestamp": datetime.now(timezone.utc).isoformat().replace("+00:00", "Z"),
            "level": record.levelname,
            "module": record.module,
            "function": record.funcName,
            "message": record.getMessage(),
        }

        # Merge extra fields from record.__dict__ (passed via extra=)
        # Filter out standard LogRecord attributes to avoid noise
        standard_attrs = {
            "name", "msg", "args", "levelname", "levelno", "pathname",
            "filename", "module", "exc_info", "exc_text", "stack_info",
            "lineno", "funcName", "created", "msecs", "relativeCreated",
            "thread", "threadName", "processName", "process", "message",
            "asctime", "taskName",
        }
        for key, value in record.__dict__.items():
            if key not in standard_attrs and not key.startswith("_"):
                log_entry[key] = value

        # Include exception info if present
        if record.exc_info:
            log_entry["exception"] = self.formatException(record.exc_info)

        return json.dumps(log_entry, default=str)


def get_logger(name: str) -> logging.Logger:
    """Get a configured logger with JSON formatting.

    Args:
        name: Logger name (typically module name).

    Returns:
        Configured logging.Logger instance.
    """
    logger = logging.getLogger(name)

    # Avoid duplicate handlers if called multiple times
    if logger.handlers:
        return logger

    logger.setLevel(logging.INFO)

    # Stream handler → stdout (CloudWatch captures stdout in ECS/Lambda)
    handler = logging.StreamHandler(sys.stdout)
    handler.setFormatter(JSONFormatter())
    logger.addHandler(handler)

    # Prevent propagation to root logger (avoids duplicate output)
    logger.propagate = False

    return logger


def log_operation(
    logger: logging.Logger,
    operation: str,
    doc_id: Optional[str] = None,
    duration_ms: Optional[float] = None,
    status: Optional[str] = None,
    extra: Optional[Dict[str, Any]] = None,
) -> None:
    """Standardized operation log entry.

    Args:
        logger: Logger instance from get_logger().
        operation: Name of the operation (e.g., 'analyze_document').
        doc_id: Document identifier if applicable.
        duration_ms: Operation duration in milliseconds.
        status: 'success', 'failure', 'started', etc.
        extra: Additional key-value pairs to include in log.
    """
    log_data = {
        "operation": operation,
    }
    if doc_id:
        log_data["doc_id"] = doc_id
    if duration_ms is not None:
        log_data["duration_ms"] = round(duration_ms, 2)
    if status:
        log_data["status"] = status
    if extra:
        log_data.update(extra)

    logger.info(f"Operation: {operation}", extra=log_data)

"""Cross-cutting decorators for retry logic, timing, and input validation.

Apply these to pipeline functions to get automatic retries, performance logging,
and schema validation without cluttering business logic.
"""

import functools
import time
from typing import Any, Callable, Dict, Tuple, Type, Union

from .logging_config import get_logger

logger = get_logger("decorators")


# Retry Decorator 
def retry(
    max_attempts: int = 3,
    backoff_seconds: int = 2,
    exceptions: Tuple[Type[Exception], ...] = (Exception,),
) -> Callable:
    """Exponential backoff retry decorator.

    Retries the decorated function up to max_attempts with exponential backoff.
    Logs each retry attempt.

    Args:
        max_attempts: Maximum number of attempts (including first call).
        backoff_seconds: Initial backoff in seconds. Doubles each retry.
        exceptions: Tuple of exception types to catch and retry.

    Returns:
        Decorated function.

    Example:
        @retry(max_attempts=3, backoff_seconds=2, exceptions=(ClientError,))
        def call_bedrock_api(...):
            ...
    """
    def decorator(func: Callable) -> Callable:
        @functools.wraps(func)
        def wrapper(*args: Any, **kwargs: Any) -> Any:
            attempt = 1
            current_backoff = backoff_seconds

            while attempt <= max_attempts:
                try:
                    return func(*args, **kwargs)
                except exceptions as exc:
                    if attempt == max_attempts:
                        logger.error(
                            f"Function {func.__name__} failed after {max_attempts} attempts",
                            extra={
                                "function": func.__name__,
                                "attempt": attempt,
                                "max_attempts": max_attempts,
                                "error": str(exc),
                                "status": "failure",
                            },
                        )
                        raise

                    logger.warning(
                        f"Function {func.__name__} failed, retrying in {current_backoff}s",
                        extra={
                            "function": func.__name__,
                            "attempt": attempt,
                            "max_attempts": max_attempts,
                            "backoff_seconds": current_backoff,
                            "error": str(exc),
                            "status": "retry",
                        },
                    )
                    time.sleep(current_backoff)
                    current_backoff *= 2
                    attempt += 1

            # Should never reach here
            raise RuntimeError(f"Unexpected exit from retry loop in {func.__name__}")

        return wrapper
    return decorator


# ── Timing Decorator ───────────────────────────────────────────────────────

def timing(logger_name: str = "performance") -> Callable:
    """Log function execution duration in milliseconds.

    Args:
        logger_name: Logger name for timing logs. Default 'performance'.

    Returns:
        Decorated function.

    Example:
        @timing()
        def process_document(doc_id: str):
            ...
    """
    perf_logger = get_logger(logger_name)

    def decorator(func: Callable) -> Callable:
        @functools.wraps(func)
        def wrapper(*args: Any, **kwargs: Any) -> Any:
            start = time.perf_counter()
            try:
                result = func(*args, **kwargs)
                duration_ms = (time.perf_counter() - start) * 1000
                perf_logger.info(
                    f"Function {func.__name__} completed",
                    extra={
                        "function": func.__name__,
                        "duration_ms": round(duration_ms, 2),
                        "status": "success",
                    },
                )
                return result
            except Exception as exc:
                duration_ms = (time.perf_counter() - start) * 1000
                perf_logger.error(
                    f"Function {func.__name__} failed",
                    extra={
                        "function": func.__name__,
                        "duration_ms": round(duration_ms, 2),
                        "status": "failure",
                        "error": str(exc),
                    },
                )
                raise

        return wrapper
    return decorator


# ── Input Validation Decorator ────────────────────────────────────────────

def validate_input(schema: Dict[str, Union[Type, Tuple[Type, ...]]]) -> Callable:
    """Validate function keyword arguments against expected types.

    Args:
        schema: Dict mapping arg name to expected type(s).
                Example: {"doc_id": str, "s3_key": str, "max_pages": (int, type(None))}

    Returns:
        Decorated function.

    Raises:
        TypeError: If argument type does not match schema.
        ValueError: If required argument (no None in types) is missing.

    Example:
        @validate_input({"doc_id": str, "bucket": str})
        def process_document(doc_id: str, bucket: str):
            ...
    """
    def decorator(func: Callable) -> Callable:
        @functools.wraps(func)
        def wrapper(*args: Any, **kwargs: Any) -> Any:
            import inspect
            sig = inspect.signature(func)
            bound = sig.bind(*args, **kwargs)
            bound.apply_defaults()

            for arg_name, expected_types in schema.items():
                if isinstance(expected_types, type):
                    expected_types = (expected_types,)

                value = bound.arguments.get(arg_name)
                if value is None and type(None) not in expected_types:
                    raise ValueError(
                        f"Missing required argument '{arg_name}' in {func.__name__}"
                    )

                if value is not None and not isinstance(value, expected_types):
                    raise TypeError(
                        f"Argument '{arg_name}' in {func.__name__} "
                        f"expected {expected_types}, got {type(value).__name__}"
                    )

            return func(*args, **kwargs)

        return wrapper
    return decorator

#textract_processor.py
"""AWS Textract document analysis with polling and S3 persistence.

Handles:
  - analyze_document (synchronous, <10 pages) or start_document_analysis (async, >10 pages)
  - Pagination of Textract block responses
  - Polling with configurable interval and max attempts
  - Raw JSON upload to S3 for downstream parsing
"""

import json
import time
from typing import Any, Dict, List, Optional

import boto3
from botocore.exceptions import ClientError

from config.settings import load_settings
from utils.aws_clients import get_textract_client, get_s3_client
from utils.s3_utils import upload_json, check_exists
from utils.logging_config import get_logger, log_operation
from utils.decorators import retry, timing

logger = get_logger("textract_processor")


class TextractProcessor:
    """Orchestrates Textract document analysis and S3 persistence."""

    def __init__(self):
        self.settings = load_settings()
        self.textract = get_textract_client()
        self.s3 = get_s3_client()
        self.bucket = self.settings.S3_BUCKET
        self.features = list(self.settings.textract.FEATURES)
        self.poll_interval = self.settings.textract.POLLING_INTERVAL_SECONDS
        self.max_polls = self.settings.textract.MAX_POLLING_ATTEMPTS

    # ── Public API ─────────────────────────────────────────────────────────

    @timing()
    def process_document(self, doc_id: str, s3_input_key: str) -> str:
        """Analyze a document via Textract and store raw output in S3.

        Args:
            doc_id: Document identifier (e.g., 'doc_001').
            s3_input_key: S3 key of the source file (e.g., 'ecmc_org/source_files/pdfs/doc_001.pdf').

        Returns:
            S3 key where raw Textract JSON was saved.

        Raises:
            RuntimeError: If Textract job fails or polling exceeds max attempts.
            ClientError: On AWS API errors.
        """
        output_key = self.settings.s3_paths.textract_doc_path(doc_id) + "layout.json"

        # Skip if already processed
        if check_exists(self.bucket, output_key):
            logger.info(
                "Textract output already exists, skipping",
                extra={"doc_id": doc_id, "s3_key": output_key},
            )
            return output_key

        logger.info(
            "Starting Textract analysis",
            extra={"doc_id": doc_id, "s3_input_key": s3_input_key, "features": self.features},
        )

        # Determine sync vs async based on file size (heuristic: >10 pages or >5MB → async)
        file_size = self._get_s3_file_size(s3_input_key)
        if file_size and file_size > 5 * 1024 * 1024:
            raw_response = self._analyze_async(s3_input_key)
        else:
            raw_response = self._analyze_sync(s3_input_key)

        # Upload raw response to S3
        upload_json(self.bucket, output_key, raw_response)

        logger.info(
            "Textract analysis complete",
            extra={
                "doc_id": doc_id,
                "s3_output_key": output_key,
                "block_count": len(raw_response.get("Blocks", [])),
            },
        )
        return output_key

    # ── Internal: Synchronous Analysis ───────────────────────────────────

    @retry(max_attempts=3, backoff_seconds=2, exceptions=(ClientError,))
    def _analyze_sync(self, s3_input_key: str) -> Dict[str, Any]:
        """Call analyze_document (synchronous, best for <10 pages)."""
        response = self.textract.analyze_document(
            Document={
                "S3Object": {"Bucket": self.bucket, "Name": s3_input_key}
            },
            FeatureTypes=self.features,
        )
        return self._flatten_response(response)

    # ── Internal: Asynchronous Analysis ────────────────────────────────────

    @retry(max_attempts=3, backoff_seconds=2, exceptions=(ClientError,))
    def _analyze_async(self, s3_input_key: str) -> Dict[str, Any]:
        """Call start_document_analysis + poll (best for >10 pages)."""
        start_resp = self.textract.start_document_analysis(
            DocumentLocation={
                "S3Object": {"Bucket": self.bucket, "Name": s3_input_key}
            },
            FeatureTypes=self.features,
        )
        job_id = start_resp["JobId"]
        logger.info(
            "Async Textract job started",
            extra={"job_id": job_id, "s3_input_key": s3_input_key},
        )

        # Poll until complete
        for attempt in range(1, self.max_polls + 1):
            status_resp = self.textract.get_document_analysis(JobId=job_id)
            status = status_resp["JobStatus"]

            if status == "SUCCEEDED":
                logger.info(
                    "Async Textract job succeeded",
                    extra={"job_id": job_id, "attempts": attempt},
                )
                return self._paginate_async_results(job_id)

            if status == "FAILED":
                error_msg = status_resp.get("StatusMessage", "Unknown error")
                logger.error(
                    "Async Textract job failed",
                    extra={"job_id": job_id, "error": error_msg},
                )
                raise RuntimeError(f"Textract job {job_id} failed: {error_msg}")

            logger.info(
                "Textract job in progress, polling...",
                extra={"job_id": job_id, "attempt": attempt, "status": status},
            )
            time.sleep(self.poll_interval)

        raise RuntimeError(
            f"Textract job {job_id} did not complete within "
            f"{self.max_polls * self.poll_interval} seconds"
        )

    # ── Internal: Pagination ───────────────────────────────────────────────

    def _paginate_async_results(self, job_id: str) -> Dict[str, Any]:
        """Paginate through all blocks of an async Textract job."""
        all_blocks: List[Dict[str, Any]] = []
        next_token: Optional[str] = None
        page_count = 0

        while True:
            kwargs = {"JobId": job_id}
            if next_token:
                kwargs["NextToken"] = next_token

            page = self.textract.get_document_analysis(**kwargs)
            page_count += 1
            blocks = page.get("Blocks", [])
            all_blocks.extend(blocks)

            next_token = page.get("NextToken")
            if not next_token:
                break

        logger.info(
            "Paginated Textract results",
            extra={"job_id": job_id, "pages": page_count, "total_blocks": len(all_blocks)},
        )

        # Return a structure compatible with analyze_document response
        return {
            "DocumentMetadata": page.get("DocumentMetadata", {}),
            "Blocks": all_blocks,
            "JobId": job_id,
        }

    # ── Internal: Helpers ──────────────────────────────────────────────────

    def _get_s3_file_size(self, key: str) -> Optional[int]:
        """Return S3 object size in bytes, or None if not found."""
        try:
            resp = self.s3.head_object(Bucket=self.bucket, Key=key)
            return resp["ContentLength"]
        except ClientError:
            return None

    def _flatten_response(self, response: Dict[str, Any]) -> Dict[str, Any]:
        """Ensure consistent structure between sync and async responses."""
        return {
            "DocumentMetadata": response.get("DocumentMetadata", {}),
            "Blocks": response.get("Blocks", []),
        }

def process_document(doc_id: str, s3_input_key: str) -> str:
    """One-shot function to process a single document.

    Args:
        doc_id: Document identifier.
        s3_input_key: S3 key of source PDF/HTML.

    Returns:
        S3 key of stored Textract output.
    """
    processor = TextractProcessor()
    return processor.process_document(doc_id, s3_input_key)

"""Parse Textract raw JSON into structured reading order and document metadata.

Enforces sequential document flow:

    def __init__(self):
        self.settings = load_settings()
        self.bucket = self.settings.S3_BUCKET

    # ── Public API ─────────────────────────────────────────────────────────

    @timing()
    def parse_document(self, doc_id: str) -> Tuple[str, str]:
        """Parse Textract output into reading_order.json and document_metadata.json.

        Args:
            doc_id: Document identifier (e.g., 'doc_001').

        Returns:
            Tuple of (reading_order_s3_key, metadata_s3_key).
        """
        # Read raw Textract JSON from S3
        textract_key = self.settings.s3_paths.textract_doc_path(doc_id) + "layout.json"
        raw_data = download_json(self.bucket, textract_key)

        if not raw_data or "Blocks" not in raw_data:
            raise ValueError(f"Invalid Textract data for {doc_id}: missing 'Blocks'")

        blocks = raw_data["Blocks"]
        doc_meta = raw_data.get("DocumentMetadata", {})

        # Build block lookup for relationship resolution
        blocks_by_id = {b["Id"]: b for b in blocks}

        # Extract layout blocks (text, headers, tables, figures)
        layout_blocks = self._extract_layout_blocks(blocks, blocks_by_id)

        # Sort by page → top → left
        sorted_blocks = self._sort_by_reading_order(layout_blocks)

        # Assign sequential reading_order integers
        for i, block in enumerate(sorted_blocks, start=1):
            block["reading_order"] = i

        # Identify section boundaries
        sections = self._identify_sections(sorted_blocks)

        # Build outputs
        reading_order = {
            "doc_id": doc_id,
            "total_pages": doc_meta.get("Pages", 0),
            "total_blocks": len(sorted_blocks),
            "blocks": sorted_blocks,
            "sections": sections,
        }

        metadata = {
            "doc_id": doc_id,
            "total_pages": doc_meta.get("Pages", 0),
            "total_blocks": len(blocks),
            "layout_blocks": len(sorted_blocks),
            "sections_count": len(sections),
            "source_s3_key": textract_key,
        }

        # Upload to S3
        reading_key = self.settings.s3_paths.json_doc_path(doc_id) + "reading_order.json"
        metadata_key = self.settings.s3_paths.json_doc_path(doc_id) + "document_metadata.json"

        upload_json(self.bucket, reading_key, reading_order)
        upload_json(self.bucket, metadata_key, metadata)

        logger.info(
            "Layout parsed",
            extra={
                "doc_id": doc_id,
                "blocks": len(sorted_blocks),
                "sections": len(sections),
                "pages": doc_meta.get("Pages", 0),
            },
        )

        return reading_key, metadata_key

    # ── Internal: Block Extraction ─────────────────────────────────────────

    def _extract_layout_blocks(
        self,
        blocks: List[Dict[str, Any]],
        blocks_by_id: Dict[str, Dict[str, Any]],
    ) -> List[Dict[str, Any]]:
        """Extract relevant layout blocks with resolved text and geometry.

        Includes: LAYOUT_TEXT, LAYOUT_TITLE, LAYOUT_SECTION_HEADER,
                  LAYOUT_TABLE, LAYOUT_FIGURE, LAYOUT_LIST, LAYOUT_PAGE_NUMBER
        """
        layout_types = {
            "LAYOUT_TEXT",
            "LAYOUT_TITLE",
            "LAYOUT_SECTION_HEADER",
            "LAYOUT_TABLE",
            "LAYOUT_FIGURE",
            "LAYOUT_LIST",
            "LAYOUT_PAGE_NUMBER",
            "LAYOUT_FOOTER",
            "LAYOUT_HEADER",
        }

        result = []
        for block in blocks:
            block_type = block.get("BlockType", "")
            if block_type not in layout_types:
                continue

            # Extract geometry
            geometry = block.get("Geometry", {})
            bbox = geometry.get("BoundingBox", {})

            # Resolve text from child WORD blocks
            text = self._resolve_text(block, blocks_by_id)

            # Extract page number from Page field
            page = block.get("Page", 1)

            result.append({
                "block_id": block["Id"],
                "block_type": block_type,
                "page_number": page,
                "text": text,
                "geometry": {
                    "top": bbox.get("Top", 0.0),
                    "left": bbox.get("Left", 0.0),
                    "width": bbox.get("Width", 0.0),
                    "height": bbox.get("Height", 0.0),
                },
                "confidence": block.get("Confidence", 0.0),
            })

        return result

    def _resolve_text(
        self,
        block: Dict[str, Any],
        blocks_by_id: Dict[str, Dict[str, Any]],
    ) -> str:
        """Resolve full text from a block by concatenating child WORD blocks."""
        text_parts = []
        for rel in block.get("Relationships", []):
            if rel.get("Type") == "CHILD":
                for child_id in rel.get("Ids", []):
                    child = blocks_by_id.get(child_id)
                    if child and child.get("BlockType") == "WORD":
                        text_parts.append(child.get("Text", ""))
        return " ".join(text_parts).strip()

    # ── Internal: Sorting ──────────────────────────────────────────────────

    def _sort_by_reading_order(
        self,
        blocks: List[Dict[str, Any]],
    ) -> List[Dict[str, Any]]:
        """Sort blocks by page number, then vertical position, then horizontal."""
        return sorted(
            blocks,
            key=lambda b: (
                b["page_number"],
                b["geometry"]["top"],
                b["geometry"]["left"],
            ),
        )

    # ── Internal: Section Identification ───────────────────────────────────

    def _identify_sections(
        self,
        sorted_blocks: List[Dict[str, Any]],
    ) -> List[Dict[str, Any]]:
        """Identify document sections bounded by LAYOUT_TITLE or LAYOUT_SECTION_HEADER.

        Returns list of sections with start_block_id, end_block_id, heading, page_range.
        """
        sections = []
        current_section = None

        for block in sorted_blocks:
            block_type = block["block_type"]
            block_id = block["block_id"]
            page = block["page_number"]
            text = block["text"]

            # Start new section at title or section header
            if block_type in ("LAYOUT_TITLE", "LAYOUT_SECTION_HEADER"):
                # Close previous section
                if current_section:
                    current_section["end_block_id"] = sorted_blocks[
                        sorted_blocks.index(block) - 1
                    ]["block_id"]
                    sections.append(current_section)

                current_section = {
                    "section_id": f"section_{len(sections) + 1:03d}",
                    "heading": text,
                    "heading_block_id": block_id,
                    "start_block_id": block_id,
                    "end_block_id": None,
                    "page_start": page,
                    "page_end": page,
                }
            elif current_section:
                # Update page range
                current_section["page_end"] = max(current_section["page_end"], page)

        # Close final section
        if current_section and sorted_blocks:
            current_section["end_block_id"] = sorted_blocks[-1]["block_id"]
            sections.append(current_section)

        return sections


# ── Convenience Function ───────────────────────────────────────────────────

def parse_layout(doc_id: str) -> Tuple[str, str]:
    """One-shot function to parse layout for a single document.

    Args:
        doc_id: Document identifier.

    Returns:
        Tuple of (reading_order_s3_key, metadata_s3_key).
    """
    parser = LayoutParser()
    return parser.parse_document(doc_id)

    def extract_and_enrich(self, doc_id: str) -> Tuple[str, str]:

        Args:

        Returns:

        """
        # Read inputs
        textract_key = self.settings.s3_paths.textract_doc_path(doc_id) + "layout.json"
        reading_key = self.settings.s3_paths.json_doc_path(doc_id) + "reading_order.json"

        textract_data = download_json(self.bucket, textract_key)
        reading_data = download_json(self.bucket, reading_key)

        if not textract_data or "Blocks" not in textract_data:

        blocks = textract_data["Blocks"]
        blocks_by_id = {b["Id"]: b for b in blocks}
        reading_blocks = reading_data.get("blocks", [])

        # Find all TABLE blocks
        table_blocks = [b for b in blocks if b.get("BlockType") == "TABLE"]

            # Still write empty files for consistency
            tables_output = {"doc_id": doc_id, "tables": []}
            enrichments_output = {"doc_id": doc_id, "enrichments": []}
        else:
            tables_output = {"doc_id": doc_id, "tables": []}
            enrichments_output = {"doc_id": doc_id, "enrichments": []}

            for i, table_block in enumerate(table_blocks, start=1):
                table_id = f"{doc_id}_table_{i:03d}"

                # Convert to markdown
                markdown = textract_table_to_markdown(table_block, blocks_by_id)

                # Extract context from reading order
                context = self._extract_table_context(
                    table_block["Id"], reading_blocks, blocks_by_id
                )

                # Get page number from table block
                page = table_block.get("Page", 1)

                # Build table record
                table_record = {
                    "table_id": table_id,
                    "textract_block_id": table_block["Id"],
                    "page_number": page,
                    "markdown": markdown,
                    "row_count": self._count_rows(table_block),
                    "column_count": self._count_columns(table_block),

                caption = self._enrich_table(table_record)

                enrichment_record = {
                    "table_id": table_id,

                }
                enrichments_output["enrichments"].append(enrichment_record)

                logger.info(

                    extra={

        tables_key = self.settings.s3_paths.json_doc_path(doc_id) + "tables.json"
        enrichments_key = (

        )

        logger.info(
            "Table extraction complete",
            extra={

                "table_count": len(tables_output["tables"]),
            },
        )

        return tables_key, enrichments_key

    # ── Internal: Context Extraction ───────────────────────────────────────

    def _extract_table_context(
        self,
        table_block_id: str,
        reading_blocks: List[Dict[str, Any]],
        blocks_by_id: Dict[str, Dict[str, Any]],
    ) -> str:
        """Extract text blocks immediately before and after a table.

        Uses reading_order blocks to find position, then collects
        LAYOUT_TEXT and LAYOUT_SECTION_HEADER blocks up to token limit.
        """
        # Find table position in reading order
        table_idx = -1
        for i, block in enumerate(reading_blocks):
            if block.get("block_id") == table_block_id:
                table_idx = i
                break

        if table_idx == -1:
            return ""

        max_tokens = self.settings.chunking.TABLE_CONTEXT_TOKENS
        half_tokens = max_tokens // 2

        before_parts = []
        after_parts = []
        before_tokens = 0
        after_tokens = 0

        # Collect before table
        for i in range(table_idx - 1, -1, -1):
            block = reading_blocks[i]
            if block["block_type"] in ("LAYOUT_TEXT", "LAYOUT_SECTION_HEADER"):
                text = block.get("text", "")
                tokens = estimate_token_count(text)
                if before_tokens + tokens > half_tokens:
                    break
                before_parts.insert(0, text)
                before_tokens += tokens

        # Collect after table
        for i in range(table_idx + 1, len(reading_blocks)):
            block = reading_blocks[i]
            if block["block_type"] in ("LAYOUT_TEXT", "LAYOUT_SECTION_HEADER"):
                text = block.get("text", "")
                tokens = estimate_token_count(text)
                if after_tokens + tokens > half_tokens:
                    break
                after_parts.append(text)
                after_tokens += tokens

        return " ".join(before_parts + after_parts)

    # ── Internal: Nova Lite Enrichment ─────────────────────────────────────

    @retry(max_attempts=3, backoff_seconds=2, exceptions=(Exception,))
    def _enrich_table(self, table_record: Dict[str, Any]) -> str:
        """Send table markdown + context to Nova Lite for semantic caption.

        Args:
            table_record: Dict with 'markdown', 'context', 'table_id'.

        Returns:
            Nova Lite generated caption string.
        """
        prompt = self._build_enrichment_prompt(table_record)

        response = self.bedrock.converse(
            modelId=self.chat_model,
            messages=[
                {
                    "role": "user",
                    "content": [{"text": prompt}],
                }
            ],
            inferenceConfig={
                "maxTokens": 500,
                "temperature": 0.3,
            },
        )

        output = response["output"]["message"]["content"][0]["text"]
        return output.strip()

    def _build_enrichment_prompt(self, table_record: Dict[str, Any]) -> str:
        """Build the prompt for table enrichment."""
        markdown = table_record.get("markdown", "")
        context = table_record.get("context", "")

        prompt_parts = [self.settings.prompts.TABLE_ENRICHMENT]

        if context:
            prompt_parts.append(f"\n\nContext from the document:\n{context}")

        prompt_parts.append(f"\n\nTable (in markdown format):\n\n{markdown}")

        return "\n".join(prompt_parts)

"""Extract figure images from documents and generate captions via Nova Lite multimodal.

Pipeline:

  2. Download original source file from S3

Nova Lite prompt (from config.paths.PROMPTS.IMAGE_CAPTION):
  "Describe this figure or chart and its key data points in 2-3 sentences."

Note: For PDF source files, image cropping requires additional libraries

        """
        # Read Textract output
        textract_key = self.settings.s3_paths.textract_doc_path(doc_id) + "layout.json"
        textract_data = download_json(self.bucket, textract_key)

        if not textract_data or "Blocks" not in textract_data:

        blocks = textract_data["Blocks"]

        # Download source file to local temp
        source_bytes = download_bytes(self.bucket, source_s3_key)

        captions_output = {"doc_id": doc_id, "figures": []}

                    "Unsupported source type for image cropping",
                    extra={"doc_id": doc_id, "figure_id": figure_id, "ext": file_ext},

                        "figure_id": figure_id,

                    },
                )
            else:
                figure_record["caption"] = (
                    f"[Figure on page {page} — image extraction not available]"
                )
                figure_record["s3_image_key"] = None
                logger.warning(
                    "Could not extract figure image",
                    extra={"doc_id": doc_id, "figure_id": figure_id, "page": page},
                )

    # ── Internal: Image Cropping ───────────────────────────────────────────

    def _crop_from_image(self, image_bytes: bytes, bbox: Dict[str, float]) -> Optional[bytes]:
        """Crop a region from an image using PIL.

        Args:
            image_bytes: Raw image bytes (PNG/JPG).
            bbox: Textract BoundingBox dict with Top, Left, Width, Height (0-1 normalized).

        Returns:
            Cropped PNG bytes, or None if PIL unavailable or error.
        """
        try:
            from PIL import Image
        except ImportError:
            logger.warning("PIL not available, skipping image crop")
            return None

        try:
            img = Image.open(io.BytesIO(image_bytes))
            width, height = img.size

            # Convert normalized bbox to pixel coordinates
            left = int(bbox.get("Left", 0) * width)
            top = int(bbox.get("Top", 0) * height)
            right = int((bbox.get("Left", 0) + bbox.get("Width", 0)) * width)
            bottom = int((bbox.get("Top", 0) + bbox.get("Height", 0)) * height)

            # Ensure valid coordinates
            left = max(0, left)
            top = max(0, top)
            right = min(width, right)
            bottom = min(height, bottom)

            if right <= left or bottom <= top:
                logger.warning("Invalid crop coordinates", extra={"bbox": bbox})
                return None

            cropped = img.crop((left, top, right, bottom))

            output = io.BytesIO()
            cropped.save(output, format="PNG")
            return output.getvalue()

        except Exception as exc:
            logger.warning("Image crop failed", extra={"error": str(exc)})
            return None

    def _crop_from_pdf(
        self,
        pdf_bytes: bytes,
        page_number: int,
        bbox: Dict[str, float],
    ) -> Optional[bytes]:
        """Crop a region from a PDF page.

        Requires PyMuPDF (fitz). If unavailable, returns None.

        Args:
            pdf_bytes: Raw PDF bytes.
            page_number: 1-based page number.
            bbox: Textract BoundingBox dict (0-1 normalized).

        Returns:
            Cropped PNG bytes, or None if library unavailable.
        """
        try:
            import fitz  # PyMuPDF
        except ImportError:
            logger.warning("PyMuPDF not available, skipping PDF image crop")
            return None

        try:
            doc = fitz.open(stream=pdf_bytes, filetype="pdf")
            if page_number < 1 or page_number > len(doc):
                logger.warning(
                    "Page number out of range",
                    extra={"page": page_number, "total_pages": len(doc)},
                )
                return None

            page = doc[page_number - 1]  # 0-based index
            rect = page.rect

            # Convert normalized bbox to page coordinates
            x0 = bbox.get("Left", 0) * rect.width
            y0 = bbox.get("Top", 0) * rect.height
            x1 = (bbox.get("Left", 0) + bbox.get("Width", 0)) * rect.width
            y1 = (bbox.get("Top", 0) + bbox.get("Height", 0)) * rect.height

            clip_rect = fitz.Rect(x0, y0, x1, y1)
            pix = page.get_pixmap(clip=clip_rect, dpi=150)

            return pix.tobytes("png")

        except Exception as exc:
            logger.warning("PDF image crop failed", extra={"error": str(exc)})
            return None

    # ── Internal: Nova Lite Multimodal Caption ───────────────────────────

    @retry(max_attempts=3, backoff_seconds=2, exceptions=(Exception,))
    def _caption_image(self, image_bytes: bytes) -> str:
        """Send image to Nova Lite multimodal API for caption.

        Args:
            image_bytes: PNG image bytes.

        Returns:
            Generated caption string.
        """
        # Encode image to base64
        b64_image = base64.b64encode(image_bytes).decode("utf-8")

        response = self.bedrock.converse(
            modelId=self.chat_model,
            messages=[
                {
                    "role": "user",
                    "content": [
                        {
                            "image": {
                                "format": "png",
                                "source": {"bytes": b64_image},
                            }
                        },
                        {"text": self.prompt},
                    ],
                }
            ],
            inferenceConfig={
                "maxTokens": 300,
                "temperature": 0.3,
            },
        )

        return response["output"]["message"]["content"][0]["text"].strip()
